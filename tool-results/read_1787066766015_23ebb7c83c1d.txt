     1→/**
     2→ * SaaS Subscription service — real payment lifecycle for tenant billing.
     3→ *
     4→ * Phase 2B.3: Implements the SaaS monetization loop.
     5→ * Phase 2B.3.1: Financial lifecycle convergence — ledger before activation,
     6→ *   reconciliation worker, no silent catches, durable idempotency.
     7→ *
     8→ * Payment model: INVOICE-STYLE RENEWAL (not automatic recurring charge).
     9→ * Each billing period, a new payment request (PaymentIntent) is created.
    10→ * The tenant must complete the payment (via the API or webhook). This is
    11→ * NOT automatic recurring billing — the platform does not store payment
    12→ * methods or charge saved cards automatically.
    13→ *
    14→ * To support automatic recurring billing in the future, the PaymentProvider
    15→ * abstraction would need to be extended with:
    16→ *   - createCustomer()
    17→ *   - savePaymentMethod()
    18→ *   - createRecurringCharge() / createSubscription()
    19→ * The current model is honest about this limitation.
    20→ *
    21→ * Financial lifecycle:
    22→ *   1. createSubscriptionIntent (tenant chooses plan → payment intent → pending invoice)
    23→ *   2. confirmSubscriptionPayment (server verifies payment → posts ledger → activates)
    24→ *   3. If ledger fails → invoice = reconciliation_required, subscription NOT activated
    25→ *   4. processDueSaasFinancialReconciliation (cron: retries ledger for reconciliation_required)
    26→ *   5. renewSubscription (creates new invoice + payment request for next period)
    27→ *   6. cancelSubscription (ends at period end, no more renewals)
    28→ *   7. processDueSaasRenewals (cron: renews subscriptions whose period has ended)
    29→ *
    30→ * All financial events post to the canonical double-entry ledger via
    31→ * ledgerSaasSubscriptionPayment. SaaS revenue (4200) is separated from
    32→ * connectivity sales revenue (4000) and platform fee revenue (4100).
    33→ *
    34→ * Idempotency: every operation uses a durable idempotencyKey. Subscription
    35→ * creation checks for an existing invoice by key before creating a new provider
    36→ * intent. Renewal uses a key derived from subscriptionId + periodEnd. Duplicate
    37→ * webhook deliveries or cron retries cannot double-charge.
    38→ */

    51→/**
    52→ * Phase 2B.3.15 P1-3 / Phase 2B.3.16: Canonical billing-interval derivation.
    53→ *
    54→ * "monthly" and "yearly" are CALENDAR intervals, not fixed day counts.
    55→ * Jan 31 + 1 month = Feb 28 (or 29 in a leap year), NOT Jan 31 + 30 days
    56→ * and NOT Mar 3 (which is what JavaScript's naive setMonth produces due to
    57→ * day overflow).
    58→ *
    59→ * Phase 2B.3.16: Billing periods are defined in UTC, not local server time.
    60→ * All Date arithmetic uses getUTCDate/getUTCMonth/setUTCMonth/setUTCFullYear
    61→ * to ensure the billing period is independent of the server's timezone
    62→ * configuration. This is critical because:
    63→ *   - paidAt is stored as a UTC timestamp in PostgreSQL
    64→ *   - periodStart/periodEnd are stored as UTC timestamps
    65→ *   - DST transitions in the server timezone must not shift the billing period
    66→ *
    67→ * This helper uses UTC calendar arithmetic WITH end-of-month clamping:
    68→ *   - If the target month doesn't have the original day (e.g. Feb 31),
    69→ *     clamp to the last day of the target month (Feb 28/29).
    70→ *   - This matches the billing-period semantics used by Stripe and other
    71→ *     billing systems: "charge on the same day each month, or the last day
    72→ *     if that day doesn't exist."
    73→ *
    74→ * The billing period for a subscription is defined as:
    75→ *   periodStart = paidAt (the financial event, UTC)
    76→ *   periodEnd   = addBillingInterval(periodStart, billingCycle)
    77→ *
    78→ * Validation of an existing period MUST re-derive the expected periodEnd
    79→ * from the recorded periodStart using this same function, then compare
    80→ * for exact equality — NOT use a duration tolerance window.
    81→ */
    82→function addBillingInterval(start: Date, billingCycle: string): Date {
    83→  const end = new Date(start.getTime());
    84→  if (billingCycle === "yearly") {
    85→    end.setUTCFullYear(end.getUTCFullYear() + 1);
    86→  } else {
    87→    // monthly (default) — with end-of-month clamping in UTC.
    88→    // JavaScript's Date.setUTCMonth overflows: Jan 31 + setUTCMonth(1) → Mar 3,
    89→    // because it tries to create "February 31" which doesn't exist.
    90→    // We detect the overflow and clamp to the last day of the target month.
    91→    const originalDay = end.getUTCDate();
    92→    end.setUTCMonth(end.getUTCMonth() + 1);
    93→    if (end.getUTCDate() < originalDay) {
    94→      // Day overflowed (e.g., Feb 31 → Mar 3). Clamp to the last day
    95→      // of the target month by setting day = 0 (last day of previous month).
    96→      end.setUTCDate(0);
    97→    }
    98→  }
    99→  return end;
   100→}
   101→
   102→/**
   103→ * Phase 2B.3.15 P1-3: Validate that (periodStart, periodEnd) is the canonical
   104→ * billing interval for the given cycle. Returns true if periodEnd ===
   105→ * addBillingInterval(periodStart, billingCycle).
   106→ */
   107→function isCanonicalBillingInterval(periodStart: Date, periodEnd: Date, billingCycle: string): boolean {
   108→  const expected = addBillingInterval(periodStart, billingCycle);
   109→  return periodEnd.getTime() === expected.getTime();
   110→}

   112→/**
   113→ * Create a subscription intent for a tenant choosing a SaaS plan.
   114→ * Creates a payment provider intent. The subscription is NOT activated until
   115→ * the payment is verified.
   116→ */
   117→export async function createSubscriptionIntent(input: {
   118→  tenantId: string;
   119→  userId: string;
   120→  planName: string;
   121→  billingCycle?: "monthly" | "yearly";
   122→  idempotencyKey: string;
   123→}): Promise<{ subscriptionId: string; providerReference: string; status: string; clientSecret?: string }> {
   124→  const { tenantId, userId, planName } = input;
   125→  const billingCycle = input.billingCycle ?? "monthly";
   126→
   127→  // Look up the plan
   128→  const plan = await db.saaasPlan.findUnique({ where: { name: planName } });
   129→  if (!plan) {
   130→    throw new AppError("not_found", "SaaS plan not found", 404, `Plan "${planName}" does not exist.`);
   131→  }
   132→  if (plan.status !== "active") {
   133→    throw new AppError("conflict", "Plan not available", 409, `Plan "${planName}" is not active.`);
   134→  }
   135→  if (plan.monthlyPriceMinor <= 0) {
   136→    throw new AppError("validation", "Cannot subscribe to free plan", 400, "The free plan does not require a subscription. It is the default when no paid subscription exists.");
   137→  }
   138→
   139→  const provider = getPaymentProvider();
   140→
   141→  // Phase 2B.3.1: Idempotency — check for an existing invoice with this idempotencyKey.
   142→  // If found, return the existing subscription intent without creating a new provider operation.
   143→  const existingInvoice = await db.tenantInvoice.findUnique({
   144→    where: { idempotencyKey: input.idempotencyKey },
   145→    include: { subscription: true },
   146→  });
   147→  if (existingInvoice && existingInvoice.subscription) {
   148→    logger.info("saas.subscribe_idempotent_replay", { idempotencyKey: input.idempotencyKey, invoiceId: existingInvoice.id });
   149→    return {
   150→      subscriptionId: existingInvoice.subscription.id,
   151→      providerReference: existingInvoice.providerReference ?? "",
   152→      status: existingInvoice.subscription.status,
   153→    };
   154→  }
   155→
   156→  // Check for existing subscription
   157→  const existing = await db.tenantSubscription.findUnique({ where: { tenantId } });
   158→  if (existing && existing.status === "active" && existing.saaasPlanId === plan.id) {
   159→    throw new AppError("conflict", "Already subscribed", 409, "This tenant is already on this plan.");
   160→  }
   161→
   162→  // Calculate the amount based on billing cycle
   163→  const amountMinor = billingCycle === "yearly" ? plan.monthlyPriceMinor * 12 : plan.monthlyPriceMinor;
   164→
   165→  // Create the payment intent with the provider
   166→  const intent = await provider.createPaymentIntent({
   167→    amountMinor,
   168→    currency: plan.currency as Currency,
   169→    description: `SaaS subscription: ${plan.displayName} (${billingCycle})`,
   170→    idempotencyKey: input.idempotencyKey,
   171→    metadata: { tenantId, planName, type: "saas_subscription" },
   172→  });
   173→
   174→  // Phase 2B.3.3 P0-1: Use PENDING_PAYMENT (not "trialing") for unpaid subscriptions.
   175→  // TRIALING is reserved for actual free trials.
   176→  // PENDING_PAYMENT subscriptions are NOT eligible for renewal processing.
   177→  // The period is NOT set until payment is confirmed — the billing clock
   178→  // starts when the customer pays, not when they express intent.
   179→  const subscription = await db.tenantSubscription.upsert({
   180→    where: { tenantId },
   181→    create: {
   182→      tenantId,
   183→      saaasPlanId: plan.id,
   184→      status: "pending_payment",
   185→      billingCycle,
   186→      currentPeriodEnd: new Date(0), // epoch — not a real period until payment
   187→      paymentProvider: provider.id,
   188→      providerReference: intent.providerReference,
   189→    },
   190→    update: {
   191→      saaasPlanId: plan.id,
   192→      status: "pending_payment",
   193→      billingCycle,
   194→      currentPeriodEnd: new Date(0),
   195→      paymentProvider: provider.id,
   196→      providerReference: intent.providerReference,
   197→      cancelledAt: null,
   198→      cancelReason: null,
   199→    },
   200→  });
   201→
   202→  // Create a pending invoice
   203→  await db.tenantInvoice.create({
   204→    data: {
   205→      tenantId,
   206→      subscriptionId: subscription.id,
   207→      saaasPlanName: plan.name,
   208→      amountMinor,
   209→      currency: plan.currency,
   210→      billingCycle,
   211→      // Phase 2B.3.4: No fake billing period for unpaid invoices.
   212→      // periodStart/periodEnd are null until payment is confirmed.
   213→      periodStart: null,
   214→      periodEnd: null,
   215→      status: "pending",
   216→      paymentProvider: provider.id,
   217→      providerReference: intent.providerReference,
   218→      idempotencyKey: input.idempotencyKey,
   219→    },
   220→  });
   221→
   222→  await audit({
   223→    tenantId,
   224→    userId,
   225→    action: "saas.subscription_intent_created",
   226→    entity: "tenant_subscription",
   227→    entityId: subscription.id,
   228→    detail: { planName, billingCycle, amountMinor, providerReference: intent.providerReference },
   229→  });
   230→
   231→  logger.info("saas.subscription_intent_created", { tenantId, subscriptionId: subscription.id, planName, providerReference: intent.providerReference });
   232→
   233→  return {
   234→    subscriptionId: subscription.id,
   235→    providerReference: intent.providerReference,
   236→    status: subscription.status,
   237→    clientSecret: intent.clientSecret,
   238→  };
   239→}

   241→/**
   242→ * Confirm a subscription payment (server-side verification).
   243→ * If the payment succeeded, activates the subscription + posts the ledger entry.
   244→ * Idempotent: if already active, returns the existing state.
   245→ */
   246→export async function confirmSubscriptionPayment(input: {
   247→  tenantId: string;
   248→  userId: string;
   249→  subscriptionId: string;
   250→}): Promise<{ status: string }> {
   251→  const subscription = await db.tenantSubscription.findUnique({
   252→    where: { id: input.subscriptionId },
   253→  });
   254→  if (!subscription) {
   255→    throw new AppError("not_found", "Subscription not found", 404, "Subscription not found.");
   256→  }
   257→  if (subscription.tenantId !== input.tenantId) {
   258→    throw new AppError("authorization", "Cross-tenant access denied", 403, "This subscription belongs to a different tenant.");
   259→  }
   260→
   261→  // Idempotent: if already active, return
   262→  if (subscription.status === "active") {
   263→    return { status: "active" };
   264→  }
   265→
   266→  // Find the pending invoice
   267→  const invoice = await db.tenantInvoice.findFirst({
   268→    where: { subscriptionId: subscription.id, status: "pending" },
   269→    orderBy: { createdAt: "desc" },
   270→  });
   271→  if (!invoice) {
   272→    throw new AppError("not_found", "No pending invoice", 404, "No pending invoice found for this subscription.");
   273→  }
   274→
   275→  // Phase 2B.3.3 P0-2: Resolve the provider from the invoice's paymentProvider field,
   276→  // NOT from the global getPaymentProvider(). This ensures an invoice created under
   277→  // Provider A continues to use Provider A even if the platform's default changes.
   278→  const provider = getPaymentProviderByKey(invoice.paymentProvider || "mock");
   279→
   280→  // For the mock provider, simulate client-side confirmation
   281→  if (provider.isMock && invoice.providerReference) {
   282→    const { mockPaymentProvider } = await import("@/lib/payments");
   283→    mockPaymentProvider.confirmIntent(invoice.providerReference);
   284→  }
   285→
   286→  const verification = await provider.verifyPayment({
   287→    providerReference: invoice.providerReference!,
   288→    idempotencyKey: invoice.idempotencyKey,
   289→  });
   290→
   291→  if (verification.status === "failed") {
   292→    // Phase 2B.3.14 P0-1: GUARDED transition — never overwrite a concurrently-paid invoice.
   293→    // The provider observation may be stale: a webhook may have finalized the invoice
   294→    // while verifyPayment() was in flight. The mutation must be PostgreSQL-atomic:
   295→    // only transition if the invoice is still in a pending state.
   296→    const guarded = await db.tenantInvoice.updateMany({
   297→      where: { id: invoice.id, status: "pending" },
   298→      data: { status: "failed", failureReason: "Payment verification failed" },
   299→    });
   300→    if (guarded.count === 0) {
   301→      // The invoice was no longer pending — a concurrent worker advanced it.
   302→      // Re-read to understand the current state and do NOT overwrite it.
   303→      const current = await db.tenantInvoice.findUnique({ where: { id: invoice.id }, select: { status: true } });
   304→      logger.info("saas.confirm_failed_invoice_already_advanced", {
   305→        invoiceId: invoice.id, currentStatus: current?.status,
   306→      });
   307→      // If it's now paid, the subscription should reflect that.
   308→      if (current?.status === "paid") {
   309→        return { status: "active" };
   310→      }
   311→      return { status: current?.status ?? "unknown" };
   312→    }
   313→    // Only transition the subscription to past_due if the invoice was actually
   314→    // transitioned to failed (guarded).
   315→    await db.tenantSubscription.updateMany({
   316→      where: { id: subscription.id, status: { in: ["pending_payment", "active", "trialing"] } },
   317→      data: { status: "past_due" },
   318→    });
   319→    return { status: "past_due" };
   320→  }
   321→
   322→  if (verification.status === "pending") {
   323→    return { status: "pending_payment" };
   324→  }
   325→
   326→  // Payment succeeded — attempt financial finalization + activation.
   327→  // Phase 2B.3.14 P1-6: Thread the provider's authoritative paidAt through.
   328→  // Phase 2B.3.17 P0-4: Pass paymentVerified=true — verifyPayment returned "succeeded".
   329→  const result = await activateSubscriptionAndPostLedger({
   330→    subscriptionId: subscription.id,
   331→    invoiceId: invoice.id,
   332→    tenantId: input.tenantId,
   333→    userId: input.userId,
   334→    paidAt: verification.paidAt,
   335→    paymentVerified: true,
   336→  });
   337→
   338→  if (result.activated) {
   339→    return { status: "active" };
   340→  } else {
   341→    // Ledger failed — invoice is reconciliation_required, subscription NOT activated
   342→    return { status: "financial_pending" };
   343→  }
   344→}
   345→
   346→/**
   347→ * Activate the subscription and post the SaaS payment to the ledger.
   348→ * Idempotent: status-guarded transitions.
   349→ */
   350→/**
   351→ * Activate the subscription and post the SaaS payment to the ledger.
   352→ *
   353→ * Phase 2B.3.1: The activation order is now:
   354→ *   1. Post the ledger entry FIRST (the canonical financial truth)
   355→ *   2. If ledger succeeds → mark invoice as paid + activate subscription
   356→ *   3. If ledger fails → mark invoice as reconciliation_required (do NOT activate)
   357→ *
   358→ * This prevents the scenario where payment succeeds, ledger fails, and the
   359→ * subscription is falsely activated without a canonical financial record.
   360→ * The processDueSaasFinancialReconciliation worker repairs reconciliation_required
   361→ * invoices by retrying the ledger posting.
   362→ *
   363→ * Idempotent: status-guarded transitions. If already paid, returns early.
   364→ */
   365→async function activateSubscriptionAndPostLedger(input: {
   366→  subscriptionId: string;
   367→  invoiceId: string;
   368→  tenantId: string;
   369→  userId: string;
   370→  /** Phase 2B.3.14 P1-6: Provider's authoritative payment timestamp. */
   371→  paidAt?: Date;
   372→  /**
   373→   * Phase 2B.3.17 P0-4: Proof that the provider payment was verified.
   374→   *
   375→   * - `true`: the caller has verified the payment with the provider (either
   376→   *   via verifyPayment() or via a webhook). The ledger MAY be posted for
   377→   *   a "pending" invoice.
   378→   * - `false` / `undefined`: the caller has NOT verified the payment. The
   379→   *   ledger MUST NOT be posted for a "pending" invoice — only "paid" or
   380→   *   "reconciliation_required" invoices are allowed (these states prove
   381→   *   the payment was verified in a prior attempt).
   382→   *
   383→   * This guard prevents ambiguous payment states from becoming recognized
   384→   * revenue. An invoice in "pending" status without paymentVerified=true
   385→   * is refused — the caller must either verify the payment or use the
   386→   * ambiguous-payment resolution path.
   387→   */
   388→  paymentVerified?: boolean;
   389→}): Promise<{ activated: boolean }> {
   390→  await ensureChartOfAccounts();
   391→
   392→  // Get the invoice
   393→  const invoice = await db.tenantInvoice.findUnique({ where: { id: input.invoiceId } });
   394→  if (!invoice) {
   395→    throw new AppError("not_found", "Invoice not found", 404, "Invoice not found during activation.");
   396→  }
   397→
   398→  // Phase 2B.3.17 P0-4: GUARD — refuse to post revenue for an unverified payment.
   399→  //
   400→  // If the invoice is "pending" AND the caller did not provide
   401→  // paymentVerified=true, this is an ambiguous payment state. The payment
   402→  // was never confirmed by the provider. We MUST NOT post the ledger —
   403→  // that would recognize revenue without proof of payment.
   404→  //
   405→  // The only callers that may pass paymentVerified=true are:
   406→  //   - confirmSubscriptionPayment (after verifyPayment returns "succeeded")
   407→  //   - handleSaasPaymentWebhook (after webhook reports "succeeded")
   408→  //   - renewSubscription direct path (after verifyPayment returns "succeeded")
   409→  //   - processDueSaasFinancialReconciliation stale-pending scan (after
   410→  //     verifyPayment returns "succeeded")
   411→  //
   412→  // The renewal RECONCILIATION_REQUIRED path does NOT pass paymentVerified
   413→  // because it relies on the invoice being in "reconciliation_required" status
   414→  // (which proves the payment was verified in a prior attempt). If the invoice
   415→  // is "pending" in that path, it means the payment was NEVER verified —
   416→  // this is the ambiguous case, and it is refused here.
   417→  if (invoice.status === "pending" && !input.paymentVerified) {
   418→    logger.error("saas.activation_refused_unverified_payment", {
   419→      invoiceId: input.invoiceId,
   420→      invoiceStatus: invoice.status,
   421→      hasProviderReference: !!invoice.providerReference,
   422→      paymentVerified: input.paymentVerified ?? false,
   423→      message: "CRITICAL: Refused to post ledger for a pending invoice without payment verification. This prevents ambiguous payment states from becoming recognized revenue.",
   424→    });
   425→    return { activated: false };
   426→  }
   427→
   428→  // Phase 2B.3.14 P1-6: Record paidAt BEFORE attempting the ledger posting.
   429→  // This is a guarded, immutable write — only sets paidAt if it's not already set.
   430→  // This ensures that if the ledger fails and the invoice goes to reconciliation_required,
   431→  // paidAt is already captured from the original payment verification. On retry,
   432→  // paidAt is preserved (not overwritten with the retry time).
   433→  //
   434→  // The billing period (set by activateInitialSaasSubscription) is derived from
   435→  // this paidAt — so it reflects the actual payment time, not the recovery time.
   436→  if (input.paidAt) {
   437→    const paidAtResult = await db.tenantInvoice.updateMany({
   438→      where: { id: input.invoiceId, paidAt: null, status: { in: ["pending", "reconciliation_required"] } },
   439→      data: { paidAt: input.paidAt },
   440→    });
   441→    if (paidAtResult.count > 0) {
   442→      logger.info("saas.paidAt_recorded", { invoiceId: input.invoiceId, paidAt: input.paidAt });
   443→    }
   444→    // If count === 0, paidAt was already set (from a prior attempt) — preserve it.
   445→  }
   446→
   447→  // Phase 2B.3.11 P0-2: Idempotent fast path must verify ledger existence.
   448→  // A paid invoice with no ledgerTransactionId is a corrupted state —
   449→  // do NOT return activated=true.
   450→  if (invoice.status === "paid") {
   451→    if (!invoice.ledgerTransactionId) {
   452→      logger.error("saas.paid_invoice_missing_ledger", { invoiceId: input.invoiceId });
   453→      await db.tenantSubscription.updateMany({
   454→        where: { id: input.subscriptionId, status: { not: "active" } },
   455→        data: { status: "reconciliation_required" },
   456→      }).catch((err) => { logger.error("saas.state_update_failed", { error: err instanceof Error ? err.message : String(err) }); });
   457→      return { activated: false };
   458→    }
   459→    const ledgerExists = await db.ledgerTransaction.findUnique({
   460→      where: { id: invoice.ledgerTransactionId },
   461→      select: { id: true },
   462→    });
   463→    if (!ledgerExists) {
   464→      logger.error("saas.paid_invoice_ledger_not_found", { invoiceId: input.invoiceId, ledgerTxnId: invoice.ledgerTransactionId });
   465→      await db.tenantSubscription.updateMany({
   466→        where: { id: input.subscriptionId, status: { not: "active" } },
   467→        data: { status: "reconciliation_required" },
   468→      }).catch((err) => { logger.error("saas.state_update_failed", { error: err instanceof Error ? err.message : String(err) }); });
   469→      return { activated: false };
   470→    }
   471→
   472→    // Phase 2B.3.11: Check if the subscription needs domain activation.
   473→    // If it's not active yet (pending_payment or reconciliation_required),
   474→    // perform the domain activation now.
   475→    const currentSub = await db.tenantSubscription.findUnique({
   476→      where: { id: input.subscriptionId },
   477→      select: { status: true },
   478→    });
   479→
   480→    if (currentSub?.status === "pending_payment" || currentSub?.status === "reconciliation_required") {
   481→      const activationResult = await activateInitialSaasSubscription({
   482→        invoiceId: input.invoiceId,
   483→        subscriptionId: input.subscriptionId,
   484→        tenantId: input.tenantId,
   485→        userId: input.userId,
   486→        billingCycle: invoice.billingCycle,
   487→        ledgerTxnId: invoice.ledgerTransactionId,
   488→        amountMinor: invoice.amountMinor,
   489→        source: "fast_path",
   490→      });
   491→      if (!activationResult.activated) {
   492→        return { activated: false };
   493→      }
   494→    }
   495→
   496→    logger.info("saas.invoice_already_paid", { invoiceId: input.invoiceId });
   497→    return { activated: true };
   498→  }
   499→
   500→  // If invoice is in reconciliation_required, we're retrying — that's fine.
   501→  // If invoice is pending, this is the first attempt.
   502→  // If invoice is failed, we should not activate.
   503→  if (invoice.status === "failed") {
   504→    throw new AppError("conflict", "Cannot activate a failed invoice", 409, "The invoice payment failed.");
   505→  }
   506→
   507→  // Step 1: Post the ledger entry FIRST (idempotent via ledger idempotencyKey)
   508→  let ledgerTxnId: string;
   509→  try {
   510→    ledgerTxnId = await ledgerSaasSubscriptionPayment({
   511→      tenantId: input.tenantId,
   512→      userId: input.userId,
   513→      amountMinor: invoice.amountMinor,
   514→      reason: `SaaS subscription: ${invoice.saaasPlanName} (${invoice.billingCycle})`,
   515→      idempotencyKey: `${invoice.idempotencyKey}:ledger`,
   516→    });
   517→  } catch (err) {
   518→    const errorMsg = err instanceof Error ? err.message : String(err);
   519→    logger.error("saas.ledger_posting_failed", { invoiceId: input.invoiceId, error: errorMsg });
   520→    await db.tenantInvoice.updateMany({
   521→      where: { id: input.invoiceId, status: { in: ["pending", "reconciliation_required"] } },
   522→      data: { status: "reconciliation_required", failureReason: `Ledger posting failed: ${errorMsg}` },
   523→    }).catch((updateErr) => {
   524→      const updateMsg = updateErr instanceof Error ? updateErr.message : String(updateErr);
   525→      logger.error("saas.reconciliation_status_update_failed", {
   526→        invoiceId: input.invoiceId,
   527→        ledgerError: errorMsg,
   528→        updateError: updateMsg,
   529→        message: "CRITICAL: Ledger failed AND status update to reconciliation_required failed.",
   530→      });
   531→    });
   532→    return { activated: false };
   533→  }
   534→
   535→  // Step 2: Ledger succeeded — mark invoice as paid + link the ledger transaction.
   536→  // Phase 2B.3.14 P1-6: Do NOT set paidAt here — it was already set guardedly above
   537→  // (or was already set from a prior attempt). This prevents overwriting the
   538→  // authoritative provider payment timestamp with the ledger-posting time.
   539→  const updated = await db.tenantInvoice.updateMany({
   540→    where: { id: input.invoiceId, status: { in: ["pending", "reconciliation_required"] } },
   541→    data: { status: "paid", ledgerTransactionId: ledgerTxnId, failureReason: null },
   542→  });
   543→
   544→  if (updated.count === 0) {
   545→    const current = await db.tenantInvoice.findUnique({ where: { id: input.invoiceId } });
   546→    if (current?.status === "paid" && current.ledgerTransactionId) {
   547→      logger.info("saas.invoice_concurrently_paid", { invoiceId: input.invoiceId });
   548→      return { activated: true };
   549→    }
   550→    logger.warn("saas.invoice_unexpected_state", { invoiceId: input.invoiceId, status: current?.status });
   551→    return { activated: false };
   552→  }
   553→
   554→  // Phase 2B.3.19: ELIMINATED the new Date() fallback for paidAt.
   555→  //
   556→  // The previous code (2B.3.14) had a fallback that set paidAt = new Date()
   557→  // when the caller didn't provide a provider paidAt. This violated the
   558→  // invariant that billing periods must be derived from the authoritative
   559→  // provider payment timestamp — a missing timestamp was treated as "use now",
   560→  // which could drift the billing period.
   561→  //
   562→  // Now: if paidAt was not provided by the caller AND the invoice doesn't
   563→  // already have one, we refuse to finalize. The invoice goes to
   564→  // reconciliation_required. The only way forward is for a caller that has
   565→  // verified the payment (and obtained paidAt) to retry.
   566→  //
   567→  // This is fail-closed: "we don't know when the customer paid" is NOT a
   568→  // reason to invent a payment timestamp.
   569→  if (!input.paidAt) {
   570→    // Check if the invoice already has a paidAt (from a prior verified attempt).
   571→    const currentInvoice = await db.tenantInvoice.findUnique({
   572→      where: { id: input.invoiceId },
   573→      select: { paidAt: true },
   574→    });
   575→    if (!currentInvoice?.paidAt) {
   576→      // No paidAt from provider and no prior paidAt — refuse to finalize.
   577→      logger.error("saas.paidAt_missing_refused", {
   578→        invoiceId: input.invoiceId, tenantId: input.tenantId,
   579→        message: "CRITICAL: Cannot finalize invoice without an authoritative paidAt timestamp. The provider did not return paidAt, and no prior paidAt exists. Invoice marked reconciliation_required — a caller with provider verification must retry.",
   580→      });
   581→      await db.tenantInvoice.updateMany({
   582→        where: { id: input.invoiceId, status: { in: ["pending", "reconciliation_required"] } },
   583→        data: { status: "reconciliation_required", failureReason: "Missing authoritative paidAt — provider did not return payment timestamp" },
   584→      }).catch((err) => {
   585→        logger.error("saas.state_update_failed", { invoiceId: input.invoiceId, error: err instanceof Error ? err.message : String(err) });
   586→      });
   587→      return { activated: false };
   588→    }
   589→    // paidAt already exists from a prior verified attempt — proceed.
   590→  }
   591→
   592→  // Step 3: Domain activation — for initial subscriptions only.
   593→  // For renewals, completeSaasRenewalCycle() handles period extension.
   594→  const currentSub = await db.tenantSubscription.findUnique({
   595→    where: { id: input.subscriptionId },
   596→    select: { status: true },
   597→  });
   598→
   599→  if (currentSub?.status === "pending_payment" || currentSub?.status === "reconciliation_required") {
   600→    return activateInitialSaasSubscription({
   601→      invoiceId: input.invoiceId,
   602→      subscriptionId: input.subscriptionId,
   603→      tenantId: input.tenantId,
   604→      userId: input.userId,
   605→      billingCycle: invoice.billingCycle,
   606→      ledgerTxnId,
   607→      amountMinor: invoice.amountMinor,
   608→      source: "step3",
   609→    });
   610→  }
   611→
   612→  // For renewals (status was already active/past_due), the caller handles
   613→  // domain completion via completeSaasRenewalCycle().
   614→  return { activated: true };
   615→}
   616→
   617→/**
   618→ * Phase 2B.3.12: Atomic initial SaaS subscription activation.
   619→ *
   620→ * Performs invoice period update + subscription activation inside ONE
   621→ * PostgreSQL transaction with FOR UPDATE locks. If either update fails,
   622→ * the transaction rolls back — no partial activation state is committed.
   623→ *
   624→ * Phase 2B.3.13 — DETERMINISTIC BILLING PERIOD:
   625→ * The initial billing period MUST be derived from a durable financial
   626→ * timestamp captured at payment finalization (invoice.paidAt), NOT from
   627→ * the retry execution time. Retries days apart produce the SAME period.
   628→ *
   629→ *   if invoice.periodStart / periodEnd already populated:
   630→ *       reuse them           (durable recorded period — never drift)
   631→ *   else if invoice.paidAt:
   632→ *       periodStart = paidAt (the authoritative financial event)
   633→ *       periodEnd   = paidAt + billingCycle
   634→ *   else:
   635→ *       refuse to activate   (paid invoice MUST have paidAt)
   636→ *
   637→ * No path inside this function may use `new Date()` for the period start.
   638→ *
   639→ * Phase 2B.3.13 — LEDGER INVARIANT ENFORCED IN THE HELPER:
   640→ * This function does NOT trust callers for the ledger-existence invariant.
   641→ * It verifies `LedgerTransaction` exists for invoice.ledgerTransactionId
   642→ * inside the same transaction. The authoritative activation function cannot
   643→ * ever be called with a dangling ledger reference.
   644→ *
   645→ * If the transaction fails, marks the subscription reconciliation_required
   646→ * so the worker can retry. The financial state (invoice paid + ledger)
   647→ * remains intact — only the domain activation is retried. On retry, the
   648→ * billing period is recomputed identically (from paidAt).
   649→ */
   650→async function activateInitialSaasSubscription(input: {
   651→  invoiceId: string;
   652→  subscriptionId: string;
   653→  tenantId: string;
   654→  userId: string;
   655→  billingCycle: string;
   656→  ledgerTxnId: string;
   657→  amountMinor: number;
   658→  source: string; // "fast_path" | "step3"
   659→}): Promise<{ activated: boolean }> {
   660→  try {
   661→    type ActivationResult = {
   662→      activated: boolean;
   663→      reason: string;
   664→      periodStart?: Date;
   665→      periodEnd?: Date;
   666→      periodSource?: "reused" | "paidAt" | "validated";
   667→    };
   668→    const result: ActivationResult = await db.$transaction(async (tx) => {
   669→      // 1. Lock the invoice — Phase 2B.3.13: also load paidAt + existing periods
   670→      //    so the billing period can be derived deterministically.
   671→      const lockedInvoice: Array<{ id: string; status: string; ledgerTransactionId: string | null; paidAt: Date | null; periodStart: Date | null; periodEnd: Date | null }> = await tx.$queryRaw`
   672→        SELECT id, status, "ledgerTransactionId", "paidAt", "periodStart", "periodEnd"
   673→        FROM "TenantInvoice"
   674→        WHERE id = ${input.invoiceId}
   675→        FOR UPDATE
   676→      `;
   677→      if (lockedInvoice.length === 0) {
   678→        return { activated: false, reason: "Invoice not found" };
   679→      }
   680→      const inv = lockedInvoice[0];
   681→
   682→      // 2. Lock the subscription
   683→      const lockedSub: Array<{ id: string; status: string }> = await tx.$queryRaw`
   684→        SELECT id, status
   685→        FROM "TenantSubscription"
   686→        WHERE id = ${input.subscriptionId}
   687→        FOR UPDATE
   688→      `;
   689→      if (lockedSub.length === 0) {
   690→        return { activated: false, reason: "Subscription not found" };
   691→      }
   692→      const sub = lockedSub[0];
   693→
   694→      // 3. Verify prerequisites — invoice must be paid AND have a ledger reference.
   695→      if (inv.status !== "paid" || !inv.ledgerTransactionId) {
   696→        return { activated: false, reason: `Invoice not ready (status: ${inv.status})` };
   697→      }
   698→
   699→      // 3b. Phase 2B.3.13 P1: Verify the referenced LedgerTransaction ACTUALLY EXISTS
   700→      // inside the helper itself. Do not rely on callers to establish this invariant.
   701→      const ledgerCheck: Array<{ id: string }> = await tx.$queryRaw`
   702→        SELECT id FROM "LedgerTransaction" WHERE id = ${inv.ledgerTransactionId}
   703→      `;
   704→      if (ledgerCheck.length === 0) {
   705→        return { activated: false, reason: "Ledger transaction referenced by invoice does not exist (dangling reference)" };
   706→      }
   707→
   708→      // 4. Only activate if subscription is pending_payment or reconciliation_required
   709→      if (sub.status !== "pending_payment" && sub.status !== "reconciliation_required") {
   710→        // Already active — idempotent success
   711→        return { activated: true, reason: "already_active" };
   712→      }
   713→
   714→      // 5. Phase 2B.3.13/2B.3.14/2B.3.15: Derive billing period DETERMINISTICALLY.
   715→      //    The billing clock starts when the customer pays — never when a
   716→      //    recovery worker happens to repair the domain state.
   717→      let periodStart: Date;
   718→      let periodEnd: Date;
   719→      let periodSource: "reused" | "paidAt" | "validated";
   720→      if (inv.periodStart && inv.periodEnd) {
   721→        // Phase 2B.3.15 P1-3: VALIDATE the existing period using CANONICAL CALENDAR
   722→        // intervals — NOT duration tolerances. "monthly" is a calendar month
   723→        // (Jan 31 → Feb 28), not 30 days. We re-derive the expected periodEnd
   724→        // from periodStart using addBillingInterval(), then compare exactly.
   725→        const ps = inv.periodStart;
   726→        const pe = inv.periodEnd;
   727→
   728→        // 5a. Temporal sanity: periodEnd must be after periodStart.
   729→        if (pe.getTime() <= ps.getTime()) {
   730→          logger.error("saas.existing_period_invalid_duration", {
   731→            invoiceId: input.invoiceId, periodStart: ps, periodEnd: pe,
   732→          });
   733→          return { activated: false, reason: "Existing invoice period is invalid (periodEnd <= periodStart)" };
   734→        }
   735→
   736→        // 5b. Phase 2B.3.15 P1-3: Canonical calendar validation.
   737→        // Re-derive expected periodEnd from periodStart using the same calendar
   738→        // arithmetic used to create it. If they don't match exactly, the recorded
   739→        // period is corrupt.
   740→        if (!isCanonicalBillingInterval(ps, pe, input.billingCycle)) {
   741→          logger.error("saas.existing_period_not_canonical", {
   742→            invoiceId: input.invoiceId, billingCycle: input.billingCycle,
   743→            periodStart: ps, periodEnd: pe,
   744→            expected: addBillingInterval(ps, input.billingCycle),
   745→          });
   746→          return { activated: false, reason: `Existing period is not the canonical ${input.billingCycle} billing interval for the recorded periodStart` };
   747→        }
   748→
   749→        // 5c. If paidAt is available, verify periodStart is consistent with it.
   750→        // For initial subscriptions, periodStart should equal paidAt (the financial event).
   751→        // If they differ significantly, the recorded period may be corrupt.
   752→        if (inv.paidAt) {
   753→          const diffMs = Math.abs(ps.getTime() - inv.paidAt.getTime());
   754→          const diffDays = diffMs / 86400000;
   755→          if (diffDays > 1) {
   756→            // periodStart doesn't match paidAt — fall through to paidAt derivation
   757→            // rather than reusing the inconsistent period. This is a repair, not a refusal.
   758→            logger.warn("saas.existing_period_inconsistent_with_paidAt", {
   759→              invoiceId: input.invoiceId, periodStart: ps, paidAt: inv.paidAt, diffDays: diffDays.toFixed(1),
   760→              message: "Existing periodStart doesn't match paidAt — re-deriving from paidAt.",
   761→            });
   762→            periodStart = inv.paidAt;
   763→            periodEnd = addBillingInterval(periodStart, input.billingCycle);
   764→            periodSource = "validated";
   765→          } else {
   766→            periodStart = ps;
   767→            periodEnd = pe;
   768→            periodSource = "reused";
   769→          }
   770→        } else {
   771→          // No paidAt to compare against — reuse the recorded period (calendar shape was validated).
   772→          periodStart = ps;
   773→          periodEnd = pe;
   774→          periodSource = "reused";
   775→        }
   776→      } else if (inv.paidAt) {
   777→        // First activation — anchor to the financial event that created the subscription.
   778→        // Phase 2B.3.15 P1-3: Use addBillingInterval for canonical calendar derivation.
   779→        periodStart = inv.paidAt;
   780→        periodEnd = addBillingInterval(periodStart, input.billingCycle);
   781→        periodSource = "paidAt";
   782→      } else {
   783→        // Defensive — a paid invoice MUST have paidAt. If missing, the schema is corrupt;
   784→        // refuse to activate rather than fabricate a billing period.
   785→        return { activated: false, reason: "Paid invoice has no paidAt timestamp (cannot derive billing period)" };
   786→      }
   787→
   788→      // 6. Update invoice period (idempotent — same input produces same row)
   789→      await tx.tenantInvoice.update({
   790→        where: { id: input.invoiceId },
   791→        data: { periodStart, periodEnd },
   792→      });
   793→
   794→      // 7. Activate the subscription
   795→      await tx.tenantSubscription.update({
   796→        where: { id: input.subscriptionId },
   797→        data: { status: "active", currentPeriodEnd: periodEnd },
   798→      });
   799→
   800→      return { activated: true, reason: "activated", periodStart, periodEnd, periodSource };
   801→    }, { timeout: 30000, maxWait: 15000 });
   802→
   803→    if (result.activated) {
   804→      await audit({
   805→        tenantId: input.tenantId,
   806→        userId: (input.userId === "renewal-worker" || input.userId === "webhook" || input.userId === "reconciliation-worker") ? undefined : input.userId,
   807→        action: "saas.subscription_activated",
   808→        entity: "tenant_subscription",
   809→        entityId: input.subscriptionId,
   810→        // Phase 2B.3.13: persist periodSource ("paidAt" | "reused") so auditors
   811→        // can prove the billing period was derived from the financial event,
   812→        // not the retry execution time.
   813→        detail: {
   814→          invoiceId: input.invoiceId,
   815→          ledgerTxnId: input.ledgerTxnId,
   816→          amount: input.amountMinor,
   817→          source: input.source,
   818→          periodSource: result.periodSource,
   819→          periodStart: result.periodStart?.toISOString(),
   820→          periodEnd: result.periodEnd?.toISOString(),
   821→        },
   822→      });
   823→      logger.info("saas.subscription_activated", {
   824→        tenantId: input.tenantId, subscriptionId: input.subscriptionId,
   825→        ledgerTxnId: input.ledgerTxnId, invoiceId: input.invoiceId, source: input.source,
   826→        periodSource: result.periodSource,
   827→      });
   828→      return { activated: true };
   829→    } else {
   830→      logger.warn("saas.initial_activation_skipped", {
   831→        subscriptionId: input.subscriptionId, invoiceId: input.invoiceId,
   832→        reason: result.reason, source: input.source,
   833→      });
   834→      return { activated: false };
   835→    }
   836→  } catch (err) {
   837→    // Transaction failed — neither invoice period nor subscription was updated.
   838→    const errorMsg = err instanceof Error ? err.message : String(err);
   839→    logger.error("saas.initial_activation_failed", {
   840→      subscriptionId: input.subscriptionId,
   841→      invoiceId: input.invoiceId,
   842→      tenantId: input.tenantId,
   843→      error: errorMsg,
   844→      source: input.source,
   845→      message: "CRITICAL: Financial finalization succeeded but domain activation failed (transactional). Subscription marked reconciliation_required.",
   846→    });
   847→    // Mark for recovery
   848→    await db.tenantSubscription.updateMany({
   849→      where: { id: input.subscriptionId, status: { not: "active" } },
   850→      data: { status: "reconciliation_required" },
   851→    }).catch((updateErr) => {
   852→      logger.error("saas.recovery_state_persist_failed", {
   853→        subscriptionId: input.subscriptionId,
   854→        tenantId: input.tenantId,
   855→        originalError: errorMsg,
   856→        persistError: updateErr instanceof Error ? updateErr.message : String(updateErr),
   857→        message: "CRITICAL: Failed to persist reconciliation_required after activation failure. The stale-status scan will recover it.",
   858→      });
   859→    });
   860→    return { activated: false };
   861→  }
   862→}
   863→
   864→/**
   865→ * Phase 2B.3.7: Complete a SaaS renewal cycle — the SINGLE authoritative
   866→ * function that extends the subscription period after financial finalization.
   867→ *
   868→ * This function is called by ALL successful renewal paths:
   869→ *   - renewSubscription (direct path)
   870→ *   - handleSaasPaymentWebhook (webhook path)
   871→ *   - processDueSaasFinancialReconciliation (reconciliation path)
   872→ *
   873→ * Phase 2B.3.7: The cycle completion + subscription period extension are
   874→ * performed inside ONE PostgreSQL transaction with FOR UPDATE locks on both
   875→ * the SaasRenewalCycle and TenantSubscription rows. This guarantees:
   876→ *
   877→ *   IF cycle = COMPLETED
   878→ *   THEN subscription.currentPeriodEnd = cycle.periodEnd
   879→ *
   880→ * No partial completion is possible. If the subscription update fails,
   881→ * the transaction rolls back and the cycle does NOT become COMPLETED.
   882→ *
   883→ * Idempotent: if already COMPLETED, verifies currentPeriodEnd == cycle.periodEnd.
   884→ * If they differ (stale legacy state), repairs the subscription period
   885→ * inside a new transaction.
   886→ */
   887→async function completeSaasRenewalCycle(input: {
   888→  invoiceId: string;
   889→  tenantId: string;
   890→}): Promise<{ completed: boolean }> {
   891→  // Find the renewal cycle linked to this invoice (read-only, outside transaction)
   892→  const cycle = await db.saasRenewalCycle.findFirst({
   893→    where: { invoiceId: input.invoiceId },
   894→  });
   895→  if (!cycle) {
   896→    // No cycle — this is an initial subscription, not a renewal. Nothing to complete.
   897→    return { completed: false };
   898→  }
   899→
   900→  try {
   901→    // Phase 2B.3.7: Perform the completion inside ONE transaction with row locks.
   902→    const result = await db.$transaction(async (tx) => {
   903→      // 1. Lock the SaasRenewalCycle row
   904→      const lockedCycle: Array<{ id: string; state: string; subscriptionId: string; periodEnd: Date; invoiceId: string | null }> = await tx.$queryRaw`
   905→        SELECT id, state, "subscriptionId", "periodEnd", "invoiceId"
   906→        FROM "SaasRenewalCycle"
   907→        WHERE id = ${cycle.id}
   908→        FOR UPDATE
   909→      `;
   910→      if (lockedCycle.length === 0) {
   911→        return { completed: false, reason: "Cycle not found" };
   912→      }
   913→      const c = lockedCycle[0];
   914→
   915→      // 2. Lock the TenantSubscription row
   916→      const lockedSub: Array<{ id: string; currentPeriodEnd: Date; status: string }> = await tx.$queryRaw`
   917→        SELECT id, "currentPeriodEnd", status
   918→        FROM "TenantSubscription"
   919→        WHERE id = ${c.subscriptionId}
   920→        FOR UPDATE
   921→      `;
   922→      if (lockedSub.length === 0) {
   923→        return { completed: false, reason: "Subscription not found" };
   924→      }
   925→      const sub = lockedSub[0];
   926→
   927→      // 3. If already COMPLETED, verify the invariant. Repair if stale.
   928→      // Phase 2B.3.19: The COMPLETED idempotent path now ALSO verifies the
   929→      // ledger transaction still exists. LedgerTransaction rows are immutable
   930→      // (no delete path in the codebase), but this check provides defense-in-depth
   931→      // and explicitly answers: "every call that recognizes COMPLETED independently
   932→      // proves the ledger still exists." If the ledger was somehow removed, the
   933→      // cycle is transitioned to RECONCILIATION_REQUIRED rather than returning
   934→      // a false success.
   935→      if (c.state === "COMPLETED") {
   936→        // Verify the invoice is still paid and the ledger still exists.
   937→        if (c.invoiceId) {
   938→          const invCheck = await tx.tenantInvoice.findUnique({
   939→            where: { id: c.invoiceId },
   940→            select: { status: true, ledgerTransactionId: true },
   941→          });
   942→          if (!invCheck || invCheck.status !== "paid" || !invCheck.ledgerTransactionId) {
   943→            logger.error("saas.cycle_completed_invoice_not_ready", {
   944→              cycleId: c.id, invoiceId: c.invoiceId, invoiceStatus: invCheck?.status,
   945→            });
   946→            return { completed: false, reason: "COMPLETED cycle's invoice is no longer paid" };
   947→          }
   948→          const ledgerStillExists: Array<{ id: string }> = await tx.$queryRaw`
   949→            SELECT id FROM "LedgerTransaction" WHERE id = ${invCheck.ledgerTransactionId}
   950→          `;
   951→          if (ledgerStillExists.length === 0) {
   952→            logger.error("saas.cycle_completed_ledger_missing", {
   953→              cycleId: c.id, invoiceId: c.invoiceId, ledgerTxnId: invCheck.ledgerTransactionId,
   954→              message: "CRITICAL: COMPLETED cycle references a ledger transaction that no longer exists. Cycle left in COMPLETED state but completion refused — manual audit required.",
   955→            });
   956→            return { completed: false, reason: "COMPLETED cycle's ledger transaction no longer exists" };
   957→          }
   958→        }
   959→
   960→        if (sub.currentPeriodEnd.getTime() === c.periodEnd.getTime()) {
   961→          // Invariant holds — idempotent success
   962→          return { completed: true, reason: "already_completed" };
   963→        }
   964→        // Stale legacy state: cycle says COMPLETED but period wasn't extended.
   965→        // Repair inside this transaction.
   966→        logger.warn("saas.cycle_stale_repairing", {
   967→          cycleId: c.id,
   968→          currentPeriodEnd: sub.currentPeriodEnd,
   969→          expectedPeriodEnd: c.periodEnd,
   970→        });
   971→        await tx.tenantSubscription.update({
   972→          where: { id: c.subscriptionId },
   973→          data: { status: "active", currentPeriodEnd: c.periodEnd },
   974→        });
   975→        return { completed: true, reason: "stale_repaired" };
   976→      }
   977→
   978→      // 4. Verify the invoice is financially ready (paid + ledger exists)
   979→      if (!c.invoiceId) {
   980→        return { completed: false, reason: "No invoice linked to cycle" };
   981→      }
   982→      const invoice = await tx.tenantInvoice.findUnique({
   983→        where: { id: c.invoiceId },
   984→        select: { status: true, ledgerTransactionId: true },
   985→      });
   986→      if (!invoice || invoice.status !== "paid" || !invoice.ledgerTransactionId) {
   987→        return { completed: false, reason: `Invoice not ready (status: ${invoice?.status})` };
   988→      }
   989→
   990→      // Phase 2B.3.14 P1-7: Verify the referenced LedgerTransaction ACTUALLY EXISTS
   991→      // inside the helper's own transaction — symmetric with activateInitialSaasSubscription.
   992→      // The authoritative completion function must NOT trust a non-null ledgerTransactionId
   993→      // without proving the row exists.
   994→      const ledgerCheck: Array<{ id: string }> = await tx.$queryRaw`
   995→        SELECT id FROM "LedgerTransaction" WHERE id = ${invoice.ledgerTransactionId}
   996→      `;
   997→      if (ledgerCheck.length === 0) {
   998→        logger.error("saas.cycle_completion_dangling_ledger", {
   999→          cycleId: c.id, invoiceId: c.invoiceId, ledgerTxnId: invoice.ledgerTransactionId,
  1000→        });
  1001→        return { completed: false, reason: "Ledger transaction referenced by invoice does not exist (dangling reference)" };
  1002→      }
  1003→
  1004→      // 5. Update the subscription period FIRST (before marking cycle COMPLETED)
  1005→      await tx.tenantSubscription.update({
  1006→        where: { id: c.subscriptionId },
  1007→        data: {
  1008→          status: "active",
  1009→          currentPeriodEnd: c.periodEnd,
  1010→        },
  1011→      });
  1012→
  1013→      // 6. Mark the cycle COMPLETED (after subscription update succeeds)
  1014→      // Status-guarded: only non-COMPLETED → COMPLETED
  1015→      await tx.saasRenewalCycle.updateMany({
  1016→        where: { id: c.id, state: { not: "COMPLETED" } },
  1017→        data: { state: "COMPLETED", failureReason: null },
  1018→      });
  1019→
  1020→      return { completed: true, reason: "completed" };
  1021→    }, { timeout: 30000, maxWait: 15000 });
  1022→
  1023→    if (result.completed) {
  1024→      if (result.reason === "stale_repaired") {
  1025→        logger.info("saas.cycle_stale_repaired", { cycleId: cycle.id });
  1026→      } else if (result.reason !== "already_completed") {
  1027→        logger.info("saas.renewal_cycle_completed", {
  1028→          cycleId: cycle.id,
  1029→          subscriptionId: cycle.subscriptionId,
  1030→          newPeriodEnd: cycle.periodEnd,
  1031→        });
  1032→      } else {
  1033→        logger.info("saas.cycle_already_completed", { cycleId: cycle.id });
  1034→      }
  1035→    } else {
  1036→      logger.warn("saas.cycle_not_completed", { cycleId: cycle.id, reason: result.reason });
  1037→    }
  1038→
  1039→    return { completed: result.completed };
  1040→  } catch (err) {
  1041→    // Transaction failed — neither the cycle nor the subscription was updated
  1042→    logger.error("saas.cycle_completion_failed", {
  1043→      cycleId: cycle.id,
  1044→      error: err instanceof Error ? err.message : String(err),
  1045→    });
  1046→    return { completed: false };
  1047→  }
  1048→}

  1050→/**
  1051→ * Cancel a subscription. The subscription remains active until the end of the
  1052→ * current period, then expires. No more renewals.
  1053→ */
  1054→export async function cancelSubscription(input: {
  1055→  tenantId: string;
  1056→  userId: string;
  1057→  reason?: string;
  1058→}): Promise<{ status: string; currentPeriodEnd: Date }> {
  1059→  const subscription = await db.tenantSubscription.findUnique({
  1060→    where: { tenantId: input.tenantId },
  1061→  });
  1062→  if (!subscription) {
  1063→    throw new AppError("not_found", "No subscription found", 404, "This tenant has no subscription.");
  1064→  }
  1065→
  1066→  // Phase 2B.3.14 P2-10: Guard cancellation against financial recovery states.
  1067→  // A subscription in reconciliation_required has a paid invoice + posted ledger
  1068→  // that is awaiting domain activation. Cancelling it would orphan the financial
  1069→  // state with no recovery path.
  1070→  if (subscription.status === "reconciliation_required") {
  1071→    throw new AppError(
  1072→      "conflict",
  1073→      "Cannot cancel subscription in recovery state",
  1074→      409,
  1075→      "This subscription has a pending financial reconciliation. Wait for the reconciliation worker to complete before cancelling, or contact support.",
  1076→    );
  1077→  }
  1078→
  1079→  // Phase 2B.3.14 P2-10: Guarded transition — only cancel if not already cancelled.
  1080→  const result = await db.tenantSubscription.updateMany({
  1081→    where: { id: subscription.id, status: { not: "cancelled" } },
  1082→    data: {
  1083→      status: "cancelled",
  1084→      cancelledAt: new Date(),
  1085→      cancelReason: input.reason ?? null,
  1086→    },
  1087→  });
  1088→  if (result.count === 0) {
  1089→    // Already cancelled — idempotent
  1090→    return { status: "cancelled", currentPeriodEnd: subscription.currentPeriodEnd };
  1091→  }
  1092→
  1093→  await audit({
  1094→    tenantId: input.tenantId,
  1095→    userId: input.userId,
  1096→    action: "saas.subscription_cancelled",
  1097→    entity: "tenant_subscription",
  1098→    entityId: subscription.id,
  1099→    detail: { reason: input.reason },
  1100→  });
  1101→
  1102→  logger.info("saas.subscription_cancelled", { tenantId: input.tenantId, subscriptionId: subscription.id });
  1103→
  1104→  return { status: "cancelled", currentPeriodEnd: subscription.currentPeriodEnd };
  1105→}

  1107→/**
  1108→ * Renew a subscription for the next billing period.
  1109→ * Creates an invoice + payment intent + charges.
  1110→ *
  1111→ * Idempotent: uses a durable renewalIdempotencyKey derived from the
  1112→ * subscription ID + period end. Duplicate cron calls or webhook retries
  1113→ * cannot double-charge.
  1114→ */
  1115→export async function renewSubscription(tenantId: string): Promise<{ success: boolean; status: string; reason?: string }> {
  1116→  const subscription = await db.tenantSubscription.findUnique({
  1117→    where: { tenantId },
  1118→    include: { saaasPlan: true },
  1119→  });
  1120→  if (!subscription) {
  1121→    return { success: false, status: "none", reason: "No subscription found" };
  1122→  }
  1123→
  1124→  // Don't renew cancelled subscriptions
  1125→  if (subscription.status === "cancelled") {
  1126→    return { success: false, status: "cancelled", reason: "Subscription is cancelled" };
  1127→  }
  1128→
  1129→  // Phase 2B.3.14 P2-13: Don't renew free plans through the paid SaaS machinery.
  1130→  // Free plans have monthlyPriceMinor = 0 — the ledger rejects zero-amount
  1131→  // transactions, and there's no payment to process. Free-plan subscriptions
  1132→  // should be handled by a separate product-level flow, not the payment pipeline.
  1133→  if (subscription.saaasPlan && subscription.saaasPlan.monthlyPriceMinor <= 0) {
  1134→    logger.info("saas.free_plan_renewal_skipped", { tenantId, planName: subscription.saaasPlan.name });
  1135→    // Extend the period without payment/ledger — free plan.
  1136→    // Phase 2B.3.15 P1-3: Use addBillingInterval for canonical calendar derivation.
  1137→    const newPeriodEnd = addBillingInterval(subscription.currentPeriodEnd, subscription.billingCycle);
  1138→    await db.tenantSubscription.updateMany({
  1139→      where: { id: subscription.id, status: { in: ["active", "past_due"] } },
  1140→      data: { status: "active", currentPeriodEnd: newPeriodEnd },
  1141→    });
  1142→    return { success: true, status: "active" };
  1143→  }
  1144→
  1145→  // Don't renew if the current period hasn't ended yet
  1146→  if (subscription.currentPeriodEnd > new Date()) {
  1147→    return { success: false, status: subscription.status, reason: "Current period has not ended" };
  1148→  }
  1149→
  1150→  const plan = subscription.saaasPlan;
  1151→  const amountMinor = subscription.billingCycle === "yearly" ? plan.monthlyPriceMinor * 12 : plan.monthlyPriceMinor;
  1152→
  1153→  // Phase 2B.3.5: Derive the durable cycle identity from immutable periodStart.
  1154→  // The cycleKey is subscriptionId + immutable periodStart (the expired period's end).
  1155→  // This is the SAME key regardless of how many workers run concurrently.
  1156→  // Phase 2B.3.15 P1-3: Use addBillingInterval for canonical calendar derivation.
  1157→  const newPeriodStart = subscription.currentPeriodEnd;
  1158→  const newPeriodEnd = addBillingInterval(newPeriodStart, subscription.billingCycle);
  1159→
  1160→  const cycleKey = `saas_renewal_${subscription.id}_${newPeriodStart.getTime()}`;
  1161→
  1162→  // Phase 2B.3.5: Atomically get-or-create the SaasRenewalCycle.
  1163→  // The unique constraint on cycleKey ensures two concurrent workers
  1164→  // resolve to the SAME cycle.
  1165→  let cycle = await db.saasRenewalCycle.findUnique({ where: { cycleKey } });
  1166→  if (!cycle) {
  1167→    try {
  1168→      cycle = await db.saasRenewalCycle.create({
  1169→        data: {
  1170→          subscriptionId: subscription.id,
  1171→          tenantId,
  1172→          cycleKey,
  1173→          state: "PENDING",
  1174→          periodStart: newPeriodStart,
  1175→          periodEnd: newPeriodEnd,
  1176→        },
  1177→      });
  1178→      logger.info("saas.renewal_cycle_created", { cycleId: cycle.id, cycleKey, subscriptionId: subscription.id });
  1179→    } catch (err: any) {
  1180→      // P2002 = another concurrent worker created it. Re-fetch.
  1181→      if (err?.code === "P2002") {
  1182→        cycle = await db.saasRenewalCycle.findUnique({ where: { cycleKey } });
  1183→        logger.info("saas.renewal_cycle_existing", { cycleId: cycle?.id, cycleKey });
  1184→      } else {
  1185→        throw err;
  1186→      }
  1187→    }
  1188→  }
  1189→  if (!cycle) {
  1190→    return { success: false, status: "error", reason: "Failed to create/find renewal cycle" };
  1191→  }
  1192→
  1193→  // Phase 2B.3.9: If the cycle is already COMPLETED, do NOT blindly return success.
  1194→  // Route through completeSaasRenewalCycle() which verifies the invariant
  1195→  // (subscription.currentPeriodEnd == cycle.periodEnd) and repairs if stale.
  1196→  if (cycle.state === "COMPLETED") {
  1197→    if (!cycle.invoiceId) {
  1198→      // COMPLETED cycle with no invoice — fail closed, don't silently succeed
  1199→      logger.error("saas.completed_cycle_no_invoice", { cycleId: cycle.id });
  1200→      return { success: false, status: "error", reason: "Completed cycle has no invoice — manual reconciliation required" };
  1201→    }
  1202→    const completion = await completeSaasRenewalCycle({ invoiceId: cycle.invoiceId, tenantId });
  1203→    if (completion.completed) {
  1204→      return { success: true, status: "active" };
  1205→    } else {
  1206→      return { success: false, status: "error", reason: "Completed cycle invariant verification failed" };
  1207→    }
  1208→  }
  1209→
  1210→  // Phase 2B.3.17 P0-1/P0-3: AMBIGUOUS_PAYMENT is a separate state from
  1211→  // RECONCILIATION_REQUIRED. It represents a payment-operation creation that
  1212→  // timed out without a providerReference — the payment may or may not exist
  1213→  // at the provider. It MUST NOT be fed into activateSubscriptionAndPostLedger(),
  1214→  // because that would post revenue without proof of payment.
  1215→  if (cycle.state === "AMBIGUOUS_PAYMENT") {
  1216→    logger.error("saas.renewal_refused_ambiguous_payment", {
  1217→      cycleId: cycle.id, tenantId, subscriptionId: subscription.id,
  1218→      message: "CRITICAL: Renewal refused for AMBIGUOUS_PAYMENT cycle. This state requires manual/provider audit to determine whether a payment operation was created. Revenue must NOT be recognized until the ambiguity is resolved.",
  1219→    });
  1220→    return {
  1221→      success: false,
  1222→      status: "error",
  1223→      reason: "Cycle is in AMBIGUOUS_PAYMENT state — manual/provider audit required before renewal can proceed",
  1224→    };
  1225→  }
  1226→
  1227→  // If the cycle is RECONCILIATION_REQUIRED, attempt to finalize the existing invoice.
  1228→  // Phase 2B.3.17 P0-4: This path does NOT pass paymentVerified=true.
  1229→  // The guard inside activateSubscriptionAndPostLedger() will refuse to post
  1230→  // the ledger if the invoice is "pending" (unverified). Only invoices in
  1231→  // "reconciliation_required" or "paid" status are allowed — these prove the
  1232→  // payment was verified in a prior attempt.
  1233→  if (cycle.state === "RECONCILIATION_REQUIRED" || cycle.state === "FINANCIAL_POSTED") {
  1234→    if (!cycle.invoiceId) {
  1235→      return { success: false, status: "error", reason: "Cycle in financial state but no invoice" };
  1236→    }
  1237→    // Phase 2B.3.17 P0-4: paymentVerified is intentionally NOT passed here.
  1238→    // If the invoice is "pending", the guard refuses the ledger posting.
  1239→    const result = await activateSubscriptionAndPostLedger({
  1240→      subscriptionId: subscription.id,
  1241→      invoiceId: cycle.invoiceId,
  1242→      tenantId,
  1243→      userId: "renewal-worker",
  1244→    });
  1245→
  1246→    if (result.activated) {
  1247→      // Phase 2B.3.10: Use the single authoritative completion function.
  1248→      // If completion fails, mark cycle RECONCILIATION_REQUIRED for recovery.
  1249→      const completion = await completeSaasRenewalCycle({ invoiceId: cycle.invoiceId!, tenantId });
  1250→      if (completion.completed) {
  1251→        return { success: true, status: "active" };
  1252→      } else {
  1253→        await db.saasRenewalCycle.updateMany({
  1254→          where: { id: cycle.id, state: { not: "COMPLETED" } },
  1255→          data: { state: "RECONCILIATION_REQUIRED", failureReason: "Period extension failed during renewal retry" },
  1256→        }).catch((err) => { logger.error("saas.state_update_failed", { error: err instanceof Error ? err.message : String(err) }); });
  1257→        return { success: false, status: "financial_pending", reason: "Period extension failed during renewal retry" };
  1258→      }
  1259→    } else {
  1260→      return { success: false, status: "financial_pending", reason: "Ledger posting still failing during renewal retry" };
  1261→    }
  1262→  }
  1263→
  1264→  // If the cycle is PAST_DUE (payment failed), don't retry automatically.
  1265→  if (cycle.state === "PAST_DUE") {
  1266→    return { success: false, status: "past_due", reason: "Renewal payment previously failed" };
  1267→  }
  1268→
  1269→  // Cycle is PENDING or PAYMENT_PENDING — proceed with the renewal.
  1270→  // Use the invoice idempotency key derived from the cycle identity.
  1271→  const renewalKey = cycleKey;
  1272→
  1273→  // Check if an invoice already exists for this cycle
  1274→  const existingInvoice = cycle.invoiceId
  1275→    ? await db.tenantInvoice.findUnique({ where: { id: cycle.invoiceId } })
  1276→    : await db.tenantInvoice.findUnique({ where: { idempotencyKey: renewalKey } });
  1277→
  1278→  if (existingInvoice && existingInvoice.status === "paid") {
  1279→    // Phase 2B.3.8: Use the single authoritative completion function.
  1280→    // NO direct cycle/subscription mutation outside completeSaasRenewalCycle().
  1281→    if (!cycle.invoiceId) {
  1282→      // Phase 2B.3.14 P2-12: Guarded — only update invoiceId if not already set.
  1283→      await db.saasRenewalCycle.updateMany({
  1284→        where: { id: cycle.id, invoiceId: null },
  1285→        data: { invoiceId: existingInvoice.id },
  1286→      });
  1287→    }
  1288→    const completion = await completeSaasRenewalCycle({ invoiceId: existingInvoice.id, tenantId });
  1289→    if (completion.completed) {
  1290→      return { success: true, status: "active" };
  1291→    } else {
  1292→      // Phase 2B.3.10: Mark cycle for recovery
  1293→      await db.saasRenewalCycle.updateMany({
  1294→        where: { id: cycle.id, state: { not: "COMPLETED" } },
  1295→        data: { state: "RECONCILIATION_REQUIRED", failureReason: "Completion failed for already-paid invoice" },
  1296→      }).catch((err) => { logger.error("saas.state_update_failed", { error: err instanceof Error ? err.message : String(err) }); });
  1297→      return { success: false, status: "financial_pending", reason: "Completion failed for already-paid invoice" };
  1298→    }
  1299→  }
  1300→
  1301→  // Create or reuse the invoice
  1302→  const invoice = await db.tenantInvoice.upsert({
  1303→    where: { idempotencyKey: renewalKey },
  1304→    create: {
  1305→      tenantId,
  1306→      subscriptionId: subscription.id,
  1307→      saaasPlanName: plan.name,
  1308→      amountMinor,
  1309→      currency: plan.currency,
  1310→      billingCycle: subscription.billingCycle,
  1311→      periodStart: newPeriodStart,
  1312→      periodEnd: newPeriodEnd,
  1313→      status: "pending",
  1314→      idempotencyKey: renewalKey,
  1315→    },
  1316→    update: {},
  1317→  });
  1318→
  1319→  // Link the invoice to the cycle + transition to PAYMENT_PENDING.
  1320→  // Phase 2B.3.14 P2-12: Guarded — don't overwrite COMPLETED or RECONCILIATION_REQUIRED.
  1321→  await db.saasRenewalCycle.updateMany({
  1322→    where: { id: cycle.id, state: { notIn: ["COMPLETED", "RECONCILIATION_REQUIRED", "PAYMENT_PENDING"] } },
  1323→    data: { state: "PAYMENT_PENDING", invoiceId: invoice.id },
  1324→  });
  1325→
  1326→  if (invoice.status === "paid") {
  1327→    // Phase 2B.3.10: Use the single authoritative completion function.
  1328→    const completion = await completeSaasRenewalCycle({ invoiceId: invoice.id, tenantId });
  1329→    if (completion.completed) {
  1330→      return { success: true, status: "active" };
  1331→    } else {
  1332→      await db.saasRenewalCycle.updateMany({
  1333→        where: { id: cycle.id, state: { not: "COMPLETED" } },
  1334→        data: { state: "RECONCILIATION_REQUIRED", failureReason: "Completion failed for paid invoice" },
  1335→      }).catch((err) => { logger.error("saas.state_update_failed", { error: err instanceof Error ? err.message : String(err) }); });
  1336→      return { success: false, status: "financial_pending", reason: "Completion failed for paid invoice" };
  1337→    }
  1338→  }
  1339→
  1340→  // If the invoice is reconciliation_required, attempt to finalize
  1341→  if (invoice.status === "reconciliation_required") {
  1342→    await db.saasRenewalCycle.updateMany({
  1343→      where: { id: cycle.id, state: { notIn: ["COMPLETED", "RECONCILIATION_REQUIRED"] } },
  1344→      data: { state: "FINANCIAL_POSTED" },
  1345→    });
  1346→    const result = await activateSubscriptionAndPostLedger({
  1347→      subscriptionId: subscription.id,
  1348→      invoiceId: invoice.id,
  1349→      tenantId,
  1350→      userId: "renewal-worker",
  1351→    });
  1352→
  1353→    if (result.activated) {
  1354→      // Phase 2B.3.10: Use the single authoritative completion function.
  1355→      const completion = await completeSaasRenewalCycle({ invoiceId: invoice.id, tenantId });
  1356→      if (completion.completed) {
  1357→        return { success: true, status: "active" };
  1358→      } else {
  1359→        await db.saasRenewalCycle.updateMany({
  1360→          where: { id: cycle.id, state: { not: "COMPLETED" } },
  1361→          data: { state: "RECONCILIATION_REQUIRED", failureReason: "Period extension failed" },
  1362→        }).catch((err) => { logger.error("saas.state_update_failed", { error: err instanceof Error ? err.message : String(err) }); });
  1363→        return { success: false, status: "financial_pending", reason: "Period extension failed" };
  1364→      }
  1365→    } else {
  1366→      await db.saasRenewalCycle.updateMany({
  1367→        where: { id: cycle.id, state: { not: "COMPLETED" } },
  1368→        data: { state: "RECONCILIATION_REQUIRED" },
  1369→      });
  1370→      return { success: false, status: "financial_pending", reason: "Ledger posting still failing during renewal retry" };
  1371→    }
  1372→  }
  1373→
  1374→  // Phase 2B.3.3: Use the subscription's stored payment provider.
  1375→  // Phase 2B.3.16: ATOMIC PAYMENT-OPERATION ACQUISITION.
  1376→  //
  1377→  // The previous code (2B.3.15) had a race: two concurrent workers could both
  1378→  // read providerReference = null, both call createPaymentIntent(), and both
  1379→  // try to persist. The invariant "ONE INVOICE → ONE PROVIDER PAYMENT OPERATION"
  1380→  // relied entirely on provider-level idempotency.
  1381→  //
  1382→  // This fix introduces a durable payment-operation state machine at the
  1383→  // APPLICATION level, using the SaasRenewalCycle's state field:
  1384→  //
  1385→  //   PAYMENT_PENDING → PAYMENT_CREATING → PAYMENT_CONFIRMED
  1386→  //
  1387→  // The transition PAYMENT_PENDING → PAYMENT_CREATING is a PostgreSQL-atomic
  1388→  // conditional mutation (updateMany WHERE state = 'PAYMENT_PENDING'). Only
  1389→  // ONE worker can win this claim. The winner calls createPaymentIntent and
  1390→  // persists the providerReference. Losers re-read and reuse the reference.
  1391→  //
  1392→  // Crash recovery: if the winner crashes after claiming PAYMENT_CREATING but
  1393→  // before persisting the providerReference, the cycle is stuck in
  1394→  // PAYMENT_CREATING. The reconciliation worker scans for cycles stuck in
  1395→  // PAYMENT_CREATING for > 5 minutes and re-claims them (see
  1396→  // processDueSaasFinancialReconciliation).
  1397→  //
  1398→  // This does NOT eliminate provider-level idempotency as a defense — it
  1399→  // adds an APPLICATION-LEVEL guarantee that only one worker calls
  1400→  // createPaymentIntent, without relying on the provider to deduplicate.
  1401→  const provider = getPaymentProviderByKey(subscription.paymentProvider || "mock");
  1402→  let providerReference: string;
  1403→
  1404→  // Re-read the invoice to get the current providerReference.
  1405→  const currentInvoice = await db.tenantInvoice.findUnique({
  1406→    where: { id: invoice.id },
  1407→    select: { providerReference: true, paymentProvider: true },
  1408→  });
  1409→
  1410→  if (currentInvoice?.providerReference) {
  1411→    // Reuse the existing provider operation — never create a second one.
  1412→    providerReference = currentInvoice.providerReference;
  1413→    logger.info("saas.renewal_reusing_provider_reference", {
  1414→      invoiceId: invoice.id, cycleId: cycle.id, providerReference,
  1415→    });
  1416→  } else {
  1417→    // Phase 2B.3.16: ATOMIC CLAIM — transition cycle to PAYMENT_CREATING.
  1418→    // Only the worker that successfully transitions PAYMENT_PENDING →
  1419→    // PAYMENT_CREATING may call createPaymentIntent. This is a
  1420→    // PostgreSQL-atomic conditional mutation.
  1421→    const claim = await db.saasRenewalCycle.updateMany({
  1422→      where: { id: cycle.id, state: "PAYMENT_PENDING" },
  1423→      data: { state: "PAYMENT_CREATING" },
  1424→    });
  1425→
  1426→    if (claim.count === 0) {
  1427→      // Another worker is already creating the payment operation (cycle is in
  1428→      // PAYMENT_CREATING) or has already moved past it. We must NOT call
  1429→      // createPaymentIntent. Wait briefly for the winner to persist the
  1430→      // providerReference, then re-read and reuse it.
  1431→      logger.info("saas.payment_creation_claim_lost", {
  1432→        cycleId: cycle.id, invoiceId: invoice.id,
  1433→      });
  1434→
  1435→      // Wait up to 10 seconds (polling every 500ms) for the providerReference
  1436→      // to appear. If it doesn't appear, return financial_pending — the
  1437→      // reconciliation worker will recover the stuck cycle.
  1438→      let waited = 0;
  1439→      let ref: string | null = null;
  1440→      while (waited < 10000) {
  1441→        await new Promise((r) => setTimeout(r, 500));
  1442→        waited += 500;
  1443→        const reRead = await db.tenantInvoice.findUnique({
  1444→          where: { id: invoice.id },
  1445→          select: { providerReference: true },
  1446→        });
  1447→        if (reRead?.providerReference) {
  1448→          ref = reRead.providerReference;
  1449→          break;
  1450→        }
  1451→      }
  1452→
  1453→      if (ref) {
  1454→        providerReference = ref;
  1455→        logger.info("saas.payment_creation_reuse_after_wait", {
  1456→          cycleId: cycle.id, invoiceId: invoice.id, providerReference, waitedMs: waited,
  1457→        });
  1458→      } else {
  1459→        // The winner didn't persist the reference within 10 seconds.
  1460→        // This could be a crash or a slow provider. Return financial_pending —
  1461→        // the reconciliation worker will re-claim the cycle after 5 minutes.
  1462→        logger.warn("saas.payment_creation_timeout", {
  1463→          cycleId: cycle.id, invoiceId: invoice.id, waitedMs: waited,
  1464→        });
  1465→        return {
  1466→          success: false,
  1467→          status: "financial_pending",
  1468→          reason: "Payment operation creation in progress by another worker — cycle will be recovered if stuck",
  1469→        };
  1470→      }
  1471→    } else {
  1472→      // We won the claim — we are the ONLY worker allowed to call createPaymentIntent.
  1473→      logger.info("saas.payment_creation_claimed", {
  1474→        cycleId: cycle.id, invoiceId: invoice.id,
  1475→      });
  1476→
  1477→      // Create the payment intent at the provider.
  1478→      const intent = await provider.createPaymentIntent({
  1479→        amountMinor,
  1480→        currency: plan.currency as Currency,
  1481→        description: `SaaS renewal: ${plan.displayName} (${subscription.billingCycle})`,
  1482→        idempotencyKey: renewalKey,
  1483→        metadata: { tenantId, planName: plan.name, type: "saas_renewal" },
  1484→      });
  1485→      providerReference = intent.providerReference;
  1486→
  1487→      // Persist the providerReference BEFORE any other work.
  1488→      // If persistence fails, the cycle stays in PAYMENT_CREATING — the
  1489→      // reconciliation worker will re-claim it after 5 minutes and retry.
  1490→      try {
  1491→        await db.tenantInvoice.update({
  1492→          where: { id: invoice.id },
  1493→          data: { providerReference, paymentProvider: provider.id },
  1494→        });
  1495→      } catch (err) {
  1496→        const errorMsg = err instanceof Error ? err.message : String(err);
  1497→        logger.error("saas.providerReference_persist_failed", {
  1498→          invoiceId: invoice.id, cycleId: cycle.id, providerReference, error: errorMsg,
  1499→          message: "CRITICAL: Failed to persist providerReference after claiming PAYMENT_CREATING. Cycle will be recovered by the reconciliation worker after 5 minutes.",
  1500→        });
  1501→        // Leave the cycle in PAYMENT_CREATING — the reconciliation worker
  1502→        // will re-claim it and retry. Do NOT transition to RECONCILIATION_REQUIRED
  1503→        // because that would allow another worker to immediately re-attempt
  1504→        // createPaymentIntent, which could create a second provider operation
  1505→        // if the first one actually succeeded at the provider but we didn't
  1506→        // receive the response.
  1507→        return {
  1508→          success: false,
  1509→          status: "financial_pending",
  1510→          reason: "ProviderReference persistence failed — cycle will be recovered after 5-minute timeout",
  1511→        };
  1512→      }
  1513→    }
  1514→  }
  1515→
  1516→  // For the mock provider, auto-confirm
  1517→  if (provider.isMock) {
  1518→    const { mockPaymentProvider } = await import("@/lib/payments");
  1519→    mockPaymentProvider.confirmIntent(providerReference);
  1520→  }
  1521→
  1522→  // Verify the payment
  1523→  const verification = await provider.verifyPayment({
  1524→    providerReference,
  1525→    idempotencyKey: renewalKey,
  1526→  });
  1527→
  1528→  if (verification.status === "failed") {
  1529→    // Phase 2B.3.14 P0-3: GUARDED transitions — never overwrite a concurrently-finalized state.
  1530→    // A webhook may have arrived during verifyPayment() and finalized the invoice/cycle.
  1531→    // Each mutation is PostgreSQL-atomic with a state guard.
  1532→    const invoiceGuarded = await db.tenantInvoice.updateMany({
  1533→      where: { id: invoice.id, status: "pending" },
  1534→      data: { status: "failed", failureReason: "Renewal payment failed" },
  1535→    });
  1536→    if (invoiceGuarded.count === 0) {
  1537→      // The invoice was no longer pending — a concurrent worker advanced it.
  1538→      logger.info("saas.renewal_failed_invoice_already_advanced", { invoiceId: invoice.id, cycleId: cycle.id });
  1539→      // Do NOT overwrite the subscription or cycle — they may have been finalized.
  1540→      // Re-read and reconcile.
  1541→      const currentCycle = await db.saasRenewalCycle.findUnique({ where: { id: cycle.id }, select: { state: true } });
  1542→      if (currentCycle?.state === "COMPLETED") {
  1543→        return { success: true, status: "active" };
  1544→      }
  1545→      return { success: false, status: "financial_pending", reason: "Invoice was concurrently advanced — skipping destructive overwrite" };
  1546→    }
  1547→    // Invoice was actually transitioned to failed — now guard the subscription + cycle.
  1548→    await db.tenantSubscription.updateMany({
  1549→      where: { id: subscription.id, status: { in: ["active", "trialing", "pending_payment"] } },
  1550→      data: { status: "past_due" },
  1551→    });
  1552→    await db.saasRenewalCycle.updateMany({
  1553→      where: { id: cycle.id, state: { notIn: ["COMPLETED", "PAST_DUE"] } },
  1554→      data: { state: "PAST_DUE", failureReason: "Payment failed" },
  1555→    });
  1556→    return { success: false, status: "past_due", reason: "Renewal payment failed" };
  1557→  }
  1558→
  1559→  if (verification.status === "pending") {
  1560→    // Phase 2B.3.14 P2-11: Guarded transition — don't overwrite a concurrently-active subscription.
  1561→    await db.tenantSubscription.updateMany({
  1562→      where: { id: subscription.id, status: { in: ["active", "trialing", "pending_payment"] } },
  1563→      data: { status: "past_due" },
  1564→    });
  1565→    // Phase 2B.3.16: Cycle stays in PAYMENT_CREATING (or transitions back to
  1566→    // PAYMENT_PENDING if it was in PAYMENT_CREATING — the payment intent exists
  1567→    // at the provider, we just don't have confirmation yet). The reconciliation
  1568→    // worker will pick it up and re-verify.
  1569→    // Phase 2B.3.17: Remove silent .catch(() => {}) — emit CRITICAL on failure.
  1570→    const pendingTransition = await db.saasRenewalCycle.updateMany({
  1571→      where: { id: cycle.id, state: "PAYMENT_CREATING" },
  1572→      data: { state: "PAYMENT_PENDING" },
  1573→    });
  1574→    if (pendingTransition.count === 0) {
  1575→      // Not in PAYMENT_CREATING — could be a concurrent transition. Log for visibility.
  1576→      logger.info("saas.payment_creating_to_pending_noop", {
  1577→        cycleId: cycle.id, tenantId, subscriptionId: subscription.id,
  1578→        message: "Cycle was not in PAYMENT_CREATING — may have been concurrently transitioned.",
  1579→      });
  1580→    }
  1581→    return { success: false, status: "past_due", reason: "Payment pending" };
  1582→  }
  1583→
  1584→  // Payment succeeded — transition cycle to PAYMENT_CONFIRMED.
  1585→  // Phase 2B.3.16: Guarded — only transition from PAYMENT_CREATING or PAYMENT_PENDING.
  1586→  // Don't overwrite COMPLETED/RECONCILIATION_REQUIRED.
  1587→  await db.saasRenewalCycle.updateMany({
  1588→    where: { id: cycle.id, state: { in: ["PAYMENT_CREATING", "PAYMENT_PENDING"] } },
  1589→    data: { state: "PAYMENT_CONFIRMED" },
  1590→  });
  1591→
  1592→  // Attempt financial finalization.
  1593→  // Phase 2B.3.14 P1-6: Thread provider paidAt through.
  1594→  // Phase 2B.3.17 P0-4: Pass paymentVerified=true — verifyPayment returned "succeeded".
  1595→  const result = await activateSubscriptionAndPostLedger({
  1596→    subscriptionId: subscription.id,
  1597→    invoiceId: invoice.id,
  1598→    tenantId,
  1599→    userId: "renewal-worker",
  1600→    paidAt: verification.paidAt,
  1601→    paymentVerified: true,
  1602→  });
  1603→
  1604→  if (!result.activated) {
  1605→    // Ledger failed — cycle → RECONCILIATION_REQUIRED, period NOT extended.
  1606→    // Phase 2B.3.14 P2-12: Guarded — don't overwrite COMPLETED.
  1607→    await db.saasRenewalCycle.updateMany({
  1608→      where: { id: cycle.id, state: { not: "COMPLETED" } },
  1609→      data: { state: "RECONCILIATION_REQUIRED", failureReason: "Ledger posting failed" },
  1610→    });
  1611→    return { success: false, status: "financial_pending", reason: "Ledger posting failed during renewal — invoice is reconciliation_required" };
  1612→  }
  1613→
  1614→  // Phase 2B.3.6: Financial finalization succeeded — use the single
  1615→  // authoritative completion function to extend the period + complete the cycle.
  1616→  const completion = await completeSaasRenewalCycle({ invoiceId: invoice.id, tenantId });
  1617→  if (completion.completed) {
  1618→    logger.info("saas.subscription_renewed", { tenantId, subscriptionId: subscription.id, cycleId: cycle.id, newPeriodEnd });
  1619→    return { success: true, status: "active" };
  1620→  } else {
  1621→    // Period extension failed — mark cycle for retry.
  1622→    // Phase 2B.3.14 P2-12: Guarded (already had state guard, but use updateMany for consistency).
  1623→    await db.saasRenewalCycle.updateMany({
  1624→      where: { id: cycle.id, state: { not: "COMPLETED" } },
  1625→      data: { state: "RECONCILIATION_REQUIRED", failureReason: "Period extension failed" },
  1626→    });
  1627→    return { success: false, status: "financial_pending", reason: "Period extension failed after successful ledger posting" };
  1628→  }
  1629→}

  1631→/**
  1632→ * Process all due SaaS subscription renewals.
  1633→ * Called by the reconciliation cron endpoint.
  1634→ */
  1635→export async function processDueSaasRenewals(): Promise<{
  1636→  renewed: number;
  1637→  failed: number;
  1638→  skipped: number;
  1639→}> {
  1640→  const result = { renewed: 0, failed: 0, skipped: 0 };
  1641→  const now = new Date();
  1642→
  1643→  // Phase 2B.3.3 P0-1: Only renew subscriptions that are active or past_due.
  1644→  // PENDING_PAYMENT subscriptions must NOT enter the renewal pipeline —
  1645→  // they haven't paid for their first period yet.
  1646→  // TRIALING is reserved for actual free trials and should not be renewed
  1647→  // by the paid renewal path.
  1648→  const due = await db.tenantSubscription.findMany({
  1649→    where: {
  1650→      currentPeriodEnd: { lt: now },
  1651→      status: { in: ["active", "past_due"] },
  1652→    },
  1653→    select: { tenantId: true },
  1654→  });
  1655→
  1656→  for (const sub of due) {
  1657→    const renewal = await renewSubscription(sub.tenantId);
  1658→    if (renewal.success) {
  1659→      result.renewed++;
  1660→    } else if (renewal.status === "past_due") {
  1661→      result.failed++;
  1662→    } else {
  1663→      result.skipped++;
  1664→    }
  1665→  }
  1666→
  1667→  return result;
  1668→}

  1670→/**
  1671→ * Handle a SaaS subscription payment webhook (from the payment provider).
  1672→ * Idempotent: if the invoice is already paid, returns without re-charging.
  1673→ */
  1674→/**
  1675→ * Phase 2B.3.4: Handle a SaaS payment webhook with provider identity.
  1676→ * The webhook must include the providerKey so we can find the invoice by
  1677→ * (paymentProvider, providerReference) — not just providerReference alone.
  1678→ * This prevents cross-provider collisions if two providers generate
  1679→ * the same reference.
  1680→ */
  1681→export async function handleSaasPaymentWebhook(input: {
  1682→  providerKey: string;
  1683→  providerReference: string;
  1684→  status: "succeeded" | "failed" | "pending";
  1685→  /** Phase 2B.3.14 P1-6: Provider's authoritative payment timestamp from the webhook event. */
  1686→  paidAt?: Date;
  1687→}): Promise<{ handled: boolean }> {
  1688→  const invoice = await db.tenantInvoice.findFirst({
  1689→    where: {
  1690→      paymentProvider: input.providerKey,
  1691→      providerReference: input.providerReference,
  1692→    },
  1693→    include: { subscription: true },
  1694→  });
  1695→  if (!invoice) {
  1696→    logger.warn("saas.webhook_no_match", { providerReference: input.providerReference });
  1697→    return { handled: false };
  1698→  }
  1699→
  1700→  // Phase 2B.3.4: State monotonicity — a paid invoice must NOT be rolled back.
  1701→  // A "failed" webhook arriving after a "succeeded" webhook must NOT change
  1702→  // the invoice from paid → failed.
  1703→  if (invoice.status === "paid") {
  1704→    logger.info("saas.webhook_idempotent_paid", { invoiceId: invoice.id, incomingStatus: input.status });
  1705→    return { handled: true };
  1706→  }
  1707→
  1708→  // Also protect reconciliation_required from being overwritten by "failed"
  1709→  // (the payment was already verified — a later failed event shouldn't undo that)
  1710→  if (invoice.status === "reconciliation_required" && input.status === "failed") {
  1711→    logger.warn("saas.webhook_failed_after_reconciliation", { invoiceId: invoice.id });
  1712→    return { handled: true };
  1713→  }
  1714→
  1715→  if (input.status === "succeeded") {
  1716→    // Phase 2B.3.17 P0-4: Pass paymentVerified=true — webhook reports "succeeded".
  1717→    const result = await activateSubscriptionAndPostLedger({
  1718→      subscriptionId: invoice.subscriptionId,
  1719→      invoiceId: invoice.id,
  1720→      tenantId: invoice.tenantId,
  1721→      userId: "webhook",
  1722→      paidAt: input.paidAt,
  1723→      paymentVerified: true,
  1724→    });
  1725→
  1726→    // Phase 2B.3.10: If financial finalization succeeded, complete the renewal cycle.
  1727→    // If completion FAILS, mark the cycle RECONCILIATION_REQUIRED so the worker
  1728→    // can retry the domain completion (not the payment or ledger).
  1729→    if (result.activated) {
  1730→      const completion = await completeSaasRenewalCycle({ invoiceId: invoice.id, tenantId: invoice.tenantId });
  1731→      if (!completion.completed) {
  1732→        // Financial work is done (invoice paid + ledger posted) but domain
  1733→        // completion failed. Mark the cycle for recovery.
  1734→        const cycle = await db.saasRenewalCycle.findFirst({ where: { invoiceId: invoice.id } });
  1735→        if (cycle && cycle.state !== "COMPLETED") {
  1736→          await db.saasRenewalCycle.updateMany({
  1737→            where: { id: cycle.id, state: { not: "COMPLETED" } },
  1738→            data: { state: "RECONCILIATION_REQUIRED", failureReason: "Domain completion failed after webhook" },
  1739→          }).catch((err) => { logger.error("saas.state_update_failed", { error: err instanceof Error ? err.message : String(err) }); });
  1740→        }
  1741→        logger.error("saas.webhook_completion_failed", {
  1742→          invoiceId: invoice.id,
  1743→          message: "CRITICAL: Financial finalization succeeded but renewal cycle completion failed. Cycle marked RECONCILIATION_REQUIRED.",
  1744→        });
  1745→      }
  1746→    }
  1747→    return { handled: true };
  1748→  }
  1749→
  1750→  if (input.status === "failed") {
  1751→    // Status-guarded: only pending → failed (don't overwrite paid/reconciliation_required)
  1752→    // Phase 2B.3.14 P0: This was already guarded for the invoice, but the subscription
  1753→    // update is now also guarded to prevent overwriting an active subscription.
  1754→    await db.tenantInvoice.updateMany({
  1755→      where: { id: invoice.id, status: "pending" },
  1756→      data: { status: "failed", failureReason: "Webhook reported payment failed" },
  1757→    });
  1758→    await db.tenantSubscription.updateMany({
  1759→      where: { id: invoice.subscriptionId, status: { in: ["pending_payment", "active", "trialing"] } },
  1760→      data: { status: "past_due" },
  1761→    });
  1762→    return { handled: true };
  1763→  }
  1764→
  1765→  return { handled: false };
  1766→}

  1779→// ---------------------------------------------------------------------------
  1780→// Phase 2B.3.1 — SaaS Financial Reconciliation Worker
  1781→// ---------------------------------------------------------------------------
  1782→
  1783→/**
  1784→ * Process SaaS invoices stuck in reconciliation_required.
  1785→ *
  1786→ * For each stuck invoice:
  1787→ *   1. Retry ledgerSaasSubscriptionPayment (idempotent — replays if key exists)
  1788→ *   2. If ledger succeeds → mark invoice as paid + activate subscription
  1789→ *   3. If ledger still fails → leave as reconciliation_required for next run
  1790→ *
  1791→ * Also scans for stale "pending" invoices (older than 5 minutes) whose payment
  1792→ * was confirmed by the provider but never financially finalized.
  1793→ *
  1794→ * Idempotent: ledger replays via idempotencyKey, status-guarded transitions.
  1795→ */
  1796→export async function processDueSaasFinancialReconciliation(): Promise<{
  1797→  retried: number;
  1798→  repaired: number;
  1799→  stillFailing: number;
  1800→}> {
  1801→  const result = { retried: 0, repaired: 0, stillFailing: 0 };
  1802→
  1803→  const staleCutoff = new Date(Date.now() - 5 * 60 * 1000);
  1804→  const due = await db.tenantInvoice.findMany({
  1805→    where: {
  1806→      OR: [
  1807→        { status: "reconciliation_required" },
  1808→        { status: "pending", createdAt: { lt: staleCutoff }, providerReference: { not: null } },
  1809→      ],
  1810→    },
  1811→    include: { subscription: true },
  1812→  });
  1813→
  1814→  for (const invoice of due) {
  1815→    if (!invoice.subscription) continue;
  1816→    result.retried++;
  1817→
  1818→    try {
  1819→      // Phase 2B.3.2 P0-2: For stale "pending" invoices, we MUST verify the
  1820→      // payment with the provider before posting any ledger entry.
  1821→      // A providerReference only proves a payment intent exists — it does NOT
  1822→      // prove the payment succeeded. Only reconciliation_required invoices
  1823→      // have already been verified (the payment was confirmed by the caller
  1824→      // before the ledger attempt failed).
  1825→      let stalePendingPaidAt: Date | undefined;
  1826→      if (invoice.status === "pending") {
  1827→        if (!invoice.providerReference || !invoice.paymentProvider) {
  1828→          // No provider reference — can't verify, skip
  1829→          logger.warn("saas.reconciliation_no_provider_ref", { invoiceId: invoice.id });
  1830→          continue;
  1831→        }
  1832→
  1833→        // Phase 2B.3.3 P0-2: Resolve provider from the invoice's paymentProvider field
  1834→        const provider = getPaymentProviderByKey(invoice.paymentProvider || "mock");
  1835→        const verification = await provider.verifyPayment({
  1836→          providerReference: invoice.providerReference,
  1837→          idempotencyKey: invoice.idempotencyKey,
  1838→        });
  1839→
  1840→        if (verification.status === "failed") {
  1841→          // Phase 2B.3.14 P0-2: GUARDED transition — never overwrite a concurrently-paid invoice.
  1842→          // The invoice was selected as "pending" but may have been finalized by a webhook
  1843→          // while verifyPayment() was in flight. The mutation must be PostgreSQL-atomic.
  1844→          const guarded = await db.tenantInvoice.updateMany({
  1845→            where: { id: invoice.id, status: "pending" },
  1846→            data: { status: "failed", failureReason: "Payment verification failed during reconciliation" },
  1847→          });
  1848→          if (guarded.count === 0) {
  1849→            // The invoice was no longer pending — a concurrent worker advanced it.
  1850→            logger.info("saas.reconciliation_failed_invoice_already_advanced", {
  1851→              invoiceId: invoice.id,
  1852→            });
  1853→            // Do NOT overwrite the subscription — it may have been activated.
  1854→            continue;
  1855→          }
  1856→          // Invoice was actually transitioned to failed — guard the subscription too.
  1857→          await db.tenantSubscription.updateMany({
  1858→            where: { id: invoice.subscriptionId, status: { in: ["pending_payment", "active", "trialing"] } },
  1859→            data: { status: "past_due" },
  1860→          });
  1861→          logger.info("saas.reconciliation_payment_failed", { invoiceId: invoice.id });
  1862→          continue;
  1863→        }
  1864→
  1865→        if (verification.status === "pending") {
  1866→          // Payment still pending — do NOT post revenue, leave as pending
  1867→          logger.info("saas.reconciliation_payment_pending", { invoiceId: invoice.id });
  1868→          continue;
  1869→        }
  1870→
  1871→        // verification.status === "succeeded" — proceed to financial finalization.
  1872→        // Phase 2B.3.14 P1-6: Capture the provider's authoritative paidAt.
  1873→        // This is the critical fix: the stale-pending path now uses the provider's
  1874→        // payment timestamp, not the reconciliation execution time.
  1875→        stalePendingPaidAt = verification.paidAt;
  1876→      }
  1877→
  1878→      // For reconciliation_required invoices, the payment was already verified
  1879→      // by the original caller. The ledger just needs to be retried.
  1880→      // For stale-pending invoices that just verified as succeeded, use the provider paidAt.
  1881→      // Phase 2B.3.17 P0-4: Pass paymentVerified=true when we just verified the
  1882→      // payment (stalePendingPaidAt is set). For reconciliation_required invoices,
  1883→      // paymentVerified is false — but the invoice status is "reconciliation_required"
  1884→      // (not "pending"), so the guard inside activateSubscriptionAndPostLedger allows it.
  1885→      const activated = await activateSubscriptionAndPostLedger({
  1886→        subscriptionId: invoice.subscriptionId,
  1887→        invoiceId: invoice.id,
  1888→        tenantId: invoice.tenantId,
  1889→        userId: "reconciliation-worker",
  1890→        paidAt: stalePendingPaidAt,
  1891→        paymentVerified: !!stalePendingPaidAt,
  1892→      });
  1893→      if (activated.activated) {
  1894→        result.repaired++;
  1895→        logger.info("saas.invoice_reconciled", { invoiceId: invoice.id, subscriptionId: invoice.subscriptionId });
