     1→/**
     2→ * Control Plane — Action Executor v2 (Phase 8.4)
     3→ *
     4→ * CHANGED FROM v1:
     5→ *   v1: SWITCH just transitions session state (no resource operations)
     6→ *   v2: SWITCH performs full resource transaction:
     7→ *     1. Reserve target resource (ownership-safe)
     8→ *     2. Activate target (mark IN_USE, update session)
     9→ *     3. Verify target is usable
    10→ *     4. Release previous resource (ownership-safe)
    11→ *     5. Any failure leaves a recoverable state
    12→ *
    13→ * ACTIVATE also performs:
    14→ *     1. Reserve resource
    15→ *     2. Mark IN_USE
    16→ *     3. Transition session to ACTIVE
    17→ *
    18→ * The action executor NEVER calls provider APIs directly — it operates
    19→ * on ProtocolResource records and session state. The frozen kernel is
    20→ * called only when provisioning is needed (future: via a bridge function).
    21→ */
    22→

    36→// ---------------------------------------------------------------------------
    37→// Create Action
    38→// ---------------------------------------------------------------------------
    39→
    40→export async function createAction(input: {
    41→  sessionId: string;
    42→  decisionId?: string;
    43→  type: ActionType;
    44→  targetResourceId?: string;
    45→  reason?: string;
    46→  policyVersion?: string;
    47→  idempotencyKey?: string; // caller-supplied durable key (Phase 8.5.3)
    48→}): Promise<{ id: string; state: string; idempotencyKey: string }> {
    49→  // Phase 8.5.3: accept caller-supplied idempotency key for durable retry.
    50→  // If not supplied, generate one (backward compat).
    51→  const idempotencyKey = input.idempotencyKey ?? `action-${input.sessionId}-${input.type}-${Date.now()}-${Math.random().toString(36).slice(2, 6)}`;
    52→
    53→  // Phase 8.5.3: idempotent create — if an action with this key already exists,
    54→  // return it instead of creating a duplicate.
    55→  // Phase 8.5.7: Handle concurrent race — if findUnique returns null but create
    56→  // throws a unique constraint violation, re-read the existing action.
    57→  const existing = await db.connectivityAction.findUnique({
    58→    where: { idempotencyKey },
    59→  });
    60→
    61→  if (existing) {
    62→    logger.info("action.idempotent_return", {
    63→      actionId: existing.id,
    64→      idempotencyKey,
    65→      state: existing.state,
    66→    });
    67→    return { id: existing.id, state: existing.state, idempotencyKey };
    68→  }
    69→
    70→  let action;
    71→  try {
    72→    action = await db.connectivityAction.create({
    73→      data: {
    74→        sessionId: input.sessionId,
    75→        decisionId: input.decisionId ?? null,
    76→        type: input.type,
    77→        targetResourceId: input.targetResourceId ?? null,
    78→        state: "PLANNED",
    79→        reason: input.reason ?? null,
    80→        policyVersion: input.policyVersion ?? null,
    81→        idempotencyKey,
    82→      },
    83→    });
    84→  } catch (err: any) {
    85→    // Phase 8.5.7: Handle unique constraint violation from concurrent create
    86→    if (err?.code === "P2002") {
    87→      // Unique constraint on idempotencyKey — another worker created it first.
    88→      // Re-read and return the existing action.
    89→      const concurrent = await db.connectivityAction.findUnique({
    90→        where: { idempotencyKey },
    91→      });
    92→      if (concurrent) {
    93→        logger.info("action.concurrent_idempotent_return", {
    94→          actionId: concurrent.id,
    95→          idempotencyKey,
    96→          state: concurrent.state,
    97→        });
    98→        return { id: concurrent.id, state: concurrent.state, idempotencyKey };
    99→      }
   100→    }
   101→    throw err;
   102→  }
   103→
   104→  logger.info("action.created", {
   105→    actionId: action.id,
   106→    sessionId: input.sessionId,
   107→    type: input.type,
   108→    targetResourceId: input.targetResourceId,
   109→    idempotencyKey,
   110→  });
   111→
   112→  return { id: action.id, state: action.state, idempotencyKey };
   113→}

   115→// ---------------------------------------------------------------------------
   116→// Transition Action State
   117→// ---------------------------------------------------------------------------
   118→
   119→export async function transitionActionState(
   120→  actionId: string,
   121→  toState: ActionState,
   122→  error?: string,
   123→): Promise<{ id: string; state: string }> {
   124→  const action = await db.connectivityAction.findUnique({
   125→    where: { id: actionId },
   126→    select: { state: true },
   127→  });
   128→
   129→  if (!action) {
   130→    throw new Error(`Action not found: ${actionId}`);
   131→  }
   132→
   133→  const currentState = action.state as ActionState;
   134→  const allowed = ACTION_TRANSITIONS[currentState] ?? [];
   135→
   136→  if (!allowed.includes(toState)) {
   137→    throw new Error(
   138→      `Illegal action transition: ${currentState} → ${toState}. Allowed: ${allowed.join(", ")}`,
   139→    );
   140→  }
   141→
   142→  const updateData: Record<string, unknown> = { state: toState };
   143→  if (toState === "EXECUTING") {
   144→    updateData.executedAt = new Date();
   145→  }
   146→  if (toState === "SUCCEEDED" || toState === "FAILED") {
   147→    updateData.completedAt = new Date();
   148→  }
   149→  if (error) {
   150→    updateData.error = error;
   151→  }
   152→
   153→  const updated = await db.connectivityAction.update({
   154→    where: { id: actionId },
   155→    data: updateData,
   156→  });
   157→
   158→  logger.info("action.transitioned", { actionId, from: currentState, to: toState });
   159→  return { id: updated.id, state: updated.state };
   160→}

   162→// ---------------------------------------------------------------------------
   163→// Execute Action
   164→// ---------------------------------------------------------------------------
   165→
   166→/**
   167→ * Execute a planned action. This is the real resource-driven execution path.
   168→ *
   169→ * SWITCH transaction:
   170→ *   1. Reserve target resource (ownership-safe)
   171→ *      → fail: action FAILED, session unchanged, target stays AVAILABLE
   172→ *   2. Activate target (mark IN_USE)
   173→ *   3. Verify target is usable
   174→ *      → fail: release target, action FAILED, session unchanged
   175→ *   4. Transition session → SWITCHING → ACTIVE
   176→ *   5. Update session.activeResourceId = target
   177→ *   6. Release previous resource (ownership-safe)
   178→ *      → fail: session is still on target (correct), mark RECONCILIATION_REQUIRED
   179→ *
   180→ * Key invariant: releasing the old resource MUST NOT invalidate the new one.
   181→ */
   182→export async function executeAction(actionId: string): Promise<{
   183→  status: "succeeded" | "failed" | "reconciliation_required";
   184→  error?: string;
   185→}> {
   186→  const action = await db.connectivityAction.findUnique({
   187→    where: { id: actionId },
   188→    include: { session: true },
   189→  });
   190→
   191→  if (!action) {
   192→    throw new Error(`Action not found: ${actionId}`);
   193→  }
   194→
   195→  if (action.state !== "PLANNED") {
   196→    throw new Error(`Action state is ${action.state}, expected PLANNED`);
   197→  }
   198→
   199→  try {
   200→    // Step 1: Authorize
   201→    await transitionActionState(actionId, "AUTHORIZED");
   202→
   203→    // Step 2: Execute
   204→    await transitionActionState(actionId, "EXECUTING");
   205→
   206→    const session = action.session;
   207→    const targetResourceId = action.targetResourceId;
   208→
   209→    if (!targetResourceId) {
   210→      throw new Error("SWITCH/ACTIVATE action requires targetResourceId");
   211→    }
   212→
   213→    // Step 3: Execute based on action type
   214→    switch (action.type) {
   215→      // -------------------------------------------------------------------
   216→      // ACTIVATE: reserve → mark IN_USE → session ACTIVE
   217→      // -------------------------------------------------------------------
   218→      case "ACTIVATE": {
   219→        // 3a. Reserve the target resource
   220→        const reserveResult = await reserveResource(targetResourceId, session.id);
   221→        if (!reserveResult.reserved) {
   222→          throw new Error(`Failed to reserve resource ${targetResourceId}: ${reserveResult.reason}`);
   223→        }
   224→
   225→        // 3b. Resolve resource binding via kernel bridge
   226→        // Phase 8.5.8: tenantId is DERIVED from the resource's capability, not caller-supplied
   227→        const bridgeResult = await resolveResourceBinding({
   228→          protocolResourceId: targetResourceId,
   229→          subjectId: session.subjectId,
   230→        });
   231→
   232→        if (bridgeResult.status === "failed") {
   233→          await releaseResource(targetResourceId, session.id);
   234→          throw new Error(`Kernel bridge failed: ${bridgeResult.error}`);
   235→        }
   236→        if (bridgeResult.status === "reconciliation_required") {
   237→          await releaseResource(targetResourceId, session.id);
   238→          throw new Error(`Kernel bridge requires reconciliation: ${bridgeResult.error}`);
   239→        }
   240→
   241→        // 3c. Mark resource as IN_USE — fail closed
   242→        const activateResult = await markResourceInUse(targetResourceId, session.id);
   243→        if (!activateResult.activated) {
   244→          await releaseResource(targetResourceId, session.id);
   245→          throw new Error(`Failed to mark resource IN_USE: ${activateResult.reason}`);
   246→        }
   247→
   248→        // 3d. Verify via kernel bridge (USABLE | NOT_USABLE | UNKNOWN)
   249→        const verifyResult = await verifyResourceUsable(targetResourceId, session.id);
   250→        if (verifyResult.status === "NOT_USABLE") {
   251→          await releaseResource(targetResourceId, session.id);
   252→          throw new Error(`Resource verification failed: ${verifyResult.reason}`);
   253→        }
   254→
   255→        // Phase 8.5.9: UNKNOWN → release target, do NOT activate session, RECONCILIATION_REQUIRED
   256→        if (verifyResult.status === "UNKNOWN") {
   257→          await releaseResource(targetResourceId, session.id);
   258→          await transitionActionState(actionId, "RECONCILIATION_REQUIRED", `Verification UNKNOWN: ${verifyResult.reason}`);
   259→          logger.warn("action.activate_unknown_verification", {
   260→            actionId, targetResourceId, reason: verifyResult.reason,
   261→          });
   262→          return { status: "failed", error: `Verification UNKNOWN — target released, reconciliation required: ${verifyResult.reason}` };
   263→        }
   264→
   265→        // 3e. Update session with entitlement link (only if USABLE)
   266→        await db.connectivitySession.update({
   267→          where: { id: session.id },
   268→          data: {
   269→            activeResourceId: targetResourceId,
   270→            entitlementId: bridgeResult.entitlementId,
   271→            startedAt: new Date(),
   272→            lastObservedAt: new Date(),
   273→          },
   274→        });
   275→
   276→        // 3f. Transition session to ACTIVE.
   277→        // Phase 8.6 runtime fix: the session state machine requires
   278→        // PLANNED → DISCOVERING → ACTIVE (PLANNED → ACTIVE is illegal). This
   279→        // bug was invisible to static tests and only surfaced when the
   280→        // DB-backed runtime test exercised ACTIVATE from a freshly-created
   281→        // (PLANNED) session.
   282→        if (session.state === "PLANNED" || session.state === "DISCOVERING" || session.state === "RESERVED") {
   283→          if (session.state === "PLANNED") {
   284→            await transitionSessionState(session.id, "DISCOVERING");
   285→          }
   286→          await transitionSessionState(session.id, "ACTIVE");
   287→        }
   288→
   289→        // Phase 8.5.10: Verify the active connectivity invariant
   290→        const invariant = await assertActiveConnectivityInvariant(session.id);
   291→        if (!invariant.valid) {
   292→          await transitionActionState(actionId, "RECONCILIATION_REQUIRED",
   293→            `Invariant check failed after ACTIVATE: ${invariant.violations.join(", ")}`);
   294→          logger.error("action.activate_invariant_failed", {
   295→            actionId, sessionId: session.id, violations: invariant.violations,
   296→          });
   297→          return { status: "failed", error: `Invariant failed: ${invariant.violations.join(", ")}` };
   298→        }
   299→
   300→        logger.info("action.activate_succeeded", {
   301→          actionId, sessionId: session.id, resourceId: targetResourceId,
   302→        });
   303→
   304→        // Phase 8.6.6: Decoupled re-observation. Previously this called
   305→        // probeAndIngest() inline, coupling the command path to the telemetry
   306→        // path and making action completion depend on a potentially slow
   307→        // provider probe. Now it emits a REOBSERVE_REQUESTED event that the
   308→        // observation worker processes asynchronously — same closed-loop
   309→        // behavior (OBSERVE→...→ACT→VERIFY→OBSERVE AGAIN) without blocking.
   310→        await emitReobserveRequest(targetResourceId, session.id);
   311→        break;
   312→      }
   313→
   314→      // -------------------------------------------------------------------
   315→      // SWITCH: reserve target → activate → verify → update session → release old
   316→      // -------------------------------------------------------------------
   317→      case "SWITCH": {
   318→        if (session.state !== "ACTIVE" && session.state !== "DEGRADED") {
   319→          throw new Error(`SWITCH requires session to be ACTIVE or DEGRADED, got ${session.state}`);
   320→        }
   321→
   322→        const previousResourceId = session.activeResourceId;
   323→
   324→        // 3a. Transition session to SWITCHING
   325→        await transitionSessionState(session.id, "SWITCHING");
   326→
   327→        // 3b. Reserve the target resource (ownership-safe)
   328→        const reserveResult = await reserveResource(targetResourceId, session.id);
   329→        if (!reserveResult.reserved) {
   330→          await transitionSessionState(session.id, session.state === "DEGRADED" ? "DEGRADED" : "ACTIVE");
   331→          throw new Error(`Failed to reserve target resource ${targetResourceId}: ${reserveResult.reason}`);
   332→        }
   333→
   334→        // 3c. Resolve target binding via kernel bridge
   335→        // Phase 8.5.8: tenantId derived from resource, not caller-supplied
   336→        const bridgeResult = await resolveResourceBinding({
   337→          protocolResourceId: targetResourceId,
   338→          subjectId: session.subjectId,
   339→        });
   340→
   341→        if (bridgeResult.status === "failed") {
   342→          await releaseResource(targetResourceId, session.id);
   343→          await transitionSessionState(session.id, session.state === "DEGRADED" ? "DEGRADED" : "ACTIVE");
   344→          throw new Error(`Kernel bridge failed for target: ${bridgeResult.error}`);
   345→        }
   346→        if (bridgeResult.status === "reconciliation_required") {
   347→          await releaseResource(targetResourceId, session.id);
   348→          await transitionSessionState(session.id, session.state === "DEGRADED" ? "DEGRADED" : "ACTIVE");
   349→          throw new Error(`Kernel bridge requires reconciliation for target: ${bridgeResult.error}`);
   350→        }
   351→
   352→        // 3d. Mark target as IN_USE — fail closed
   353→        const activateResult = await markResourceInUse(targetResourceId, session.id);
   354→        if (!activateResult.activated) {
   355→          await releaseResource(targetResourceId, session.id);
   356→          await transitionSessionState(session.id, session.state === "DEGRADED" ? "DEGRADED" : "ACTIVE");
   357→          throw new Error(`Failed to mark target IN_USE: ${activateResult.reason}`);
   358→        }
   359→
   360→        // 3e. Verify target is usable (USABLE | NOT_USABLE | UNKNOWN)
   361→        const verifyResult = await verifyResourceUsable(targetResourceId, session.id);
   362→        if (verifyResult.status === "NOT_USABLE") {
   363→          await releaseResource(targetResourceId, session.id);
   364→          await transitionSessionState(session.id, session.state === "DEGRADED" ? "DEGRADED" : "ACTIVE");
   365→          throw new Error(`Target verification failed: ${verifyResult.reason}`);
   366→        }
   367→        if (verifyResult.status === "UNKNOWN") {
   368→          // Phase 8.5.8: UNKNOWN → RECONCILIATION_REQUIRED (NOT SUCCEEDED)
   369→          // The session stays on the old resource — we don't switch to an unverified target.
   370→          await releaseResource(targetResourceId, session.id);
   371→          await transitionSessionState(session.id, session.state === "DEGRADED" ? "DEGRADED" : "ACTIVE");
   372→          await transitionActionState(actionId, "RECONCILIATION_REQUIRED", `Switch verification UNKNOWN: ${verifyResult.reason}`);
   373→          logger.warn("action.switch_verification_unknown", {
   374→            actionId, targetResourceId, reason: verifyResult.reason,
   375→          });
   376→          return { status: "failed", error: `Switch verification UNKNOWN — reconciliation required: ${verifyResult.reason}` };
   377→        }
   378→
   379→        // 3f. Atomically update session to point to new resource + entitlement
   380→        await db.connectivitySession.update({
   381→          where: { id: session.id },
   382→          data: {
   383→            activeResourceId: targetResourceId,
   384→            entitlementId: bridgeResult.entitlementId,
   385→            lastObservedAt: new Date(),
   386→          },
   387→        });
   388→
   389→        // 3g. Transition session back to ACTIVE
   390→        await transitionSessionState(session.id, "ACTIVE");
   391→
   392→        // 3g. Release the previous resource (ownership-safe)
   393→        // IMPORTANT: failure here does NOT invalidate the new resource.
   394→        // Phase 8.5.8: Persist a durable reconciliation marker on failure.
   395→        let oldReleaseFailed = false;
   396→        if (previousResourceId && previousResourceId !== targetResourceId) {
   397→          const releaseResult = await releaseResource(previousResourceId, session.id);
   398→          if (!releaseResult.released) {
   399→            oldReleaseFailed = true;
   400→            logger.warn("action.switch_old_release_failed", {
   401→              actionId,
   402→              sessionId: session.id,
   403→              oldResourceId: previousResourceId,
   404→              reason: releaseResult.reason,
   405→            });
   406→          }
   407→        }
   408→
   409→        logger.info("action.switch_succeeded", {
   410→          actionId,
   411→          sessionId: session.id,
   412→          fromResource: previousResourceId,
   413→          toResource: targetResourceId,
   414→          oldReleaseFailed,
   415→        });
   416→
   417→        // Phase 8.5.8: If old resource release failed, mark RECONCILIATION_REQUIRED
   418→        // instead of SUCCEEDED. The switch itself succeeded (session is on target),
   419→        // but the old resource is leaked and needs cleanup.
   420→        // Phase 8.6.6: Return "reconciliation_required" (NOT "succeeded") so the
   421→        // decision-executor propagates the correct terminal state to the Decision.
   422→        // Overloading "succeeded" to mean "switch worked but cleanup remains"
   423→        // was an inconsistent state model.
   424→        if (oldReleaseFailed) {
   425→          await transitionActionState(actionId, "RECONCILIATION_REQUIRED", `Old resource ${previousResourceId} release failed — session is on target but cleanup needed`);
   426→          return { status: "reconciliation_required", error: `Switch succeeded but old resource release failed — reconciliation required` };
   427→        }
   428→
   429→        // Phase 8.5.10: Verify the active connectivity invariant after SWITCH
   430→        const switchInvariant = await assertActiveConnectivityInvariant(session.id);
   431→        if (!switchInvariant.valid) {
   432→          await transitionActionState(actionId, "RECONCILIATION_REQUIRED",
   433→            `Invariant check failed after SWITCH: ${switchInvariant.violations.join(", ")}`);
   434→          logger.error("action.switch_invariant_failed", {
   435→            actionId, sessionId: session.id, violations: switchInvariant.violations,
   436→          });
   437→          return { status: "failed", error: `Invariant failed: ${switchInvariant.violations.join(", ")}` };
   438→        }
   439→
   440→        // Phase 8.6.6: Decoupled re-observation — emit REOBSERVE_REQUESTED
   441→        // instead of inline probeAndIngest. The observation worker processes
   442→        // it asynchronously, keeping the command path (action completion)
   443→        // separate from the telemetry path (observation).
   444→        await emitReobserveRequest(targetResourceId, session.id);
   445→        break;
   446→      }
   447→
   448→      // -------------------------------------------------------------------
   449→      // SUSPEND: session ACTIVE → DEGRADED
   450→      // -------------------------------------------------------------------
   451→      case "SUSPEND": {
   452→        if (session.state === "ACTIVE") {
   453→          await transitionSessionState(session.id, "DEGRADED");
   454→        }
   455→        break;
   456→      }
   457→
   458→      // -------------------------------------------------------------------
   459→      // RESUME: session DEGRADED → ACTIVE
   460→      // -------------------------------------------------------------------
   461→      case "RESUME": {
   462→        if (session.state === "DEGRADED") {
   463→          await transitionSessionState(session.id, "ACTIVE");
   464→        }
   465→        break;
   466→      }
   467→
   468→      // -------------------------------------------------------------------
   469→      // RELEASE: release resource + session ENDED
   470→      // -------------------------------------------------------------------
   471→      case "RELEASE": {
   472→        if (session.state !== "ENDED") {
   473→          // Release the active resource (ownership-safe)
   474→          if (session.activeResourceId) {
   475→            await releaseResource(session.activeResourceId, session.id);
   476→          }
   477→          await transitionSessionState(session.id, "ENDED");
   478→        }
   479→        break;
   480→      }
   481→
   482→      // -------------------------------------------------------------------
   483→      // Not yet implemented
   484→      // -------------------------------------------------------------------
   485→      case "DISCOVER":
   486→      case "RESERVE":
   487→      case "RENEW":
   488→      case "TRANSFER":
   489→        logger.info("action.type_not_yet_implemented", { actionId, type: action.type });
   490→        break;
   491→
   492→      default:
   493→        throw new Error(`Unknown action type: ${action.type}`);
   494→    }
   495→
   496→    // Step 4: Mark success
   497→    await transitionActionState(actionId, "SUCCEEDED");
   498→
   499→    logger.info("action.executed", { actionId, type: action.type, sessionId: session.id });
   500→
   501→    return { status: "succeeded" };
   502→  } catch (err) {
   503→    const errorMsg = err instanceof Error ? err.message : String(err);
   504→    await transitionActionState(actionId, "FAILED", errorMsg);
   505→
   506→    logger.error("action.failed", { actionId, error: errorMsg });
   507→
   508→    return { status: "failed", error: errorMsg };
   509→  }
   510→}

   526→// ---------------------------------------------------------------------------
   527→// Phase 8.5.3: Durable Action Recovery Worker
   528→// ---------------------------------------------------------------------------
   529→
   530→/**
   531→ * Recover EXECUTING actions that were interrupted by a crash.
   532→ *
   533→ * An action in EXECUTING state means the process died mid-execution.
   534→ * The recovery worker inspects the session + resource state and converges:
   535→ *
   536→ *   - If the target resource is IN_USE and owned by the session → SUCCEEDED
   537→ *   - If the target resource is RESERVED but not IN_USE → release + FAILED
   538→ *   - If the target resource is AVAILABLE (never reserved) → FAILED
   539→ *   - If the session is SWITCHING and target is IN_USE → complete the switch
   540→ *   - Otherwise → RECONCILIATION_REQUIRED
   541→ *
   542→ * This is the control-plane equivalent of the kernel's reconcileProvisioning().
   543→ */
   544→export async function recoverStaleActions(): Promise<{
   545→  recovered: number;
   546→  succeeded: number;
   547→  failed: number;
   548→  reconciliationRequired: number;
   549→}> {
   550→  // Phase 8.5.10: Recovery with stale predicate + lease expiry.
   551→  //
   552→  // 1. Only claim EXECUTING actions older than RECOVERY_TIMEOUT_MS
   553→  //    (prevents stealing actions from a legitimate executor that's still running)
   554→  // 2. Also reclaim expired RECOVERY_CLAIMED actions (recoveryClaimExpiresAt < now)
   555→  //    (allows a dead recovery worker's claims to be reclaimed)
   556→  // 3. Generate a unique claim ID for this worker invocation
   557→  // 4. Atomically transition to RECOVERY_CLAIMED with the claim ID + lease expiry
   558→  // 5. Query ONLY actions claimed by THIS worker
   559→  const claimId = `recovery-${Date.now()}-${Math.random().toString(36).slice(2, 10)}`;
   560→  const now = new Date();
   561→  const staleCutoff = new Date(now.getTime() - RECOVERY_TIMEOUT_MS);
   562→  const leaseExpiry = new Date(now.getTime() + RECOVERY_LEASE_MS);
   563→
   564→  // Claim stale EXECUTING actions + expired RECOVERY_CLAIMED actions
   565→  const claimResult = await db.connectivityAction.updateMany({
   566→    where: {
   567→      OR: [
   568→        // Stale EXECUTING: running longer than RECOVERY_TIMEOUT_MS
   569→        {
   570→          state: "EXECUTING",
   571→          executedAt: { lt: staleCutoff },
   572→        },
   573→        // Expired RECOVERY_CLAIMED: a previous recovery worker's lease has expired
   574→        {
   575→          state: "RECOVERY_CLAIMED",
   576→          recoveryClaimExpiresAt: { lt: now },
   577→        },
   578→      ],
   579→    },
   580→    data: {
   581→      state: "RECOVERY_CLAIMED",
   582→      recoveryClaimId: claimId,
   583→      recoveryClaimedAt: now,
   584→      recoveryClaimExpiresAt: leaseExpiry,
   585→    },
   586→  });
   587→
   588→  if (claimResult.count === 0) {
   589→    return { recovered: 0, succeeded: 0, failed: 0, reconciliationRequired: 0 };
   590→  }
   591→
   592→  // Query ONLY the actions claimed by this worker
   593→  const staleActions = await db.connectivityAction.findMany({
   594→    where: { recoveryClaimId: claimId },
   595→    include: { session: true },
   596→  });
   597→
   598→  let succeeded = 0;
   599→  let failed = 0;
   600→  let reconciliationRequired = 0;
   601→
   602→  for (const action of staleActions) {
   603→    const session = action.session;
   604→    const targetResourceId = action.targetResourceId;
   605→
   606→    if (!targetResourceId) {
   607→      await transitionActionState(action.id, "FAILED", "No targetResourceId on EXECUTING action");
   608→      failed++;
   609→      continue;
   610→    }
   611→
   612→    // Check the target resource state
   613→    const resource = await db.protocolResource.findUnique({
   614→      where: { id: targetResourceId },
   615→      select: { state: true, reservedBy: true, providerBindingId: true },
   616→    });
   617→
   618→    if (!resource) {
   619→      await transitionActionState(action.id, "FAILED", "Target resource not found during recovery");
   620→      failed++;
   621→      continue;
   622→    }
   623→
   624→    if (resource.state === "IN_USE" && resource.reservedBy === session.id) {
   625→      // Phase 8.5.9: Use the kernel bridge to validate binding ownership
   626→      // (same authority as normal execution, not a weaker manual lookup)
   627→      const bridgeResult = await resolveResourceBinding({
   628→        protocolResourceId: targetResourceId,
   629→        subjectId: session.subjectId,
   630→      });
   631→
   632→      if (bridgeResult.status === "failed") {
   633→        await releaseResource(targetResourceId, session.id);
   634→        if (session.state === "SWITCHING") {
   635→          await transitionSessionState(session.id, "ACTIVE");
   636→        }
   637→        await transitionActionState(action.id, "FAILED", `Recovery binding validation failed: ${bridgeResult.error}`);
   638→        failed++;
   639→        continue;
   640→      }
   641→
   642→      if (bridgeResult.status === "reconciliation_required") {
   643→        if (session.state === "SWITCHING") {
   644→          await transitionSessionState(session.id, "ACTIVE");
   645→        }
   646→        await transitionActionState(action.id, "RECONCILIATION_REQUIRED", `Recovery binding reconciliation: ${bridgeResult.error}`);
   647→        reconciliationRequired++;
   648→        continue;
   649→      }
   650→
   651→      // Bridge validated — now verify provider truth
   652→      const verifyResult = await verifyResourceUsable(targetResourceId, session.id);
   653→
   654→      if (verifyResult.status === "NOT_USABLE") {
   655→        await releaseResource(targetResourceId, session.id);
   656→        if (session.state === "SWITCHING") {
   657→          await transitionSessionState(session.id, "ACTIVE");
   658→        }
   659→        await transitionActionState(action.id, "FAILED", `Recovery verification NOT_USABLE: ${verifyResult.reason}`);
   660→        failed++;
   661→        continue;
   662→      }
   663→
   664→      if (verifyResult.status === "UNKNOWN") {
   665→        if (session.state === "SWITCHING") {
   666→          await transitionSessionState(session.id, "ACTIVE");
   667→        }
   668→        await transitionActionState(action.id, "RECONCILIATION_REQUIRED", `Recovery verification UNKNOWN: ${verifyResult.reason}`);
   669→        reconciliationRequired++;
   670→        continue;
   671→      }
   672→
   673→      // USABLE — complete the switch if needed
   674→      if (session.state === "SWITCHING") {
   675→        const previousResourceId = session.activeResourceId;
   676→
   677→        // entitlementId comes from the validated bridge result
   678→        await db.connectivitySession.update({
   679→          where: { id: session.id },
   680→          data: {
   681→            activeResourceId: targetResourceId,
   682→            entitlementId: bridgeResult.entitlementId,
   683→            lastObservedAt: new Date(),
   684→          },
   685→        });
   686→        await transitionSessionState(session.id, "ACTIVE");
   687→
   688→        // Phase 8.5.9: Recovery old-release failure → RECONCILIATION_REQUIRED
   689→        // (not swallowed .catch — same invariant as normal SWITCH)
   690→        let oldReleaseFailed = false;
   691→        if (previousResourceId && previousResourceId !== targetResourceId) {
   692→          const releaseResult = await releaseResource(previousResourceId, session.id);
   693→          if (!releaseResult.released) {
   694→            oldReleaseFailed = true;
   695→            logger.warn("recovery.switch_old_release_failed", {
   696→              actionId: action.id,
   697→              oldResourceId: previousResourceId,
   698→              reason: releaseResult.reason,
   699→            });
   700→          }
   701→        }
   702→
   703→        if (oldReleaseFailed) {
   704→          await transitionActionState(action.id, "RECONCILIATION_REQUIRED", `Recovery: old resource ${previousResourceId} release failed — session on target but cleanup needed`);
   705→          reconciliationRequired++;
   706→          continue;
   707→        }
   708→      }
   709→
   710→      // Phase 8.5.10: Verify invariant before declaring recovery SUCCEEDED
   711→      const recoveryInvariant = await assertActiveConnectivityInvariant(session.id);
   712→      if (!recoveryInvariant.valid) {
   713→        await transitionActionState(action.id, "RECONCILIATION_REQUIRED",
   714→          `Recovery invariant failed: ${recoveryInvariant.violations.join(", ")}`);
   715→        reconciliationRequired++;
   716→        logger.warn("action.recovered_invariant_failed", {
   717→          actionId: action.id, violations: recoveryInvariant.violations,
   718→        });
   719→        continue;
   720→      }
   721→
   722→      await transitionActionState(action.id, "SUCCEEDED");
   723→      succeeded++;
   724→      logger.info("action.recovered_succeeded", { actionId: action.id, targetResourceId, verified: true, invariantChecked: true });
   725→    } else if (resource.state === "RESERVED" && resource.reservedBy === session.id) {
   726→      await releaseResource(targetResourceId, session.id);
   727→      if (session.state === "SWITCHING") {
   728→        await transitionSessionState(session.id, "ACTIVE");
   729→      }
   730→      await transitionActionState(action.id, "FAILED", "Resource was RESERVED but never IN_USE — recovered by releasing");
   731→      failed++;
   732→      logger.info("action.recovered_failed_reserved", { actionId: action.id, targetResourceId });
   733→    } else {
   734→      await transitionActionState(action.id, "RECONCILIATION_REQUIRED", `Resource in unexpected state: ${resource.state}, reservedBy: ${resource.reservedBy}`);
   735→      reconciliationRequired++;
   736→      logger.warn("action.recovered_reconciliation_required", {
   737→        actionId: action.id,
   738→        targetResourceId,
   739→        resourceState: resource.state,
   740→        reservedBy: resource.reservedBy,
   741→      });
   742→    }
   743→  }
   744→
   745→  const recovered = staleActions.length;
   746→  logger.info("action.recovery_completed", {
   747→    recovered, succeeded, failed, reconciliationRequired, claimId,
   748→  });
   749→
   750→  return { recovered, succeeded, failed, reconciliationRequired };
   751→}

   753→// ---------------------------------------------------------------------------
   754→// Phase 8.6.6: Decoupled re-observation (REOBSERVE_REQUESTED event)
   755→// ---------------------------------------------------------------------------
   756→
   757→/**
   758→ * Emit a REOBSERVE_REQUESTED event for a resource. The observation worker
   759→ * processes this asynchronously, probing the resource and ingesting a fresh
   760→ * measurement.
   761→ *
   762→ * This replaces the Phase 8.6.5 inline `triggerReobservation` which called
   763→ * `probeAndIngest()` synchronously inside the action executor. That coupled
   764→ * the command path (action completion) to the telemetry path (observation),
   765→ * making action completion depend on a potentially slow provider probe.
   766→ *
   767→ * The event is idempotent (deduped by idempotencyKey) so multiple emissions
   768→ * for the same resource/session collapse to one observation.
   769→ *
   770→ * Failures are non-fatal — the observation worker will also probe on its
   771→ * regular cycle. A re-observation failure must NEVER corrupt the session or
   772→ * the action that just succeeded.
   773→ */
   774→async function emitReobserveRequest(resourceId: string, sessionId: string): Promise<void> {
   775→  try {
   776→    // Idempotent key: one reobservation per (resource, session, action-completion).
   777→    // If the same resource is reobserved within the dedup window, the event
   778→    // collapses to one.
   779→    const idempotencyKey = `reobserve-${resourceId}-${sessionId}-${Date.now().toString(36)}`;
   780→    await db.reevaluationEvent.create({
   781→      data: {
   782→        type: "MEASUREMENT_RECEIVED", // reuse the measurement event type — the observation worker drains it
   783→        resourceId,
   784→        sessionId,
   785→        subjectId: null,
   786→        payload: JSON.stringify({ reobserve: true, resourceId, sessionId, reason: "post-action-reobservation" }),
   787→        idempotencyKey,
   788→        state: "PENDING",
   789→      },
   790→    }).catch(() => {
   791→      // Idempotent — if the key already exists, that's fine.
   792→    });
   793→    logger.info("action.reobserve_requested", { resourceId, sessionId });
   794→  } catch (err) {
   795→    // Non-fatal — the observation worker will probe the resource on its cycle.
   796→    logger.warn("action.reobserve_request_failed", {
   797→      resourceId, sessionId,
   798→      error: err instanceof Error ? err.message : String(err),
   799→    });
   800→  }
   801→}
