// =============================================================================
// eSIM Marketplace — Prisma Schema
// -----------------------------------------------------------------------------
// DATABASE: PostgreSQL (Neon) is the canonical database for development,
// staging, and production. SQLite is NOT supported. Money is stored as Int
// (minor units) — never float.
// =============================================================================

generator client {
  provider        = "prisma-client-js"
  binaryTargets   = ["native", "rhel-openssl-3.0.x", "linux-arm64-openssl-3.0.x"]
}

datasource db {
  provider  = "postgresql"
  url       = env("DATABASE_URL")
  directUrl = env("DIRECT_URL")
}

// ---------------------------------------------------------------------------
// User & Authentication
// ---------------------------------------------------------------------------

model User {
  id            String   @id @default(cuid())
  email         String   @unique
  name          String?
  passwordHash  String
  role          String   @default("customer") // "customer" | "admin"
  isDemo        Boolean  @default(false) // demo accounts (quick-login)
  emailVerified DateTime?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  sessions              Session[]
  orders                Order[]
  esims                 Esim[]
  payments              Payment[]
  topUps                TopUp[]
  auditLogs             AuditLog[]
  passwordResetTokens   PasswordResetToken[]
  emailVerificationTokens EmailVerificationToken[]
  approvedFrom          WaitlistEntry?
  approvedWaitlist      WaitlistEntry[] @relation("WaitlistApprover")
  orgMemberships        OrganizationMember[]
  virtualNumbers        VirtualNumber[]
  numberSubscriptions   NumberSubscription[]
  financialTxns         FinancialTransaction[]
  referral              Referral?             @relation("ReferralUser")
  referralUses          ReferralUse[]          @relation("RefereeUser")
  credit                CustomerCredit?
  creditTxns            CreditTransaction[]
}

model WaitlistEntry {
  id          String   @id @default(cuid())
  email       String   @unique
  name        String?
  status      String   @default("pending") // pending | approved | rejected
  note        String?  // admin note
  approvedById String? // admin user who approved
  approvedAt  DateTime?
  createdUserId String? @unique // the User created when approved (1:1)
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  approvedBy  User?    @relation("WaitlistApprover", fields: [approvedById], references: [id])
  createdUser User?    @relation(fields: [createdUserId], references: [id])

  @@index([status])
}

// ---------------------------------------------------------------------------
// Plans (canonical marketplace model — never provider-native shape)
// ---------------------------------------------------------------------------

model Plan {
  id              String   @id @default(cuid())
  providerId      String   // internal provider key, e.g. "mock"
  providerPlanId  String   // provider's native plan id
  name            String
  description     String?
  country         String
  countryCode     String   // ISO 3166-1 alpha-2
  region          String   // e.g. "Africa", "Europe", "North America"
  dataAmount      Int      // in MB (1024 = 1 GB)
  dataUnit        String   @default("MB")
  validityDays    Int
  // Money — integer minor units (cents). See lib/money.ts.
  price           Int      // retail price in minor units
  wholesalePrice  Int      // internal cost, NEVER exposed to customers
  currency        String   @default("USD")
  coverage        String?  // comma-separated countries / description
  networks        String?  // JSON string: ["MTN","Vodafone"]
  roaming         Boolean  @default(false)
  hotspot         Boolean  @default(true)
  speed           String?  // e.g. "4G/5G"
  topUpSupported  Boolean  @default(true)
  status          String   @default("active") // "active" | "inactive"
  // Pricing rule that produced `price` (audit). JSON string.
  pricingRule     String?
  // Provider-native payload (never sent to client). JSON string.
  metadata        String?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  orders          Order[]

  @@unique([providerId, providerPlanId])
  @@index([countryCode])
  @@index([region])
  @@index([status])
}

// ---------------------------------------------------------------------------
// Orders
// ---------------------------------------------------------------------------

model Order {
  id               String   @id @default(cuid())
  userId           String
  planId           String?  // nullable for virtual-number orders (no Plan row)
  productType      String   @default("esim") // esim | virtual_number
  // State machine: PLAN_SELECTED | CHECKOUT_CREATED | PAYMENT_PENDING |
  // PAYMENT_CONFIRMED | ESIM_PROVISIONING | ESIM_PROVISIONED | COMPLETED |
  // PAYMENT_FAILED | PROVISIONING_FAILED | CANCELLED | REFUNDED
  status           String   @default("PLAN_SELECTED")
  amount           Int      // total charged (minor units)
  currency         String   @default("USD")
  paymentStatus    String   @default("pending") // pending | succeeded | failed | refunded
  paymentProvider  String?
  paymentReference String?  // provider's payment intent / charge id
  providerOrderId  String?  // eSIM provider order id
  // Idempotency: prevents duplicate orders / payments / provisioning on retry.
  idempotencyKey   String   @unique
  // Snapshot of plan at purchase time (so historical orders stay accurate).
  planSnapshot     String?  // JSON string
  failureReason    String?
  createdAt        DateTime @default(now())
  updatedAt        DateTime @updatedAt

  user             User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  plan             Plan?    @relation(fields: [planId], references: [id])
  payments         Payment[]
  esim             Esim?
  corporateOrder   CorporateOrder?
  virtualNumber    VirtualNumber? @relation("VirtualNumberOrder")
  financialTxns    FinancialTransaction[] @relation("FinancialOrder")
  promoRedemption  PromoRedemption?       @relation("PromoOrder")

  @@index([userId])
  @@index([status])
  @@index([paymentStatus])
}

  @@index([orderId])
  @@index([status])
}

// ---------------------------------------------------------------------------
// eSIMs

model Esim {
  id              String    @id @default(cuid())
  userId          String
  orderId         String    @unique // one eSIM per order (1:1)
  provider        String
  providerESIMId  String?   // provider's native esim id
  iccid           String?   // 19-20 digit Integrated Circuit Card ID
  smdpAddress     String?   // SM-DP+ server address
  activationCode  String?   // LPA activation code
  matchId         String?   // optional confirmation code
  qrCode          String?   // generated QR payload (LPA string)
  status          String    @default("pending") // pending|active|expired|suspended|exhausted|cancelled
  dataAmount      Int       // total allowance in MB
  dataRemaining   Int       // remaining in MB
  validityDays    Int
  expiresAt       DateTime?
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt

  user            User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  order           Order     @relation(fields: [orderId], references: [id], onDelete: Cascade)
  usages          Usage[]
  topUps          TopUp[]
  orgAssignments  OrganizationESIM[]

  @@index([userId])
  @@index([status])
  @@index([iccid])
}

// ---------------------------------------------------------------------------
// Financial Ledger — immutable record of all financial events.
// Financial truth must not depend exclusively on mutable order records.
// ---------------------------------------------------------------------------

model FinancialTransaction {
  id              String   @id @default(cuid())
  type            String   // CUSTOMER_PAYMENT | PAYMENT_FEE | PROVIDER_PURCHASE | PROVIDER_LIABILITY | PROVIDER_INVOICE | PROVIDER_PAYMENT | REFUND | CHARGEBACK | PROMOTIONAL_CREDIT | CUSTOMER_CREDIT | TRIAL_COST | COMMISSION | ADJUSTMENT
  userId          String?
  organizationId  String?
  orderId         String?
  provider        String?  // esim provider or payment provider
  providerTxnId   String?  // provider's transaction id
  amountMinor     Int      // positive = debit (cost), negative = credit (revenue). In minor units.
  currency        String   @default("USD")
  // Financial breakdown (all in minor units, may be 0 if not applicable)
  customerPrice   Int      @default(0)  // what customer paid
  providerCost    Int      @default(0)  // wholesale cost
  paymentFee      Int      @default(0)  // payment processing fee
  grossProfit     Int      @default(0)  // customerPrice - providerCost
  contributionProfit Int   @default(0)  // grossProfit - paymentFee - refundCost - fraudLoss
  refundCost      Int      @default(0)
  fraudLoss       Int      @default(0)
  // Metadata
  source          String   @default("system") // system | admin | provider | webhook
  reason          String?
  state           String   @default("pending") // pending | settled | disputed | reversed
  idempotencyKey  String?  @unique
  createdAt       DateTime @default(now())
  settledAt       DateTime?

  user            User?          @relation(fields: [userId], references: [id], onDelete: SetNull)
  order           Order?         @relation("FinancialOrder", fields: [orderId], references: [id], onDelete: SetNull)

  @@index([userId])
  @@index([orderId])
  @@index([type])
  @@index([createdAt])
}

// ---------------------------------------------------------------------------
// Provider Credit Account — tracks provider credit facilities (e.g., Airalo $10K).
// This is a financial liability, NOT cash. Must be tracked for risk control.
// ---------------------------------------------------------------------------

model ProviderCreditAccount {
  id              String   @id @default(cuid())
  provider        String   @unique // e.g. "airalo", "mock"
  creditLimit     Int      // in minor units (e.g., 1000000 = $10,000)
  currency        String   @default("USD")
  outstandingLiability Int  @default(0) // accumulated unbilled provider purchases
  pendingCommitments Int   @default(0) // in-flight provider orders not yet confirmed
  invoicedAmount  Int      @default(0) // amount invoiced by provider
  paidAmount      Int      @default(0) // amount paid to provider
  // Thresholds (percentage of credit limit)
  thresholdInfo   Int      @default(50) // informational
  thresholdWarn   Int      @default(75) // warning
  thresholdElevated Int    @default(80) // elevated
  thresholdCritical Int    @default(90) // critical — restrict new commitments
  thresholdEmergency Int   @default(95) // emergency
  // State
  active          Boolean  @default(true)
  lastInvoiceDate DateTime?
  nextPaymentDue  DateTime?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  invoices        ProviderInvoice[]
}

model ProviderInvoice {

  dueDate         DateTime?

  // Reconciliation

  mismatchedTransactions  Int @default(0)

  @@index([status])
}

// ---------------------------------------------------------------------------
// Pricing Snapshot — preserves the pricing used for each transaction so
// historical economics can't be retroactively changed.
// ---------------------------------------------------------------------------

model PricingSnapshot {
  id              String   @id @default(cuid())
  orderId         String?  @unique
  productId       String?  // planId or virtualNumberId
  productType     String   // esim | virtual_number
  wholesaleMinor  Int      // provider cost at time of purchase
  retailMinor     Int      // customer price at time of purchase
  currency        String   @default("USD")
  markupPercent   Float    // (retail - wholesale) / wholesale * 100
  grossMarginPercent Float // (retail - wholesale) / retail * 100
  pricingRule     String?  // JSON of the rule that produced this price
  createdAt       DateTime @default(now())

  @@index([productType])
}

// ---------------------------------------------------------------------------
// Promo Codes — promotional discounts at checkout.
// ---------------------------------------------------------------------------

model PromoCode {
  id              String   @id @default(cuid())
  code            String   @unique // e.g. "TRAVEL20"
  description     String?
  type            String   // "percentage" | "fixed"
  value           Int      // percentage: 20 = 20% off; fixed: 500 = $5.00 off (minor units)
  currency        String   @default("USD")
  minOrderAmount  Int      @default(0) // minimum order amount in minor units (0 = no minimum)
  maxDiscount     Int?     // cap on discount amount (minor units, null = no cap)

  maxUsesPerUser  Int      @default(1)
  usesCount       Int      @default(0)
  validFrom       DateTime @default(now())
  validUntil      DateTime?
  active          Boolean  @default(true)
  // Profitability guardrail
  minMarginPercent Float?  // reject if discount would push margin below this
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  redemptions     PromoRedemption[]

  @@index([active])
}

  order           Order     @relation("PromoOrder", fields: [orderId], references: [id], onDelete: Cascade)

  @@index([promoCodeId])
}

// ---------------------------------------------------------------------------
// Referrals — customer referral codes + credit rewards.
// ---------------------------------------------------------------------------

model Referral {
  id              String   @id @default(cuid())
  referrerUserId  String   @unique // each user has one referral code
  referralCode    String   @unique // e.g. "ROAM-ABC123"
  // Reward config (in minor units)
  referrerReward  Int      @default(200) // $2.00 credit for referrer
  refereeReward   Int      @default(200) // $2.00 credit for referee
  currency        String   @default("USD")
  // Stats
  totalReferrals  Int      @default(0)
  completedReferrals Int   @default(0) // referee made a purchase
  totalRewardPaid Int      @default(0)
  createdAt       DateTime @default(now())

  referrer        User     @relation("ReferralUser", fields: [referrerUserId], references: [id], onDelete: Cascade)
  referrals       ReferralUse[]

  @@index([referrerUserId])
}

model ReferralUse {
  id              String   @id @default(cuid())
  referralId      String
  refereeUserId   String
  orderId         String?  // first purchase that completed the referral
  status          String   @default("pending") // pending | completed | rejected
  referrerCredited Boolean @default(false)
  refereeCredited Boolean @default(false)
  createdAt       DateTime @default(now())

  referral        Referral @relation(fields: [referralId], references: [id], onDelete: Cascade)
  referee         User     @relation("RefereeUser", fields: [refereeUserId], references: [id], onDelete: Cascade)

  @@unique([referralId, refereeUserId])
  @@index([refereeUserId])
}

// ---------------------------------------------------------------------------
// Customer Credit / Wallet — unified balance across products.
// ---------------------------------------------------------------------------

model CustomerCredit {
  id              String   @id @default(cuid())
  userId          String   @unique
  balanceMinor    Int      @default(0) // current balance in minor units
  currency        String   @default("USD")
  totalEarned     Int      @default(0) // lifetime credits earned
  totalSpent      Int      @default(0) // lifetime credits spent
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  user            User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  transactions    CreditTransaction[]

  @@index([userId])
}

model CreditTransaction {
  id              String   @id @default(cuid())
  creditId        String
  userId          String
  type            String   // referral_reward | promo_credit | admin_adjustment | purchase_credit | refund_credit
  amountMinor     Int      // positive = credit added, negative = credit spent
  balanceAfter    Int      // balance after this transaction
  reason          String?
  orderId         String?
  referrerId      String?
  createdAt       DateTime @default(now())

  credit          CustomerCredit @relation(fields: [creditId], references: [id], onDelete: Cascade)
  user            User           @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([creditId])
}
