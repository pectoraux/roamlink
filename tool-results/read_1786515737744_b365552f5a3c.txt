   116→model Plan {
   117→  id              String   @id @default(cuid())
   118→  providerId      String   // internal provider key, e.g. "mock"
   119→  providerPlanId  String   // provider's native plan id
   120→  name            String
   121→  description     String?
   122→  country         String
   123→  countryCode     String   // ISO 3166-1 alpha-2
   124→  region          String   // e.g. "Africa", "Europe", "North America"
   125→  dataAmount      Int      // in MB (1024 = 1 GB)
   126→  dataUnit        String   @default("MB")
   127→  validityDays    Int
   128→  // Money — integer minor units (cents). See lib/money.ts.
   129→  price           Int      // retail price in minor units
   130→  wholesalePrice  Int      // internal cost, NEVER exposed to customers
   131→  currency        String   @default("USD")
   132→  coverage        String?  // comma-separated countries / description
   133→  networks        String?  // JSON string: ["MTN","Vodafone"]
   134→  roaming         Boolean  @default(false)
   135→  hotspot         Boolean  @default(true)
   136→  speed           String?  // e.g. "4G/5G"
   137→  topUpSupported  Boolean  @default(true)
   138→  status          String   @default("active") // "active" | "inactive"
   139→  // Pricing rule that produced `price` (audit). JSON string.
   140→  pricingRule     String?
   141→  // Provider-native payload (never sent to client). JSON string.
   142→  metadata        String?
   143→  createdAt       DateTime @default(now())
   144→  updatedAt       DateTime @updatedAt
   145→
   146→  orders          Order[]
   147→
   148→  @@unique([providerId, providerPlanId])
   149→  @@index([countryCode])
   150→  @@index([region])
   151→  @@index([status])
   152→}

   158→model Order {
   159→  id               String   @id @default(cuid())
   160→  userId           String
   161→  planId           String?  // nullable for virtual-number orders (no Plan row)
   162→  productType      String   @default("esim") // esim | virtual_number
   163→  // State machine: PLAN_SELECTED | CHECKOUT_CREATED | PAYMENT_PENDING |
   164→  // PAYMENT_CONFIRMED | ESIM_PROVISIONING | ESIM_PROVISIONED | COMPLETED |
   165→  // PAYMENT_FAILED | PROVISIONING_FAILED | CANCELLED | REFUNDED
   166→  status           String   @default("PLAN_SELECTED")
   167→  amount           Int      // total charged (minor units)
   168→  currency         String   @default("USD")
   169→  paymentStatus    String   @default("pending") // pending | succeeded | failed | refunded
   170→  paymentProvider  String?
   171→  paymentReference String?  // provider's payment intent / charge id
   172→  providerOrderId  String?  // eSIM provider order id
   173→  // Idempotency: prevents duplicate orders / payments / provisioning on retry.
   174→  idempotencyKey   String   @unique
   175→  // Snapshot of plan at purchase time (so historical orders stay accurate).
   176→  planSnapshot     String?  // JSON string
   177→  failureReason    String?
   178→  createdAt        DateTime @default(now())
   179→  updatedAt        DateTime @updatedAt
   180→
   181→  user             User     @relation(fields: [userId], references: [id], onDelete: Cascade)
   182→  plan             Plan?    @relation(fields: [planId], references: [id])
   183→  payments         Payment[]
   184→  esim             Esim?
   185→  corporateOrder   CorporateOrder?
   186→  virtualNumber    VirtualNumber? @relation("VirtualNumberOrder")
   187→  financialTxns    FinancialTransaction[] @relation("FinancialOrder")
   188→  promoRedemption  PromoRedemption?       @relation("PromoOrder")
   189→
   190→  @@index([userId])
   191→  @@index([status])
   192→  @@index([paymentStatus])
   193→}

   594→// ---------------------------------------------------------------------------

   596→model FinancialTransaction {
   597→  id              String   @id @default(cuid())
   598→  type            String   // CUSTOMER_PAYMENT | PAYMENT_FEE | PROVIDER_PURCHASE | PROVIDER_LIABILITY | PROVIDER_INVOICE | PROVIDER_PAYMENT | REFUND | CHARGEBACK | PROMOTIONAL_CREDIT | CUSTOMER_CREDIT | TRIAL_COST | COMMISSION | ADJUSTMENT
   599→  userId          String?

   603→  providerTxnId   String?  // provider's transaction id
   604→  amountMinor     Int      // positive = debit (cost), negative = credit (revenue). In minor units.
   605→  currency        String   @default("USD")

   621→
   622→  user            User?          @relation(fields: [userId], references: [id], onDelete: SetNull)
   623→  order           Order?         @relation("FinancialOrder", fields: [orderId], references: [id], onDelete: SetNull)
