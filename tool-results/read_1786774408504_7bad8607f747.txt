     1→/**
     2→ * Phase 2C.1 — Connectivity Entitlement Kernel
     3→ *
     4→ * This module implements the connectivity entitlement layer that sits between
     5→ * the SaaS billing kernel (FROZEN) and provider-specific adapters (MikroTik,
     6→ * WiFi ISP, eSIM).
     7→ *
     8→ * Three layers of truth:
     9→ *   Commercial Truth: TenantSubscription (billing kernel — FROZEN)
    10→ *   Entitlement Truth: ConnectivityEntitlement (this layer)
    11→ *   Infrastructure Truth: ProviderResourceBinding (this layer)
    12→ *
    13→ * The entitlement kernel does NOT know about MikroTik, RADIUS, eSIM suppliers,
    14→ * or ISP APIs. It only expresses what connectivity capability a customer owns.
    15→ *
    16→ * Carryover principle from SaaS hardening:
    17→ *   "Every external side effect needs a durable state machine and reconciliation path."
    18→ *
    19→ * Entitlement Lifecycle: PENDING → ACTIVE → (SUSPENDED | EXPIRED | REVOKED)
    20→ * Binding Lifecycle:     UNBOUND → PROVISIONING → BOUND → (DEGRADED | FAILED | RELEASED)
    21→ *
    22→ * The separation matters:
    23→ *   An ACTIVE entitlement can coexist with a FAILED binding (router offline).
    24→ *   Billing must not interpret infrastructure unhealthiness as a payment problem.
    25→ */
    26→

    51→export const ENTITLEMENT_STATES = {
    52→  PENDING: "PENDING",
    53→  ACTIVE: "ACTIVE",
    54→  SUSPENDED: "SUSPENDED",
    55→  EXPIRED: "EXPIRED",
    56→  REVOKED: "REVOKED",
    57→} as const;
    58→
    59→export type EntitlementState = typeof ENTITLEMENT_STATES[keyof typeof ENTITLEMENT_STATES];
    60→
    61→/**
    62→ * Legal entitlement state transitions.
    63→ * Any transition not listed here is rejected.
    64→ */
    65→const ENTITLEMENT_TRANSITIONS: Record<EntitlementState, EntitlementState[]> = {
    66→  PENDING: [ENTITLEMENT_STATES.ACTIVE, ENTITLEMENT_STATES.REVOKED],
    67→  ACTIVE: [ENTITLEMENT_STATES.SUSPENDED, ENTITLEMENT_STATES.EXPIRED, ENTITLEMENT_STATES.REVOKED],
    68→  SUSPENDED: [ENTITLEMENT_STATES.ACTIVE, ENTITLEMENT_STATES.REVOKED],
    69→  EXPIRED: [], // terminal
    70→  REVOKED: [], // terminal
    71→};
    72→
    73→// ---------------------------------------------------------------------------
    74→// Binding State Machine
    75→// ---------------------------------------------------------------------------
    76→
    77→export const BINDING_STATES = {
    78→  UNBOUND: "UNBOUND",
    79→  PROVISIONING: "PROVISIONING",
    80→  BOUND: "BOUND",
    81→  DEGRADED: "DEGRADED",
    82→  FAILED: "FAILED",
    83→  RELEASED: "RELEASED",
    84→} as const;
    85→
    86→export type BindingState = typeof BINDING_STATES[keyof typeof BINDING_STATES];
    87→
    88→/**
    89→ * Legal binding state transitions.
    90→ * Any transition not listed here is rejected.
    91→ */
    92→const BINDING_TRANSITIONS: Record<BindingState, BindingState[]> = {
    93→  UNBOUND: [BINDING_STATES.PROVISIONING, BINDING_STATES.FAILED],
    94→  PROVISIONING: [BINDING_STATES.BOUND, BINDING_STATES.FAILED],
    95→  BOUND: [BINDING_STATES.DEGRADED, BINDING_STATES.FAILED, BINDING_STATES.RELEASED],
    96→  DEGRADED: [BINDING_STATES.BOUND, BINDING_STATES.FAILED, BINDING_STATES.RELEASED],
    97→  FAILED: [BINDING_STATES.PROVISIONING, BINDING_STATES.RELEASED], // can retry provisioning
    98→  RELEASED: [], // terminal
    99→};

   181→/**
   182→ * Create a connectivity entitlement for a tenant.
   183→ *
   184→ * The entitlement represents the customer's RIGHT to consume connectivity.
   185→ * It starts in PENDING state and must be explicitly activated.
   186→ *
   187→ * The entitlement is linked to a subscription (commercial truth) but does
   188→ * not depend on the subscription's payment state — that's the billing kernel's
   189→ * responsibility. This layer only tracks the capability grant.
   190→ */
   191→export async function createEntitlement(input: {
   192→  tenantId: string;
   193→  subscriptionId: string;
   194→  capabilityType: CapabilityType;
   195→  capabilitySet: Record<string, unknown>;
   196→  policy?: Record<string, unknown>;
   197→  validFrom: Date;
   198→  validUntil?: Date;
   199→  userId?: string;
   200→}): Promise<{ id: string; status: string }> {
   201→  // Look up the capability by type
   202→  const capability = await db.connectivityCapability.findUnique({
   203→    where: { type: input.capabilityType },
   204→  });
   205→  if (!capability) {
   206→    throw new AppError("not_found", "Capability not found", 404, `Capability type "${input.capabilityType}" does not exist. Run seedConnectivityCapabilities() first.`);
   207→  }
   208→
   209→  const entitlement = await db.connectivityEntitlement.create({
   210→    data: {
   211→      tenantId: input.tenantId,
   212→      subscriptionId: input.subscriptionId,
   213→      capabilityId: capability.id,
   214→      status: ENTITLEMENT_STATES.PENDING,
   215→      capabilitySet: JSON.stringify(input.capabilitySet),
   216→      policy: input.policy ? JSON.stringify(input.policy) : null,
   217→      validFrom: input.validFrom,
   218→      validUntil: input.validUntil ?? null,
   219→    },
   220→  });
   221→
   222→  await audit({
   223→    tenantId: input.tenantId,
   224→    userId: input.userId,
   225→    action: "connectivity.entitlement_created",
   226→    entity: "connectivity_entitlement",
   227→    entityId: entitlement.id,
   228→    detail: {
   229→      capabilityType: input.capabilityType,
   230→      subscriptionId: input.subscriptionId,
   231→      capabilitySet: input.capabilitySet,
   232→    },
   233→  });
   234→
   235→  logger.info("connectivity.entitlement_created", {
   236→    entitlementId: entitlement.id, tenantId: input.tenantId,
   237→    capabilityType: input.capabilityType, subscriptionId: input.subscriptionId,
   238→  });
   239→
   240→  return { id: entitlement.id, status: entitlement.status };
   241→}

   243→/**
   244→ * Transition an entitlement to a new state.
   245→ * Uses a guarded updateMany — only the legal predecessor state can transition.
   246→ * Returns { transitioned: false } if the guard doesn't match (idempotent).
   247→ */
   248→export async function transitionEntitlement(input: {
   249→  entitlementId: string;
   250→  toState: EntitlementState;
   251→  reason?: string;
   252→  userId?: string;
   253→}): Promise<{ transitioned: boolean; status: string }> {
   254→  const entitlement = await db.connectivityEntitlement.findUnique({
   255→    where: { id: input.entitlementId },
   256→    select: { id: true, status: true, tenantId: true },
   257→  });
   258→  if (!entitlement) {
   259→    throw new AppError("not_found", "Entitlement not found", 404, "Connectivity entitlement not found.");
   260→  }
   261→
   262→  const currentState = entitlement.status as EntitlementState;
   263→  const legalTransitions = ENTITLEMENT_TRANSITIONS[currentState] || [];
   264→
   265→  // Check if this is a valid transition
   266→  if (!legalTransitions.includes(input.toState)) {
   267→    if (currentState === input.toState) {
   268→      // Idempotent — already in the target state
   269→      return { transitioned: false, status: currentState };
   270→    }
   271→    throw new AppError(
   272→      "conflict",
   273→      "Invalid entitlement transition",
   274→      409,
   275→      `Cannot transition entitlement from ${currentState} to ${input.toState}. Legal transitions from ${currentState}: ${legalTransitions.join(", ") || "none (terminal state)"}.`,
   276→    );
   277→  }
   278→
   279→  // Guarded transition — only the expected predecessor state can transition
   280→  const result = await db.connectivityEntitlement.updateMany({
   281→    where: { id: input.entitlementId, status: currentState },
   282→    data: { status: input.toState, failureReason: input.reason ?? null },
   283→  });
   284→
   285→  if (result.count === 0) {
   286→    // Concurrent transition — another worker moved it first
   287→    logger.info("connectivity.entitlement_transition_concurrent", {
   288→      entitlementId: input.entitlementId, fromState: currentState, toState: input.toState,
   289→    });
   290→    return { transitioned: false, status: currentState };
   291→  }
   292→
   293→  await audit({
   294→    tenantId: entitlement.tenantId,
   295→    userId: input.userId,
   296→    action: "connectivity.entitlement_transitioned",
   297→    entity: "connectivity_entitlement",
   298→    entityId: input.entitlementId,
   299→    detail: { from: currentState, to: input.toState, reason: input.reason },
   300→  });
   301→
   302→  logger.info("connectivity.entitlement_transitioned", {
   303→    entitlementId: input.entitlementId, from: currentState, to: input.toState,
   304→  });
   305→
   306→  return { transitioned: true, status: input.toState };
   307→}

   344→/**
   345→ * Create a provider resource binding for an entitlement.
   346→ *
   347→ * The binding represents the actual external resource that delivers the capability.
   348→ * It starts in UNBOUND state and must transition through PROVISIONING → BOUND.
   349→ *
   350→ * The providerType is provider-neutral ("mock", "mikrotik", "wifi-isp", "esim").
   351→ * The providerMetadata is opaque JSON — the entitlement kernel does NOT interpret it.
   352→ */
   353→export async function createResourceBinding(input: {
   354→  entitlementId: string;
   355→  providerType: string;
   356→  /** Phase 2C.3: Resource type within the provider (e.g., "hotspot_user", "radius_subscriber") */
   357→  resourceType?: string;
   358→  /** Phase 2C.3.1: Provider instance — the specific infrastructure endpoint */
   359→  providerInstanceId?: string;
   360→  providerMetadata?: Record<string, unknown>;
   361→  userId?: string;
   362→}): Promise<{ id: string; status: string }> {
   363→  // Phase 2C.3.1: If a providerInstanceId is supplied, verify it belongs to
   364→  // the same tenant as the entitlement and matches the providerType.
   365→  if (input.providerInstanceId) {
   366→    const entitlement = await db.connectivityEntitlement.findUnique({
   367→      where: { id: input.entitlementId },
   368→      select: { tenantId: true },
   369→    });
   370→    if (!entitlement) {
   371→      throw new AppError("not_found", "Entitlement not found", 404, "Cannot create binding — entitlement not found.");
   372→    }
   373→    const instance = await db.connectivityProviderInstance.findUnique({
   374→      where: { id: input.providerInstanceId },
   375→      select: { tenantId: true, providerType: true, status: true },
   376→    });
   377→    if (!instance) {
   378→      throw new AppError("not_found", "Provider instance not found", 404, "Provider instance does not exist.");
   379→    }
   380→    // Tenant isolation: the instance must belong to the same tenant
   381→    if (instance.tenantId !== entitlement.tenantId) {
   382→      throw new AppError(
   383→        "authorization",
   384→        "Cross-tenant provider instance access denied",
   385→        403,
   386→        "The provider instance belongs to a different tenant.",
   387→      );
   388→    }
   389→    // Type match: the instance's providerType must match the binding's providerType
   390→    if (instance.providerType !== input.providerType) {
   391→      throw new AppError(
   392→        "validation",
   393→        "Provider type mismatch",
   394→        400,
   395→        `Instance providerType is "${instance.providerType}" but binding providerType is "${input.providerType}".`,
   396→      );
   397→    }
   398→    // Instance must be active
   399→    if (instance.status !== "active") {
   400→      throw new AppError(
   401→        "conflict",
   402→        "Provider instance not active",
   403→        409,
   404→        `Provider instance status is "${instance.status}". Only "active" instances can be used.`,
   405→      );
   406→    }
   407→  }
   408→
   409→  const binding = await db.providerResourceBinding.create({
   410→    data: {
   411→      entitlementId: input.entitlementId,
   412→      providerType: input.providerType,
   413→      resourceType: input.resourceType ?? null,
   414→      providerInstanceId: input.providerInstanceId ?? null,
   415→      providerMetadata: input.providerMetadata ? JSON.stringify(input.providerMetadata) : null,
   416→      status: BINDING_STATES.UNBOUND,
   417→    },
   418→  });
   419→
   420→  await audit({
   421→    tenantId: undefined, // bindings don't have direct tenantId — it's on the entitlement
   422→    userId: input.userId,
   423→    action: "connectivity.binding_created",
   424→    entity: "provider_resource_binding",
   425→    entityId: binding.id,
   426→    detail: {
   427→      entitlementId: input.entitlementId,
   428→      providerType: input.providerType,
   429→    },
   430→  });
   431→
   432→  logger.info("connectivity.binding_created", {
   433→    bindingId: binding.id, entitlementId: input.entitlementId,
   434→    providerType: input.providerType,
   435→  });
   436→
   437→  return { id: binding.id, status: binding.status };
   438→}

   440→/**
   441→ * Transition a resource binding to a new state.
   442→ * Uses a guarded updateMany — only the legal predecessor state can transition.
   443→ */
   444→export async function transitionBinding(input: {
   445→  bindingId: string;
   446→  toState: BindingState;
   447→  providerResourceId?: string;
   448→  providerMetadata?: Record<string, unknown>;
   449→  provisioningState?: string;
   450→  reason?: string;
   451→  userId?: string;
   452→}): Promise<{ transitioned: boolean; status: string }> {
   453→  const binding = await db.providerResourceBinding.findUnique({
   454→    where: { id: input.bindingId },
   455→    select: { id: true, status: true, entitlementId: true },
   456→  });
   457→  if (!binding) {
   458→    throw new AppError("not_found", "Binding not found", 404, "Provider resource binding not found.");
   459→  }
   460→
   461→  const currentState = binding.status as BindingState;
   462→  const legalTransitions = BINDING_TRANSITIONS[currentState] || [];
   463→
   464→  if (!legalTransitions.includes(input.toState)) {
   465→    if (currentState === input.toState) {
   466→      return { transitioned: false, status: currentState };
   467→    }
   468→    throw new AppError(
   469→      "conflict",
   470→      "Invalid binding transition",
   471→      409,
   472→      `Cannot transition binding from ${currentState} to ${input.toState}. Legal transitions from ${currentState}: ${legalTransitions.join(", ") || "none (terminal state)"}.`,
   473→    );
   474→  }
   475→
   476→  const updateData: Record<string, unknown> = {
   477→    status: input.toState,
   478→    failureReason: input.reason ?? null,
   479→  };
   480→  if (input.providerResourceId !== undefined) {
   481→    updateData.providerResourceId = input.providerResourceId;
   482→  }
   483→  if (input.providerMetadata !== undefined) {
   484→    updateData.providerMetadata = JSON.stringify(input.providerMetadata);
   485→  }
   486→  if (input.provisioningState !== undefined) {
   487→    updateData.provisioningState = input.provisioningState;
   488→  }
   489→
   490→  const result = await db.providerResourceBinding.updateMany({
   491→    where: { id: input.bindingId, status: currentState },
   492→    data: updateData,
   493→  });
   494→
   495→  if (result.count === 0) {
   496→    logger.info("connectivity.binding_transition_concurrent", {
   497→      bindingId: input.bindingId, fromState: currentState, toState: input.toState,
   498→    });
   499→    return { transitioned: false, status: currentState };
   500→  }
   501→
   502→  logger.info("connectivity.binding_transitioned", {
   503→    bindingId: input.bindingId, from: currentState, to: input.toState,
   504→  });
   505→
   506→  return { transitioned: true, status: input.toState };
   507→}

   519→// ---------------------------------------------------------------------------
   520→// Reconciliation Framework
   521→// ---------------------------------------------------------------------------
   522→
   523→/**
   524→ * Reconcile connectivity entitlements and their resource bindings.
   525→ *
   526→ * This is the Phase 2C equivalent of the SaaS reconciliation worker.
   527→ * It checks for drift between desired and observed state:
   528→ *
   529→ *   - Entitlements that should be ACTIVE but have no BOUND binding
   530→ *   - Bindings that are FAILED but the entitlement is still ACTIVE
   531→ *   - Entitlements past their validUntil that haven't been EXPIRED
   532→ *   - Bindings in PROVISIONING for too long (stuck)
   533→ *
   534→ * This function is idempotent — running it multiple times produces the same
   535→ * state. It does NOT create or destroy resources — only transitions state
   536→ * and marks reconciliation flags.
   537→ */
   538→export async function reconcileConnectivityEntitlements(): Promise<{
   539→  entitlementsExpired: number;
   540→  bindingsStuck: number;
   541→  driftDetected: number;
   542→  reconciled: number;
   543→}> {
   544→  const result = { entitlementsExpired: 0, bindingsStuck: 0, driftDetected: 0, reconciled: 0 };
   545→  const now = new Date();
   546→
   547→  // 1. Expire entitlements past their validUntil
   548→  const expired = await db.connectivityEntitlement.updateMany({
   549→    where: {
   550→      status: { in: [ENTITLEMENT_STATES.PENDING, ENTITLEMENT_STATES.ACTIVE, ENTITLEMENT_STATES.SUSPENDED] },
   551→      validUntil: { lt: now },
   552→    },
   553→    data: {
   554→      status: ENTITLEMENT_STATES.EXPIRED,
   555→      reconciliationState: "RECONCILED",
   556→    },
   557→  });
   558→  result.entitlementsExpired = expired.count;
   559→
   560→  // 2. Detect stuck PROVISIONING bindings (older than 5 minutes)
   561→  const stuckCutoff = new Date(now.getTime() - 5 * 60 * 1000);
   562→  const stuckBindings = await db.providerResourceBinding.findMany({
   563→    where: {
   564→      status: BINDING_STATES.PROVISIONING,
   565→      updatedAt: { lt: stuckCutoff },
   566→    },
   567→    select: { id: true, entitlementId: true },
   568→  });
   569→
   570→  for (const binding of stuckBindings) {
   571→    // Mark as RECONCILIATION_REQUIRED — the provider adapter should retry
   572→    const stuckResult = await db.providerResourceBinding.updateMany({
   573→      where: { id: binding.id, status: BINDING_STATES.PROVISIONING },
   574→      data: {
   575→        reconciliationState: "RECONCILIATION_REQUIRED",
   576→        failureReason: "Provisioning stuck for > 5 minutes — provider adapter should retry",
   577→      },
   578→    });
   579→    if (stuckResult.count > 0) {
   580→      result.bindingsStuck++;
   581→      logger.warn("connectivity.binding_stuck", {
   582→        bindingId: binding.id, entitlementId: binding.entitlementId,
   583→      });
   584→    }
   585→  }
   586→
   587→  // 3. Detect drift: ACTIVE entitlements with no BOUND binding
   588→  const activeEntitlements = await db.connectivityEntitlement.findMany({
   589→    where: { status: ENTITLEMENT_STATES.ACTIVE },
   590→    select: { id: true, tenantId: true },
   591→  });
   592→
   593→  for (const ent of activeEntitlements) {
   594→    const bindings = await db.providerResourceBinding.findMany({
   595→      where: { entitlementId: ent.id },
   596→      select: { status: true },
   597→    });
   598→
   599→    const hasBound = bindings.some((b) => b.status === BINDING_STATES.BOUND);
   600→    if (!hasBound && bindings.length > 0) {
   601→      // Entitlement is ACTIVE but no binding is BOUND — drift
   602→      const driftResult = await db.connectivityEntitlement.updateMany({
   603→        where: { id: ent.id, status: ENTITLEMENT_STATES.ACTIVE },
   604→        data: { reconciliationState: "RECONCILIATION_REQUIRED" },
   605→      });
   606→      if (driftResult.count > 0) {
   607→        result.driftDetected++;
   608→        logger.warn("connectivity.entitlement_drift", {
   609→          entitlementId: ent.id, tenantId: ent.tenantId,
   610→          message: "ACTIVE entitlement has no BOUND binding — reconciliation required",
   611→        });
   612→      }
   613→    } else if (hasBound) {
   614→      // Entitlement is ACTIVE and has a BOUND binding — mark reconciled
   615→      const reconResult = await db.connectivityEntitlement.updateMany({
   616→        where: { id: ent.id, status: ENTITLEMENT_STATES.ACTIVE, reconciliationState: "RECONCILIATION_REQUIRED" },
   617→        data: { reconciliationState: "RECONCILED" },
   618→      });
   619→      if (reconResult.count > 0) {
   620→        result.reconciled++;
   621→      }
   622→    }
   623→  }
   624→
   625→  logger.info("connectivity.reconciliation_completed", result);
   626→  return result;
   627→}
   628→
   629→// ---------------------------------------------------------------------------
   630→// Phase 2C.2.2: Reconciliation Boundary — Adapter Reports, Kernel Decides
   631→// (Atomic, Stale-Observation-Safe, Kernel-Validated Transitions)
   632→// ---------------------------------------------------------------------------
   633→
   634→/**
   635→ * Phase 2C.2.2: Kernel-owned legal transition matrix for reconciliation-driven
   636→ * binding state transitions.
   637→ *
   638→ * The adapter may RETURN a recommendedBindingState, but the kernel VALIDATES
   639→ * it against this matrix. If the transition is not legal, the kernel refuses
   640→ * it and marks the binding for manual intervention.
   641→ *
   642→ * This prevents an adapter from putting a binding into an illegal state.
   643→ */
   644→const RECONCILIATION_LEGAL_TRANSITIONS: Record<string, BindingState[]> = {
   645→  // From BOUND: can degrade, fail, or release (not unbind or re-provision)
   646→  BOUND: [BINDING_STATES.DEGRADED, BINDING_STATES.FAILED, BINDING_STATES.RELEASED],
   647→  // From DEGRADED: can recover to BOUND, fail, or release
   648→  DEGRADED: [BINDING_STATES.BOUND, BINDING_STATES.FAILED, BINDING_STATES.RELEASED],
   649→  // From PROVISIONING: can succeed to BOUND or fail
   650→  PROVISIONING: [BINDING_STATES.BOUND, BINDING_STATES.FAILED],
   651→  // From UNBOUND: can start provisioning or fail
   652→  UNBOUND: [BINDING_STATES.PROVISIONING, BINDING_STATES.FAILED],
   653→  // From FAILED: can retry provisioning or release (NOT directly to BOUND)
   654→  FAILED: [BINDING_STATES.PROVISIONING, BINDING_STATES.RELEASED],
   655→  // From RELEASED: terminal — no transitions
   656→  RELEASED: [],
   657→};
   658→
   659→/**
   660→ * Phase 2C.2.2: Check if a reconciliation-driven transition is legal.
   661→ */
   662→function isLegalReconciliationTransition(fromState: string, toState: BindingState): boolean {
   663→  const legal = RECONCILIATION_LEGAL_TRANSITIONS[fromState] || [];
   664→  return legal.includes(toState);
   665→}
   666→
   667→/**
   668→ * Phase 2C.2.2: Map a ReconciliationResult to a target binding state + reconciliation metadata.
   669→ *
   670→ * The kernel owns this mapping — the adapter's recommendation is a SIGNAL,
   671→ * not authority. The kernel validates the recommended transition against
   672→ * the legal transition matrix.
   673→ *
   674→ * Returns:
   675→ *   - targetBindingState: the state to transition to (if any)
   676→ *   - reconciliationState: the durable reconciliation metadata
   677→ *   - failureReason: optional failure reason
   678→ */
   679→function mapReconciliationResult(
   680→  currentBindingStatus: string,
   681→  result: { status: string; recommendedBindingState?: string; details?: string },
   682→): {
   683→  targetBindingState: BindingState | null;
   684→  reconciliationState: string;
   685→  failureReason: string | null;
   686→} {
   687→  switch (result.status) {
   688→    case "in_sync":
   689→      return {
   690→        targetBindingState: null, // no transition
   691→        reconciliationState: "RECONCILED",
   692→        failureReason: null,
   693→      };
   694→
   695→    case "drift_detected": {
   696→      // The adapter recommends a state — the kernel validates it
   697→      const recommended = result.recommendedBindingState as BindingState | undefined;
   698→      if (recommended && isLegalReconciliationTransition(currentBindingStatus, recommended)) {
   699→        return {
   700→          targetBindingState: recommended,
   701→          reconciliationState: "RECONCILED",
   702→          failureReason: result.details ?? "Provider drift detected",
   703→        };
   704→      }
   705→      // Recommendation is missing or illegal — fail closed
   706→      return {
   707→        targetBindingState: null,
   708→        reconciliationState: "MANUAL_INTERVENTION_REQUIRED",
   709→        failureReason: `Adapter recommended "${recommended}" from "${currentBindingStatus}" — illegal or missing. Manual intervention required.`,
   710→      };
   711→    }
   712→
   713→    case "resource_missing":
   714→      // Resource is gone — transition to FAILED if legal
   715→      if (isLegalReconciliationTransition(currentBindingStatus, BINDING_STATES.FAILED)) {
   716→        return {
   717→          targetBindingState: BINDING_STATES.FAILED,
   718→          reconciliationState: "RECONCILED",
   719→          failureReason: "Provider resource missing",
   720→        };
   721→      }
   722→      // Can't transition to FAILED from current state — manual intervention
   723→      return {
   724→        targetBindingState: null,
   725→        reconciliationState: "MANUAL_INTERVENTION_REQUIRED",
   726→        failureReason: `resource_missing from "${currentBindingStatus}" — cannot transition to FAILED`,
   727→      };
   728→
   729→    case "failed_retryable":
   730→      // Keep current state — retry later
   731→      return {
   732→        targetBindingState: null,
   733→        reconciliationState: "RECONCILIATION_REQUIRED",
   734→        failureReason: result.details ?? "Retryable reconciliation failure",
   735→      };
   736→
   737→    case "failed_permanent":
   738→      // Phase 2C.2.2: Use MANUAL_INTERVENTION_REQUIRED (NOT RECONCILIATION_REQUIRED)
   739→      // to prevent the automatic retry loop from picking this up.
   740→      // Transition to FAILED if legal, but mark as manual intervention.
   741→      if (isLegalReconciliationTransition(currentBindingStatus, BINDING_STATES.FAILED)) {
   742→        return {
   743→          targetBindingState: BINDING_STATES.FAILED,
   744→          reconciliationState: "MANUAL_INTERVENTION_REQUIRED",
   745→          failureReason: result.details ?? "Permanent reconciliation failure",
   746→        };
   747→      }
   748→      return {
   749→        targetBindingState: null,
   750→        reconciliationState: "MANUAL_INTERVENTION_REQUIRED",
   751→        failureReason: `Permanent failure from "${currentBindingStatus}" — cannot transition to FAILED`,
   752→      };
   753→
   754→    default:
   755→      return {
   756→        targetBindingState: null,
   757→        reconciliationState: "MANUAL_INTERVENTION_REQUIRED",
   758→        failureReason: `Unknown reconciliation status: ${result.status}`,
   759→      };
   760→  }
   761→}
   762→
   763→/**
   764→ * Phase 2C.2.2: Reconcile a single binding with its provider.
   765→ *
   766→ * ATOMIC RECONCILIATION:
   767→ *   The entire operation (observe → validate → transition → commit) is
   768→ *   performed inside ONE PostgreSQL transaction with FOR UPDATE on the
   769→ *   ProviderResourceBinding row. This prevents:
   770→ *     - Partial reconciliation (transition succeeds but metadata write fails)
   771→ *     - Concurrent reconciliation overwriting each other
   772→ *     - Stale observations overwriting newer state
   773→ *
   774→ * STALE OBSERVATION PREVENTION:
   775→ *   The binding's status is captured BEFORE the adapter call. After the
   776→ *   adapter returns, the transaction locks the row and verifies the status
   777→ *   hasn't changed. If it has, the reconciliation is a no-op (the observation
   778→ *   is stale).
   779→ *
   780→ * RECONCILIATION BOUNDARY:
   781→ *   The adapter returns observations only. The kernel owns all state
   782→ *   transitions, validated against a legal transition matrix.
   783→ *
   784→ * Idempotent — safe to call multiple times.
   785→ */
   786→export async function reconcileBindingWithProvider(bindingId: string): Promise<{
   787→  status: "in_sync" | "transitioned" | "no_action" | "stale_observation" | "error";
   788→  observedState?: string;
   789→  transition?: { from: string; to: string };
   790→  error?: string;
   791→}> {
   792→  try {
   793→    // Phase 2C.3.2: Use the canonical runtime resolver.
   794→    // This validates tenant isolation, type match, and instance status
   795→    // at RUNTIME — not just at creation time.
   796→    const { adapter, binding: bindingInput, entitlement: entitlementInput } = await resolveBindingRuntime(bindingId);
   797→
   798→    // Capture the observed binding status BEFORE the adapter call.
   799→    const observedBindingStatus = bindingInput.status;
   800→
   801→    // Call the adapter's reconcile() — this is an OBSERVATION, not a mutation.
   802→    // The adapter receives the full binding input INCLUDING providerInstanceId
   803→    // and providerInstanceConfiguration.
   804→    const adapterResult = await adapter.reconcile({
   805→      entitlement: entitlementInput,
   806→      binding: bindingInput,
   807→    });
   808→
   809→    // Step 4: Map the observation to a kernel decision (target state + metadata).
   810→    // The kernel validates the transition against the legal transition matrix.
   811→    const decision = mapReconciliationResult(observedBindingStatus, adapterResult);
   812→
   813→    // Step 5: ATOMIC COMMIT — perform the transition + metadata update in ONE transaction.
   814→    // Lock the binding row with FOR UPDATE. Verify the status hasn't changed
   815→    // since we observed it (stale-observation prevention).
   816→    const txResult = await db.$transaction(async (tx) => {
   817→      // Lock the binding row
   818→      const lockedBinding: Array<{ id: string; status: string }> = await tx.$queryRaw`
   819→        SELECT id, status FROM "ProviderResourceBinding" WHERE id = ${bindingId} FOR UPDATE
   820→      `;
   821→      if (lockedBinding.length === 0) {
   822→        return { committed: false as const, reason: "Binding not found in transaction" };
   823→      }
   824→
   825→      const currentStatus = lockedBinding[0].status;
   826→
   827→      // STALE OBSERVATION CHECK: if the binding's status changed since we observed it,
   828→      // the adapter's observation is stale. Do NOT apply the transition.
   829→      if (currentStatus !== observedBindingStatus) {
   830→        return {
   831→          committed: false as const,
   832→          reason: `Stale observation: binding was "${observedBindingStatus}" when observed, now "${currentStatus}". Skipping — another worker changed it.`,
   833→        };
   834→      }
   835→
   836→      // Apply the transition (if any) + metadata update atomically
   837→      if (decision.targetBindingState) {
   838→        // Guarded transition: only if the status is still what we observed
   839→        const transitionResult = await tx.providerResourceBinding.updateMany({
   840→          where: { id: bindingId, status: observedBindingStatus },
   841→          data: {
   842→            status: decision.targetBindingState,
   843→            failureReason: decision.failureReason,
   844→            reconciliationState: decision.reconciliationState,
   845→            lastReconciledAt: new Date(),
   846→          },
   847→        });
   848→        if (transitionResult.count === 0) {
   849→          return { committed: false as const, reason: "Transition guard failed — concurrent modification" };
   850→        }
   851→        return {
   852→          committed: true as const,
   853→          transitioned: true,
   854→          from: observedBindingStatus,
   855→          to: decision.targetBindingState,
   856→        };
   857→      } else {
   858→        // No state transition — just update reconciliation metadata
   859→        await tx.providerResourceBinding.update({
   860→          where: { id: bindingId },
   861→          data: {
   862→            reconciliationState: decision.reconciliationState,
   863→            failureReason: decision.failureReason,
   864→            lastReconciledAt: new Date(),
   865→          },
   866→        });
   867→        return { committed: true as const, transitioned: false };
   868→      }
   869→    }, { timeout: 30000, maxWait: 15000 });
   870→
   871→    if (!txResult.committed) {
   872→      if (txResult.reason.includes("Stale observation")) {
   873→        return { status: "stale_observation", error: txResult.reason };
   874→      }
   875→      return { status: "no_action", error: txResult.reason };
   876→    }
   877→
   878→    if (txResult.transitioned) {
   879→      return {
   880→        status: "transitioned",
   881→        observedState: adapterResult.observedState,
   882→        transition: { from: txResult.from, to: txResult.to },
   883→      };
   884→    }
   885→
   886→    return { status: "in_sync", observedState: adapterResult.observedState };
   887→  } catch (err) {
   888→    const errorMsg = err instanceof Error ? err.message : String(err);
   889→    logger.error("connectivity.reconcile_binding_failed", { bindingId, error: errorMsg });
   890→    return { status: "error", error: errorMsg };
   891→  }
   892→}

   978→export async function resolveBindingWithInstance(bindingId: string): Promise<{
   979→  adapter: import("./adapter").ConnectivityProviderAdapter;
   980→  binding: { id: string; providerType: string; providerInstanceId: string | null; status: string; providerResourceId: string | null };
   981→  providerInstance: { id: string; tenantId: string; providerType: string; name: string; status: string } | null;
   982→}> {

   984→  const { resolveBindingAdapter } = await import("./registry");
   985→

   987→

   999→    }
  1000→  }
  1001→
  1002→  return { adapter, binding, providerInstance };
  1003→}

  1005→// ---------------------------------------------------------------------------
  1006→// Phase 2C.3.2: Canonical Runtime Resolution
  1007→// ---------------------------------------------------------------------------
  1008→
  1009→/**
  1010→ * Phase 2C.3.2: The canonical runtime resolver.
  1011→ *
  1012→ * This is the ONLY correct way to resolve a binding's full runtime context:
  1013→ *   1. Load the ProviderResourceBinding
  1014→ *   2. Load the associated ConnectivityEntitlement (for tenantId)
  1015→ *   3. Load the ConnectivityProviderInstance (if providerInstanceId is set)
  1016→ *   4. Verify tenant isolation: instance.tenantId === entitlement.tenantId
  1017→ *   5. Verify type match: instance.providerType === binding.providerType
  1018→ *   6. Verify instance status: must be "active"
  1019→ *   7. Resolve the adapter through the registry
  1020→ *
  1021→ * This resolver does NOT rely on creation-time authorization. It validates
  1022→ * the CURRENT relationship every time. If the instance's ownership, type,
  1023→ * or status has changed since binding creation, the resolver fails closed.
  1024→ *
  1025→ * Returns the full runtime context: adapter, binding, entitlement, providerInstance.
  1026→ */
  1027→export async function resolveBindingRuntime(bindingId: string): Promise<{
  1028→  adapter: import("./adapter").ConnectivityProviderAdapter;
  1029→  binding: import("./adapter").ProviderResourceBindingInput;
  1030→  entitlement: import("./adapter").ConnectivityEntitlementInput;
  1031→  providerInstance: {
  1032→    id: string;
  1033→    tenantId: string;
  1034→    providerType: string;
  1035→    name: string;
  1036→    status: string;
  1037→    configuration: Record<string, unknown> | null;
  1038→    configurationKey: string | null;
  1039→  } | null;
  1040→}> {
  1041→  const { resolveBindingAdapter } = await import("./registry");
  1042→
  1043→  // Step 1: Load the binding (includes providerInstanceId from 2C.3.1)
  1044→  const { adapter, binding: bindingSummary } = await resolveBindingAdapter(bindingId);
  1045→
  1046→  // Step 2: Load the full binding with all fields needed for the adapter input
  1047→  const fullBinding = await db.providerResourceBinding.findUnique({
  1048→    where: { id: bindingId },
  1049→    select: {
  1050→      id: true, entitlementId: true, providerType: true,
  1051→      providerResourceId: true, providerMetadata: true,
  1052→      status: true, provisioningState: true,
  1053→      providerInstanceId: true,
  1054→    },
  1055→  });
  1056→  if (!fullBinding) {
  1057→    throw new Error(`ProviderResourceBinding not found: ${bindingId}`);
  1058→  }
  1059→
  1060→  // Step 3: Load the entitlement (for tenantId + capability info)
  1061→  const entitlement = await db.connectivityEntitlement.findUnique({
  1062→    where: { id: fullBinding.entitlementId },
  1063→    include: { capability: true },
  1064→  });
  1065→  if (!entitlement) {
  1066→    throw new Error(`Entitlement not found for binding ${bindingId}`);
  1067→  }
  1068→
  1069→  // Step 4: Load the provider instance (if set)
  1070→  let providerInstance: {
  1071→    id: string; tenantId: string; providerType: string;
  1072→    name: string; status: string;
  1073→    configuration: Record<string, unknown> | null;
  1074→    configurationKey: string | null;
  1075→  } | null = null;
  1076→
  1077→  if (fullBinding.providerInstanceId) {
  1078→    const instance = await db.connectivityProviderInstance.findUnique({
  1079→      where: { id: fullBinding.providerInstanceId },
  1080→      select: { id: true, tenantId: true, providerType: true, name: true, status: true, configuration: true, configurationKey: true },
  1081→    });
  1082→
  1083→    if (!instance) {
  1084→      throw new Error(
  1085→        `Provider instance not found: ${fullBinding.providerInstanceId} (binding ${bindingId}). ` +
  1086→        `The instance may have been deleted after binding creation.`,
  1087→      );
  1088→    }
  1089→
  1090→    // Step 5: Runtime tenant isolation — verify CURRENT ownership
  1091→    if (instance.tenantId !== entitlement.tenantId) {
  1092→      throw new Error(
  1093→        `Cross-tenant provider instance access denied: binding ${bindingId} belongs to tenant ` +
  1094→        `"${entitlement.tenantId}" but instance "${instance.id}" belongs to tenant "${instance.tenantId}".`,
  1095→      );
  1096→    }
  1097→
  1098→    // Step 6: Runtime type match — verify CURRENT providerType
  1099→    if (instance.providerType !== fullBinding.providerType) {
  1100→      throw new Error(
  1101→        `Provider type mismatch: binding providerType is "${fullBinding.providerType}" ` +
  1102→        `but instance "${instance.id}" has providerType "${instance.providerType}".`,
  1103→      );
  1104→    }
  1105→
  1106→    // Step 7: Instance status — must be active
  1107→    if (instance.status !== "active") {
  1108→      throw new Error(
  1109→        `Provider instance "${instance.id}" status is "${instance.status}". ` +
  1110→        `Only "active" instances can be used for runtime operations.`,
  1111→      );
  1112→    }
  1113→
  1114→    providerInstance = {
  1115→      ...instance,
  1116→      configuration: instance.configuration ? JSON.parse(instance.configuration) : null,
  1117→    };
  1118→  }
  1119→
  1120→  // Build the adapter inputs
  1121→  const bindingInput: import("./adapter").ProviderResourceBindingInput = {
  1122→    id: fullBinding.id,
  1123→    entitlementId: fullBinding.entitlementId,
  1124→    providerType: fullBinding.providerType,
  1125→    providerResourceId: fullBinding.providerResourceId,
  1126→    providerMetadata: fullBinding.providerMetadata ? JSON.parse(fullBinding.providerMetadata) : null,
  1127→    status: fullBinding.status,
  1128→    provisioningState: fullBinding.provisioningState,
  1129→    providerInstanceId: fullBinding.providerInstanceId,
  1130→    providerInstanceConfiguration: providerInstance?.configuration ?? null,
  1131→  };
  1132→
  1133→  const entitlementInput: import("./adapter").ConnectivityEntitlementInput = {
  1134→    id: entitlement.id,
  1135→    tenantId: entitlement.tenantId,
  1136→    subscriptionId: entitlement.subscriptionId,
  1137→    status: entitlement.status,
  1138→    capabilityType: entitlement.capability.type,
  1139→    capabilitySet: JSON.parse(entitlement.capabilitySet),
  1140→    policy: entitlement.policy ? JSON.parse(entitlement.policy) : null,
  1141→    validFrom: entitlement.validFrom,
  1142→    validUntil: entitlement.validUntil,
  1143→  };
  1144→
  1145→  return { adapter, binding: bindingInput, entitlement: entitlementInput, providerInstance };
  1146→}
  1147→
  1148→// ---------------------------------------------------------------------------
  1149→// Phase 2C.4.5: Durable Provisioning Claim with Lease + Attempt Identity
  1150→// ---------------------------------------------------------------------------
  1151→
  1152→/**
  1153→ * Phase 2C.4.5: Lease duration for provisioning claims (5 minutes).
  1154→ * After this time, another worker may take over the claim.
  1155→ */
  1156→const PROVISIONING_LEASE_MS = 5 * 60 * 1000;
  1157→
  1158→/**
  1159→ * Phase 2C.4.5: Atomic provisioning claim with attempt identity + lease.
  1160→ *
  1161→ * Two claim paths:
  1162→ *   1. UNBOUND → PROVISIONING (initial claim)
  1163→ *   2. PROVISIONING + expired lease → PROVISIONING (lease takeover)
  1164→ *
  1165→ * In both cases, a new provisioningAttemptId is generated and claimExpiresAt
  1166→ * is set. Only the worker holding the attemptId may execute provider
  1167→ * operations or finalize the binding.
  1168→ *
  1169→ * A stale worker (whose attemptId no longer matches) cannot mutate the binding
  1170→ * — the finalization writes are guarded by WHERE provisioningAttemptId = X.
  1171→ *
  1172→ * Returns:
  1173→ *   { claimed: true, attemptId } — this worker owns the provisioning attempt
  1174→ *   { claimed: false, currentStatus } — another active worker owns it
  1175→ */
  1176→export async function claimProvisioning(bindingId: string): Promise<{
  1177→  claimed: boolean;
  1178→  attemptId?: string;
  1179→  currentStatus?: string;
  1180→}> {
  1181→  const now = new Date();
  1182→  const attemptId = `attempt-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`;
  1183→  const claimExpiresAt = new Date(now.getTime() + PROVISIONING_LEASE_MS);
  1184→
  1185→  // Path 1: UNBOUND → PROVISIONING (initial claim)
  1186→  const initialClaim = await db.providerResourceBinding.updateMany({
  1187→    where: { id: bindingId, status: BINDING_STATES.UNBOUND },
  1188→    data: {
  1189→      status: BINDING_STATES.PROVISIONING,
  1190→      provisioningState: "PENDING",
  1191→      provisioningAttemptId: attemptId,
  1192→      claimExpiresAt,
  1193→    },
  1194→  });
  1195→
  1196→  if (initialClaim.count > 0) {
  1197→    logger.info("connectivity.provisioning_claimed", { bindingId, attemptId, claimType: "initial" });
  1198→    return { claimed: true, attemptId };
  1199→  }
  1200→
  1201→  // Path 2: PROVISIONING + expired lease → takeover
  1202→  // The current claim's lease has expired — another worker may take over.
  1203→  const current = await db.providerResourceBinding.findUnique({
  1204→    where: { id: bindingId },
  1205→    select: { status: true, claimExpiresAt: true, provisioningAttemptId: true },
  1206→  });
  1207→
  1208→  if (!current) {
  1209→    return { claimed: false, currentStatus: "not_found" };
  1210→  }
  1211→
  1212→  // If the current status is not PROVISIONING, we can't claim (it's BOUND, FAILED, etc.)
  1213→  if (current.status !== BINDING_STATES.PROVISIONING) {
  1214→    return { claimed: false, currentStatus: current.status };
  1215→  }
  1216→
  1217→  // Check if the lease has expired
  1218→  if (current.claimExpiresAt && current.claimExpiresAt > now) {
  1219→    // Lease is still active — another worker owns it
  1220→    return { claimed: false, currentStatus: current.status };
  1221→  }
  1222→
  1223→  // Lease has expired — take over atomically
  1224→  const takeoverClaim = await db.providerResourceBinding.updateMany({
  1225→    where: {
  1226→      id: bindingId,
  1227→      status: BINDING_STATES.PROVISIONING,
  1228→      // Only take over if the lease has expired (or is null — legacy)
  1229→      OR: [
  1230→        { claimExpiresAt: { lt: now } },
  1231→        { claimExpiresAt: null },
  1232→      ],
  1233→    },
  1234→    data: {
  1235→      provisioningAttemptId: attemptId,
  1236→      claimExpiresAt,
  1237→    },
  1238→  });
  1239→
  1240→  if (takeoverClaim.count > 0) {
  1241→    logger.info("connectivity.provisioning_claimed", {
  1242→      bindingId, attemptId, claimType: "takeover",
  1243→      previousAttemptId: current.provisioningAttemptId,
  1244→    });
  1245→    return { claimed: true, attemptId };
  1246→  }
  1247→
  1248→  // Another worker took over between our read and write
  1249→  return { claimed: false, currentStatus: current.status };
  1250→}
  1251→
  1252→/**
  1253→ * Phase 2C.4.5: Claim-guarded finalization.
  1254→ *
  1255→ * Transitions the binding to a new state ONLY if the caller holds the
  1256→ * current provisioning attempt. This prevents a stale worker from
  1257→ * mutating the binding after its lease has expired and another worker
  1258→ * has taken over.
  1259→ *
  1260→ * WHERE id = bindingId
  1261→ *   AND provisioningAttemptId = attemptId
  1262→ *   AND status = expectedStatus
  1263→ */
  1264→async function claimGuardedTransition(input: {
  1265→  bindingId: string;
  1266→  attemptId: string;
  1267→  toState: BindingState;
  1268→  providerResourceId?: string;
  1269→  providerMetadata?: Record<string, unknown>;
  1270→  provisioningState?: string;
  1271→  reason?: string;
  1272→}): Promise<{ transitioned: boolean }> {
  1273→  const updateData: Record<string, unknown> = {
  1274→    status: input.toState,
  1275→    failureReason: input.reason ?? null,
  1276→    provisioningAttemptId: null, // Clear the claim after finalization
  1277→    claimExpiresAt: null,
  1278→  };
  1279→  if (input.providerResourceId !== undefined) {
  1280→    updateData.providerResourceId = input.providerResourceId;
  1281→  }
  1282→  if (input.providerMetadata !== undefined) {
  1283→    updateData.providerMetadata = JSON.stringify(input.providerMetadata);
  1284→  }
  1285→  if (input.provisioningState !== undefined) {
  1286→    updateData.provisioningState = input.provisioningState;
  1287→  }
  1288→
  1289→  const result = await db.providerResourceBinding.updateMany({
  1290→    where: {
  1291→      id: input.bindingId,
  1292→      provisioningAttemptId: input.attemptId,
  1293→      status: BINDING_STATES.PROVISIONING,
  1294→    },
  1295→    data: updateData,
  1296→  });
  1297→
  1298→  if (result.count === 0) {
  1299→    // The caller's attemptId no longer matches — lease was taken over by another worker
  1300→    logger.warn("connectivity.provisioning_stale_worker", {
  1301→      bindingId: input.bindingId,
  1302→      attemptId: input.attemptId,
  1303→      message: "CRITICAL: Stale worker attempted finalization — claim was taken over by another worker.",
  1304→    });
  1305→    return { transitioned: false };
  1306→  }
  1307→
  1308→  return { transitioned: true };
  1309→}
  1310→
  1311→/**
  1312→ * Phase 2C.4.5: Kernel-level provisioning orchestration with lease.
  1313→ *
  1314→ * This is the canonical way to provision a binding. It:
  1315→ *   1. Atomically claims the provisioning (UNBOUND → PROVISIONING)
  1316→ *      with a unique attemptId + lease expiry
  1317→ *   2. Only the winner calls the adapter's provision()
  1318→ *   3. The loser gets claim_lost and reconciles instead
  1319→ *   4. On success: claim-guarded transition to BOUND
  1320→ *   5. On failure: claim-guarded transition to FAILED
  1321→ *   6. If the winner crashes, the lease expires and another worker takes over
  1322→ *
  1323→ * A stale worker (whose attemptId no longer matches) cannot finalize the binding.
  1324→ */
  1325→export async function provisionBinding(bindingId: string): Promise<{
  1326→  status: "success" | "failed_retryable" | "failed_permanent" | "already_provisioned" | "claim_lost";
  1327→  providerResourceId?: string;
  1328→  error?: string;
  1329→}> {
  1330→  // Step 1: Resolve the runtime context
  1331→  const { adapter, binding, entitlement } = await resolveBindingRuntime(bindingId);
  1332→
  1333→  // Step 2: If already BOUND, return success (idempotent)
  1334→  if (binding.status === BINDING_STATES.BOUND) {
  1335→    return { status: "already_provisioned", providerResourceId: binding.providerResourceId ?? undefined };
  1336→  }
  1337→
  1338→  // Step 3: Atomic claim (with lease + attempt identity)
  1339→  const claim = await claimProvisioning(bindingId);
  1340→  if (!claim.claimed) {
  1341→    return { status: "claim_lost", error: `Another worker owns the provisioning claim (status: ${claim.currentStatus})` };
  1342→  }
  1343→
  1344→  const attemptId = claim.attemptId!;
  1345→
  1346→  // Step 4: Winner — call the adapter's provision()
  1347→  try {
  1348→    // Re-resolve the binding (it's now PROVISIONING with our attemptId)
  1349→    const { binding: updatedBinding, entitlement: updatedEntitlement } = await resolveBindingRuntime(bindingId);
  1350→
  1351→    const result = await adapter.provision({
  1352→      entitlement: updatedEntitlement,
  1353→      binding: updatedBinding,
  1354→    });
  1355→
  1356→    if (result.status === "success") {
  1357→      // Step 5: Claim-guarded transition to BOUND
  1358→      const transitioned = await claimGuardedTransition({
  1359→        bindingId,
  1360→        attemptId,
  1361→        toState: BINDING_STATES.BOUND,
  1362→        providerResourceId: result.providerResourceId,
  1363→        providerMetadata: result.providerMetadata,
  1364→        provisioningState: "COMPLETED",
  1365→      });
  1366→
  1367→      if (!transitioned) {
  1368→        // Our lease was taken over by another worker — our result is stale
  1369→        logger.warn("connectivity.provisioning_result_stale", {
  1370→          bindingId, attemptId,
  1371→          message: "Provider operation succeeded but claim was taken over — result is stale.",
  1372→        });
  1373→        return { status: "claim_lost", error: "Claim was taken over before finalization" };
  1374→      }
  1375→
  1376→      logger.info("connectivity.provisioning_succeeded", {
  1377→        bindingId, attemptId, providerResourceId: result.providerResourceId,
  1378→      });
  1379→
  1380→      return { status: "success", providerResourceId: result.providerResourceId };
  1381→    } else {
  1382→      // Provisioning failed — claim-guarded transition to FAILED
  1383→      const transitioned = await claimGuardedTransition({
  1384→        bindingId,
  1385→        attemptId,
  1386→        toState: BINDING_STATES.FAILED,
  1387→        provisioningState: "FAILED",
  1388→        reason: result.error,
  1389→      });
  1390→
  1391→      if (!transitioned) {
  1392→        // Our lease was taken over — another worker is handling it now
  1393→        return { status: "claim_lost", error: "Claim was taken over before failure finalization" };
  1394→      }
  1395→
  1396→      logger.warn("connectivity.provisioning_failed", {
  1397→        bindingId, attemptId, error: result.error, classification: result.status,
  1398→      });
  1399→
  1400→      return { status: result.status, error: result.error };
  1401→    }
  1402→  } catch (err) {
  1403→    // Unexpected error — claim-guarded transition to FAILED
  1404→    const errorMsg = err instanceof Error ? err.message : String(err);
  1405→    await claimGuardedTransition({
  1406→      bindingId,
  1407→      attemptId,
  1408→      toState: BINDING_STATES.FAILED,
  1409→      provisioningState: "FAILED",
  1410→      reason: errorMsg,
  1411→    }).catch(() => {});
  1412→
  1413→    logger.error("connectivity.provisioning_error", { bindingId, attemptId, error: errorMsg });
  1414→
  1415→    return { status: "failed_permanent", error: errorMsg };
  1416→  }
  1417→}
