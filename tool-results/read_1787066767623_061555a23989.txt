     1→/**
     2→ * Reseller Balance service — prepaid balance + reservation lifecycle.
     3→ *
     4→ * Phase 2B.2: The reseller balance uses a RESERVE → SETTLE / RELEASE
     5→ * lifecycle around fulfillment, exactly like provider credit reservations.
     6→ *
     7→ * Flow:
     8→ *   1. reserveResellerBalance(orderId, amount) — moves funds from available
     9→ *      to reserved (balanceMinor decreases, reservation created as RESERVED)
    10→ *   2a. On successful fulfillment: settleResellerReservation(orderId) —
    11→ *       recognizes revenue (Dr Reseller Funds Liability, Cr Sales Revenue +
    12→ *       Cr Platform Fee Revenue)
    13→ *   2b. On failed fulfillment: releaseResellerReservation(orderId) —
    14→ *       returns reserved funds to available balance (no revenue recognized)
    15→ *
    16→ * Financial atomicity: operational balance + reservation + ledger posting
    17→ * are linked via idempotency keys. If the ledger posting fails, the
    18→ * reservation is marked reconciliation_required (NOT silently swallowed).
    19→ * A background worker (processDueCreditIssuances) retries the posting.
    20→ *
    21→ * Deposits (Phase 2B.2): balance is ONLY credited after a real payment
    22→ * event. The deposit flow is:
    23→ *   1. createDepositIntent(amount) — creates a TenantDepositPayment +
    24→ *      payment provider intent
    25→ *   2. confirmDepositPayment(providerReference) — server-side verifies
    26→ *      the payment succeeded
    27→ *   3. creditDepositBalance(depositPaymentId) — credits TenantBalance +
    28→ *      posts ledger entry (Dr Cash, Cr Reseller Funds Liability)
    29→ *
    30→ * The mock deposit route is blocked in production (PAYMENT_PROVIDER !== "mock").
    31→ */
    32→

    45→/** Get or create the tenant's balance record. */
    46→export async function getOrCreateTenantBalance(tenantId: string) {
    47→  let balance = await db.tenantBalance.findUnique({ where: { tenantId } });
    48→  if (!balance) {
    49→    balance = await db.tenantBalance.create({ data: { tenantId } });
    50→  }
    51→  return balance;
    52→}
    53→
    54→/**
    55→ * Phase 2B.2.7: Concurrency-safe per-tenant sequence allocation.
    56→ *
    57→ * This MUST be called within a transaction that holds a FOR UPDATE lock on
    58→ * the TenantBalance row. The caller is responsible for acquiring the lock
    59→ * (either via an explicit $queryRaw`SELECT ... FOR UPDATE` or by being inside
    60→ * a transaction that already locked the balance row).
    61→ *
    62→ * The function reads nextTransactionSequence from TenantBalance, returns it,
    63→ * and increments it. Because the caller holds the FOR UPDATE lock, no other
    64→ * transaction can read/increment the same sequence number — this prevents
    65→ * the MAX+1 race condition.
    66→ *
    67→ * For call sites outside an existing transaction, use
    68→ * `allocateSequenceAndCreateTransaction()` which wraps the lock + allocate +
    69→ * create in a single transaction.
    70→ */
    71→async function getNextSequenceNumber(tx: any, tenantId: string): Promise<number> {
    72→  // Read and increment nextTransactionSequence atomically.
    73→  // The caller MUST hold a FOR UPDATE lock on the TenantBalance row.
    74→  // Phase 2B.2.8: The TenantBalance row MUST already exist — the sequence
    75→  // allocator must NOT silently create financial state. If the row doesn't
    76→  // exist, throw explicitly so the caller can initialize it through the
    77→  // canonical getOrCreateTenantBalance path before any financial transaction.
    78→  const balance = await tx.tenantBalance.findUnique({
    79→    where: { tenantId },
    80→    select: { nextTransactionSequence: true },
    81→  });
    82→
    83→  if (!balance) {
    84→    throw new AppError(
    85→      "internal",
    86→      `TenantBalance not found for tenant ${tenantId}`,
    87→      500,
    88→      "TenantBalance row must be created before any financial transaction. The sequence allocator does not create financial state.",
    89→    );
    90→  }
    91→
    92→  const seq = balance.nextTransactionSequence;
    93→  await tx.tenantBalance.update({
    94→    where: { tenantId },
    95→    data: { nextTransactionSequence: seq + 1 },
    96→  });
    97→  return seq;
    98→}
    99→
   100→/**
   101→ * Phase 2B.2.7: Acquire a FOR UPDATE lock on the TenantBalance row.
   102→ * Must be called at the start of a transaction before any sequence allocation.
   103→ */
   104→async function lockTenantBalance(tx: any, tenantId: string): Promise<void> {
   105→  await tx.$executeRaw`SELECT 1 FROM "TenantBalance" WHERE "tenantId" = ${tenantId} FOR UPDATE`;
   106→}
   107→
   108→/**
   109→ * Phase 2B.2.6: Log (not silently swallow) a projectionReconciled update failure.
   110→ * If this update fails, the worker will re-scan the reservation on the next cron
   111→ * run (projectionReconciled stays false). That's safe but wasteful, so we log it.
   112→ */
   113→function logProjectionUpdateFailure(reservationId: string, context: string, err: unknown) {
   114→  const errMsg = err instanceof Error ? err.message : String(err);
   115→  logger.warn("reseller.projection_reconciled_update_failed", {
   116→    reservationId,
   117→    context,
   118→    error: errMsg,
   119→    message: `projectionReconciled update failed (${context}). The worker will re-scan this reservation on the next cron run. This is safe but wasteful.`,
   120→  });
   121→}

   123→/** Get the current available balance (minor units). */
   124→export async function getTenantBalanceMinor(tenantId: string): Promise<number> {
   125→  const balance = await getOrCreateTenantBalance(tenantId);
   126→  return balance.balanceMinor;

   129→/**
   130→ * Phase 2B.2.8: Get the current available balance using a transaction client.
   131→ * This MUST be used inside a $transaction to ensure the balance read comes
   132→ * from the same transactional snapshot as the sequence allocation and
   133→ * TenantTransaction creation. Using the global db helper inside a transaction
   134→ * would read from a different connection/snapshot, breaking coherence.
   135→ */
   136→async function getTenantBalanceMinorTx(tx: any, tenantId: string): Promise<number> {
   137→  const balance = await tx.tenantBalance.findUnique({
   138→    where: { tenantId },
   139→    select: { balanceMinor: true },
   140→  });
   141→  return balance?.balanceMinor ?? 0;
   142→}

   148→/**
   149→ * Reserve funds for an order. Moves `amountMinor` from available balance to
   150→ * a RESERVED reservation. The available balance decreases immediately, but
   151→ * NO revenue is recognized — that happens on SETTLE.
   152→ *
   153→ * Concurrency-safe: FOR UPDATE lock on the balance row.
   154→ * Idempotent: same idempotencyKey returns the same reservation.
   155→ * Fails if insufficient available balance (no negative balance).
   156→ */
   157→export async function reserveResellerBalance(input: {
   158→  tenantId: string;
   159→  userId: string;
   160→  orderId: string;
   161→  amountMinor: number;
   162→  platformFeeMinor: number;
   163→  idempotencyKey: string;
   164→  description?: string;
   165→}): Promise<{ reservationId: string; balanceMinor: number }> {
   166→  if (input.amountMinor <= 0) {
   167→    throw new AppError("validation", "Reserve amount must be positive", 400, "Reserve amount must be greater than zero.");
   168→  }
   169→
   170→  const result = await db.$transaction(async (tx) => {
   171→    // Idempotency: check for existing reservation
   172→    const existing = await tx.tenantBalanceReservation.findUnique({
   173→      where: { idempotencyKey: input.idempotencyKey },
   174→    });
   175→    if (existing) {
   176→      logger.info("reseller.reserve_idempotent_replay", { idempotencyKey: input.idempotencyKey, reservationId: existing.id });
   177→      // Fetch current balance for the response
   178→      const bal = await tx.tenantBalance.findUnique({ where: { tenantId: input.tenantId } });
   179→      return { reservationId: existing.id, balanceMinor: bal?.balanceMinor ?? 0, existing: true };
   180→    }
   181→
   182→    // Also check by orderId (one reservation per order)
   183→    const existingByOrder = await tx.tenantBalanceReservation.findUnique({
   184→      where: { orderId: input.orderId },
   185→    });
   186→    if (existingByOrder) {
   187→      logger.info("reseller.reserve_existing_order", { orderId: input.orderId, reservationId: existingByOrder.id });
   188→      const bal = await tx.tenantBalance.findUnique({ where: { tenantId: input.tenantId } });
   189→      return { reservationId: existingByOrder.id, balanceMinor: bal?.balanceMinor ?? 0, existing: true };
   190→    }
   191→
   192→    // Lock the balance row for safe concurrent reservation
   193→    const locked: Array<{ balanceMinor: number }> = await tx.$queryRaw`
   194→      SELECT "balanceMinor" FROM "TenantBalance" WHERE "tenantId" = ${input.tenantId} FOR UPDATE
   195→    `;
   196→    const currentBalance = locked.length ? locked[0].balanceMinor : 0;
   197→
   198→    if (currentBalance < input.amountMinor) {
   199→      throw new AppError(
   200→        "validation",
   201→        `Insufficient balance: ${currentBalance} < ${input.amountMinor}`,
   202→        402,
   203→        `Insufficient reseller balance. Current: $${(currentBalance / 100).toFixed(2)}, required: $${(input.amountMinor / 100).toFixed(2)}. Please deposit more funds.`,
   204→      );
   205→    }
   206→
   207→    // Decrement available balance (funds are now reserved, not spent)
   208→    const balance = await tx.tenantBalance.upsert({
   209→      where: { tenantId: input.tenantId },
   210→      create: { tenantId: input.tenantId, balanceMinor: -input.amountMinor },
   211→      update: { balanceMinor: { decrement: input.amountMinor } },
   212→    });
   213→
   214→    // Create reservation record
   215→    const reservation = await tx.tenantBalanceReservation.create({
   216→      data: {
   217→        tenantId: input.tenantId,
   218→        orderId: input.orderId,
   219→        amountMinor: input.amountMinor,
   220→        platformFeeMinor: input.platformFeeMinor,
   221→        state: "RESERVED",
   222→        idempotencyKey: input.idempotencyKey,
   223→      },
   224→    });
   225→
   226→    // Create a TenantTransaction for the reservation (negative = funds held)
   227→    const reserveSeq = await getNextSequenceNumber(tx, input.tenantId);
   228→    await tx.tenantTransaction.create({
   229→      data: {
   230→        tenantId: input.tenantId,
   231→        type: "reservation",
   232→        amountMinor: -input.amountMinor,
   233→        balanceAfter: balance.balanceMinor,
   234→        orderId: input.orderId,
   235→        description: input.description ?? "Balance reserved for order",
   236→        idempotencyKey: `reserve_${input.idempotencyKey}`,
   237→        sequenceNumber: reserveSeq,
   238→      },
   239→    });
   240→
   241→    return { reservationId: reservation.id, balanceMinor: balance.balanceMinor, existing: false };
   242→  }, { timeout: 30000, maxWait: 15000 });
   243→
   244→  if (!result.existing) {
   245→    await audit({
   246→      tenantId: input.tenantId,
   247→      userId: input.userId,
   248→      action: "reseller.balance_reserved",
   249→      entity: "tenant_balance",
   250→      entityId: input.tenantId,
   251→      detail: { amount: input.amountMinor, orderId: input.orderId, balanceAfter: result.balanceMinor },
   252→    });
   253→    logger.info("reseller.balance_reserved", { tenantId: input.tenantId, amount: input.amountMinor, orderId: input.orderId, balance: result.balanceMinor });
   254→  }
   255→
   256→  return { reservationId: result.reservationId, balanceMinor: result.balanceMinor };
   257→}

   259→/**
   260→ * Settle a reservation after successful fulfillment. Recognizes revenue:
   261→ *   Dr Reseller Funds Liability (the reserved amount)
   262→ *   Cr Sales Revenue (retail - platform fee)
   263→ *   Cr Platform Fee Revenue (platform fee)
   264→ *
   265→ * Phase 2B.2.1: If the ledger posting fails, the reservation transitions
   266→ * RESERVED → RECONCILIATION_REQUIRED (NOT thrown, NOT released). The funds
   267→ * remain reserved — the service is already active, so we must NOT return
   268→ * funds to the reseller. The processDueResellerReservationReconciliation
   269→ * worker retries the ledger posting until it succeeds.
   270→ *
   271→ * The reservation state machine:
   272→ *   RESERVED → SETTLED (ledger posted, revenue recognized)
   273→ *   RESERVED → RECONCILIATION_REQUIRED (ledger failed, retry pending)
   274→ *   RECONCILIATION_REQUIRED → SETTLED (reconciliation worker succeeds)
   275→ *   RESERVED → RELEASED (fulfillment failed, funds returned)
   276→ *
   277→ * Idempotent: if already SETTLED, returns the existing ledgerTxnId.
   278→ * If RECONCILIATION_REQUIRED, returns that state (caller knows to retry).
   279→ */
   280→export async function settleResellerReservation(input: {
   281→  tenantId: string;
   282→  userId: string;
   283→  orderId: string;
   284→}): Promise<{ reservationId: string; ledgerTransactionId: string; state: string }> {
   285→  await ensureChartOfAccounts();
   286→
   287→  const reservation = await db.tenantBalanceReservation.findUnique({
   288→    where: { orderId: input.orderId },
   289→  });
   290→  if (!reservation) {
   291→    throw new AppError("not_found", "Reservation not found", 404, `No reservation found for order ${input.orderId}.`);
   292→  }
   293→  if (reservation.tenantId !== input.tenantId) {
   294→    throw new AppError("authorization", "Cross-tenant access denied", 403, "This reservation belongs to a different tenant.");
   295→  }
   296→
   297→  // Idempotent: if already settled, return the existing ledger txn
   298→  if (reservation.state === "SETTLED") {
   299→    logger.info("reseller.settle_idempotent_replay", { orderId: input.orderId, reservationId: reservation.id });
   300→    return {
   301→      reservationId: reservation.id,
   302→      ledgerTransactionId: reservation.ledgerTransactionId ?? "",
   303→      state: reservation.state,
   304→    };
   305→  }
   306→
   307→  if (reservation.state === "RELEASED") {
   308→    throw new AppError("conflict", "Cannot settle a released reservation", 409, "This reservation was released (fulfillment failed).");
   309→  }
   310→
   311→  // State must be RESERVED or RECONCILIATION_REQUIRED to attempt settlement
   312→  if (reservation.state !== "RESERVED" && reservation.state !== "RECONCILIATION_REQUIRED") {
   313→    throw new AppError("conflict", `Cannot settle reservation in state ${reservation.state}`, 409, "The reservation is in an unexpected state.");
   314→  }
   315→
   316→  // Post the ledger entry (Dr Reseller Funds Liability, Cr Sales Revenue + Cr Platform Fee Revenue)
   317→  // Idempotent: ledgerResellerPurchase replays if the idempotencyKey already exists
   318→  let ledgerTxnId: string;
   319→  try {
   320→    ledgerTxnId = await ledgerResellerPurchase({
   321→      tenantId: input.tenantId,
   322→      userId: input.userId,
   323→      orderId: input.orderId,
   324→      retailPriceMinor: reservation.amountMinor,
   325→      platformFeeMinor: reservation.platformFeeMinor,
   326→      idempotencyKey: `${reservation.idempotencyKey}:ledger`,
   327→    });
   328→  } catch (err) {
   329→    // Phase 2B.2.1: Transition to RECONCILIATION_REQUIRED (do NOT throw, do NOT release).
   330→    // The service is already active — we must preserve the reservation and
   331→    // retry the ledger posting via the reconciliation worker.
   332→    const errorMsg = err instanceof Error ? err.message : String(err);
   333→    logger.error("reseller.settle_ledger_failed", { orderId: input.orderId, reservationId: reservation.id, error: errorMsg });
   334→    await db.tenantBalanceReservation.updateMany({
   335→      where: { id: reservation.id, state: { in: ["RESERVED", "RECONCILIATION_REQUIRED"] } },
   336→      data: { state: "RECONCILIATION_REQUIRED", reconciliationReason: "LEDGER_POSTING_FAILED", failureReason: `Ledger posting failed: ${errorMsg}` },
   337→    }).catch((updateErr) => {
   338→      const updateMsg = updateErr instanceof Error ? updateErr.message : String(updateErr);
   339→      logger.error("reseller.settle_state_update_failed", {
   340→        reservationId: reservation.id,
   341→        ledgerError: errorMsg,
   342→        updateError: updateMsg,
   343→        message: "CRITICAL: Reservation state update to RECONCILIATION_REQUIRED failed after ledger failure. The reconciliation worker will recover it via the stale-state scan.",
   344→      });
   345→    });
   346→    return {
   347→      reservationId: reservation.id,
   348→      ledgerTransactionId: "",
   349→      state: "RECONCILIATION_REQUIRED",
   350→    };
   351→  }
   352→
   353→  // Transition RESERVED/RECONCILIATION_REQUIRED → SETTLED (status-guarded)
   354→  // Phase 2B.2.4: Clear BOTH failureReason AND reconciliationReason on successful settlement.
   355→  const updated = await db.tenantBalanceReservation.updateMany({
   356→    where: { id: reservation.id, state: { in: ["RESERVED", "RECONCILIATION_REQUIRED"] } },
   357→    data: { state: "SETTLED", settledAt: new Date(), ledgerTransactionId: ledgerTxnId, failureReason: null, reconciliationReason: null },
   358→  });
   359→
   360→  if (updated.count === 0) {
   361→    // Another concurrent call settled it — fetch the existing ledger txn
   362→    const existing = await db.tenantBalanceReservation.findUnique({ where: { id: reservation.id } });
   363→    return {
   364→      reservationId: reservation.id,
   365→      ledgerTransactionId: existing?.ledgerTransactionId ?? ledgerTxnId,
   366→      state: existing?.state ?? "SETTLED",
   367→    };
   368→  }
   369→
   370→  // Phase 2B.2.1 §6: Create a TenantTransaction for the settlement.
   371→  // Phase 2B.2.7: Use a transaction with FOR UPDATE lock on TenantBalance
   372→  // for concurrency-safe sequence allocation.
   373→  // Do NOT silently swallow — if this fails, the reconciliation worker will
   374→  // repair it (the reservation is already SETTLED with the ledgerTransactionId).
   375→  try {
   376→    await db.$transaction(async (tx) => {
   377→      await lockTenantBalance(tx, input.tenantId);
   378→      const settleSeq = await getNextSequenceNumber(tx, input.tenantId);
   379→      const currentBalance = await getTenantBalanceMinorTx(tx, input.tenantId);
   380→      await tx.tenantTransaction.create({
   381→        data: {
   382→          tenantId: input.tenantId,
   383→          type: "purchase",
   384→          amountMinor: -reservation.amountMinor,
   385→          balanceAfter: currentBalance,
   386→          orderId: input.orderId,
   387→          description: "Connectivity purchase settled",
   388→          idempotencyKey: `settle_${reservation.idempotencyKey}`,
   389→          ledgerTransactionId: ledgerTxnId,
   390→          sequenceNumber: settleSeq,
   391→        },
   392→      });
   393→    }, { timeout: 30000, maxWait: 15000 });
   394→
   395→    // Phase 2B.2.5: Mark the projection as reconciled so the worker doesn't
   396→    // need to scan this reservation on future cron runs.
   397→    await db.tenantBalanceReservation.update({
   398→      where: { id: reservation.id },
   399→      data: { projectionReconciled: true },
   400→    }).catch((err) => logProjectionUpdateFailure(reservation.id, "settleResellerReservation.normal", err));
   401→  } catch (err) {
   402→    // The unique constraint on idempotencyKey may have caught a duplicate
   403→    // (concurrent settlement). If so, this is fine. If it's a real error,
   404→    // log it — the reservation is SETTLED and the ledger is posted, so the
   405→    // financial truth is correct. The operational projection (TenantTransaction)
   406→    // can be repaired by the reconciliation worker.
   407→    // Phase 2B.2.5: projectionReconciled stays false, so the worker will check.
   408→    const errMsg = err instanceof Error ? err.message : String(err);
   409→    if (errMsg.includes("Unique constraint") || errMsg.includes("P2002")) {
   410→      logger.info("reseller.settle_txn_idempotent_replay", { reservationId: reservation.id, orderId: input.orderId });
   411→      // The TenantTransaction already exists (concurrent call) — mark as reconciled
   412→      await db.tenantBalanceReservation.update({
   413→        where: { id: reservation.id },
   414→        data: { projectionReconciled: true },
   415→      }).catch((err) => logProjectionUpdateFailure(reservation.id, "settleResellerReservation.idempotent_replay", err));
   416→    } else {
   417→      logger.error("reseller.settle_txn_creation_failed", {
   418→        reservationId: reservation.id,
   419→        orderId: input.orderId,
   420→        ledgerTxnId,
   421→        error: errMsg,
   422→        message: "CRITICAL: Reservation SETTLED and ledger posted but TenantTransaction creation failed. The reconciliation worker will repair the operational projection.",
   423→      });
   424→    }
   425→  }
   426→
   427→  await audit({
   428→    tenantId: input.tenantId,
   429→    userId: input.userId,
   430→    action: "reseller.reservation_settled",
   431→    entity: "tenant_balance",
   432→    entityId: reservation.id,
   433→    detail: { amount: reservation.amountMinor, orderId: input.orderId, ledgerTxnId },
   434→  });
   435→  logger.info("reseller.reservation_settled", { tenantId: input.tenantId, orderId: input.orderId, reservationId: reservation.id, ledgerTxnId });
   436→
   437→  return { reservationId: reservation.id, ledgerTransactionId: ledgerTxnId, state: "SETTLED" };
   438→}

   440→/**
   441→ * Release a reservation after failed fulfillment. Returns the reserved funds
   442→ * to the available balance. NO revenue is recognized.
   443→ *
   444→ * The reservation transitions RESERVED → RELEASED. The available balance is
   445→ * incremented (the funds that were held are returned).
   446→ *
   447→ * Idempotent: if already RELEASED, returns the existing state.
   448→ */
   449→export async function releaseResellerReservation(input: {
   450→  tenantId: string;
   451→  userId: string;
   452→  orderId: string;
   453→  reason?: string;
   454→}): Promise<{ reservationId: string; balanceMinor: number; state: string }> {
   455→  const reservation = await db.tenantBalanceReservation.findUnique({
   456→    where: { orderId: input.orderId },
   457→  });
   458→  if (!reservation) {
   459→    throw new AppError("not_found", "Reservation not found", 404, `No reservation found for order ${input.orderId}.`);
   460→  }
   461→  if (reservation.tenantId !== input.tenantId) {
   462→    throw new AppError("authorization", "Cross-tenant access denied", 403, "This reservation belongs to a different tenant.");
   463→  }
   464→
   465→  // Idempotent: if already released, return the existing state
   466→  if (reservation.state === "RELEASED") {
   467→    logger.info("reseller.release_idempotent_replay", { orderId: input.orderId, reservationId: reservation.id });
   468→    const bal = await getTenantBalanceMinor(input.tenantId);
   469→    return { reservationId: reservation.id, balanceMinor: bal, state: reservation.state };
   470→  }
   471→
   472→  if (reservation.state === "SETTLED") {
   473→    throw new AppError("conflict", "Cannot release a settled reservation", 409, "This reservation was already settled (fulfillment succeeded).");
   474→  }
   475→
   476→  // Return funds to available balance + transition RESERVED → RELEASED
   477→  const result = await db.$transaction(async (tx) => {
   478→    // Status-guarded transition: only RESERVED → RELEASED
   479→    const updated = await tx.tenantBalanceReservation.updateMany({
   480→      where: { id: reservation.id, state: "RESERVED" },
   481→      data: { state: "RELEASED", releasedAt: new Date(), failureReason: input.reason ?? null },
   482→    });
   483→    if (updated.count === 0) {
   484→      // Another concurrent call released/settled it
   485→      const existing = await tx.tenantBalanceReservation.findUnique({ where: { id: reservation.id } });
   486→      const bal = await tx.tenantBalance.findUnique({ where: { tenantId: input.tenantId } });
   487→      return { balanceMinor: bal?.balanceMinor ?? 0, state: existing?.state ?? "RELEASED", alreadyDone: true };
   488→    }
   489→
   490→    // Phase 2B.2.8: Lock the balance row for concurrency-safe sequence allocation.
   491→    await lockTenantBalance(tx, input.tenantId);
   492→
   493→    // Return funds to available balance
   494→    const balance = await tx.tenantBalance.upsert({
   495→      where: { tenantId: input.tenantId },
   496→      create: { tenantId: input.tenantId, balanceMinor: reservation.amountMinor },
   497→      update: { balanceMinor: { increment: reservation.amountMinor } },
   498→    });
   499→
   500→    // Create a TenantTransaction for the release (positive = funds returned)
   501→    const releaseSeq = await getNextSequenceNumber(tx, input.tenantId);
   502→    await tx.tenantTransaction.create({
   503→      data: {
   504→        tenantId: input.tenantId,
   505→        type: "release",
   506→        amountMinor: reservation.amountMinor,
   507→        balanceAfter: balance.balanceMinor,
   508→        orderId: input.orderId,
   509→        description: "Reservation released (fulfillment failed)",
   510→        idempotencyKey: `release_${reservation.idempotencyKey}`,
   511→        sequenceNumber: releaseSeq,
   512→      },
   513→    });
   514→
   515→    return { balanceMinor: balance.balanceMinor, state: "RELEASED" as const, alreadyDone: false };
   516→  }, { timeout: 30000, maxWait: 15000 });
   517→
   518→  if (!result.alreadyDone) {
   519→    await audit({
   520→      tenantId: input.tenantId,
   521→      userId: input.userId,
   522→      action: "reseller.reservation_released",
   523→      entity: "tenant_balance",
   524→      entityId: reservation.id,
   525→      detail: { amount: reservation.amountMinor, orderId: input.orderId, reason: input.reason },
   526→    });
   527→    logger.info("reseller.reservation_released", { tenantId: input.tenantId, orderId: input.orderId, reservationId: reservation.id, balance: result.balanceMinor });
   528→  }
   529→
   530→  return { reservationId: reservation.id, balanceMinor: result.balanceMinor, state: result.state };
   531→}

   537→/**
   538→ * Create a deposit intent. Creates a TenantDepositPayment record + a payment
   539→ * provider intent. The balance is NOT credited until the payment is verified.
   540→ *
   541→ * Phase 2B.2 §9: The mock provider is only allowed when PAYMENT_PROVIDER=mock
   542→ * (development/test). In production, this route requires a real provider.
   543→ */
   544→export async function createDepositIntent(input: {
   545→  tenantId: string;
   546→  userId: string;
   547→  amountMinor: number;
   548→  currency?: Currency;
   549→  idempotencyKey: string;
   550→}): Promise<{ depositPaymentId: string; providerReference: string; status: string; clientSecret?: string }> {
   551→  if (input.amountMinor <= 0) {
   552→    throw new AppError("validation", "Deposit amount must be positive", 400, "Deposit amount must be greater than zero.");
   553→  }
   554→
   555→  const provider = getPaymentProvider();
   556→
   557→  // Phase 2B.2 §9: Block mock provider in production
   558→  if (provider.isMock && process.env.NODE_ENV === "production") {
   559→    throw new AppError(
   560→      "validation",
   561→      "Mock payment provider not allowed in production",
   562→      403,
   563→      "The mock payment provider cannot be used in production. Configure a real payment provider (PAYMENT_PROVIDER=stripe|paystack|flutterwave).",
   564→    );
   565→  }
   566→
   567→  // Idempotency: check for existing deposit payment
   568→  const existing = await db.tenantDepositPayment.findUnique({
   569→    where: { idempotencyKey: input.idempotencyKey },
   570→  });
   571→  if (existing) {
   572→    logger.info("reseller.deposit_intent_idempotent_replay", { idempotencyKey: input.idempotencyKey, depositId: existing.id });
   573→    return {
   574→      depositPaymentId: existing.id,
   575→      providerReference: existing.providerReference ?? "",
   576→      status: existing.status,
   577→    };
   578→  }
   579→
   580→  // Create the payment intent with the provider
   581→  const intent = await provider.createPaymentIntent({
   582→    amountMinor: input.amountMinor,
   583→    currency: (input.currency ?? "USD") as Currency,
   584→    description: `Reseller deposit for tenant ${input.tenantId}`,
   585→    idempotencyKey: input.idempotencyKey,
   586→    metadata: { tenantId: input.tenantId, type: "reseller_deposit" },
   587→  });
   588→
   589→  // Create the deposit payment record
   590→  const deposit = await db.tenantDepositPayment.create({
   591→    data: {
   592→      tenantId: input.tenantId,
   593→      userId: input.userId,
   594→      amountMinor: input.amountMinor,
   595→      currency: input.currency ?? "USD",
   596→      paymentProvider: provider.id,
   597→      providerReference: intent.providerReference,
   598→      status: intent.status === "succeeded" ? "PAYMENT_SUCCEEDED" : "PAYMENT_PENDING",
   599→      idempotencyKey: input.idempotencyKey,
   600→    },
   601→  });
   602→
   603→  await audit({
   604→    tenantId: input.tenantId,
   605→    userId: input.userId,
   606→    action: "reseller.deposit_intent_created",
   607→    entity: "tenant_deposit_payment",
   608→    entityId: deposit.id,
   609→    detail: { amount: input.amountMinor, providerReference: intent.providerReference },
   610→  });
   611→  logger.info("reseller.deposit_intent_created", { tenantId: input.tenantId, depositId: deposit.id, providerReference: intent.providerReference });
   612→
   613→  return {
   614→    depositPaymentId: deposit.id,
   615→    providerReference: intent.providerReference,
   616→    status: deposit.status,
   617→    clientSecret: intent.clientSecret,
   618→  };
   619→}

   621→/**
   622→ * Confirm a deposit payment (server-side verification). If the payment
   623→ * succeeded, credit the reseller balance + post the ledger entry.
   624→ *
   625→ * Idempotent: if the deposit is already COMPLETED, returns the existing state.
   626→ */
   627→export async function confirmDepositPayment(input: {
   628→  depositPaymentId: string;
   629→  tenantId: string;
   630→  userId: string;
   631→}): Promise<{ status: string; balanceMinor?: number }> {
   632→  const deposit = await db.tenantDepositPayment.findUnique({
   633→    where: { id: input.depositPaymentId },
   634→  });
   635→  if (!deposit) {
   636→    throw new AppError("not_found", "Deposit payment not found", 404, "Deposit payment not found.");
   637→  }
   638→  if (deposit.tenantId !== input.tenantId) {
   639→    throw new AppError("authorization", "Cross-tenant access denied", 403, "This deposit belongs to a different tenant.");
   640→  }
   641→
   642→  // Idempotent: if already completed, return the existing state
   643→  if (deposit.status === "COMPLETED") {
   644→    const bal = await getTenantBalanceMinor(input.tenantId);
   645→    return { status: deposit.status, balanceMinor: bal };
   646→  }
   647→  if (deposit.status === "PAYMENT_FAILED") {
   648→    return { status: deposit.status };
   649→  }
   650→
   651→  // Server-side verification (never trust the client)
   652→  const provider = getPaymentProvider();
   653→
   654→  // For the mock provider (development only), simulate the client-side
   655→  // confirmation step before verification. In production, the client
   656→  // completes payment on the provider's hosted page/SDK, then a webhook
   657→  // triggers confirmation. The mock provider needs an explicit confirmIntent
   658→  // call to transition the intent from "pending" to "succeeded".
   659→  if (provider.isMock && deposit.providerReference) {
   660→    mockPaymentProvider.confirmIntent(deposit.providerReference);
   661→  }
   662→
   663→  const verification = await provider.verifyPayment({
   664→    providerReference: deposit.providerReference!,
   665→    idempotencyKey: deposit.idempotencyKey,
   666→  });
   667→
   668→  if (verification.status === "failed") {
   669→    await db.tenantDepositPayment.update({
   670→      where: { id: deposit.id },
   671→      data: { status: "PAYMENT_FAILED", failureReason: "Payment verification failed" },
   672→    });
   673→    return { status: "PAYMENT_FAILED" };
   674→  }
   675→
   676→  if (verification.status === "pending") {
   677→    await db.tenantDepositPayment.update({
   678→      where: { id: deposit.id },
   679→      data: { status: "PAYMENT_PENDING" },
   680→    }).catch(() => {});
   681→    return { status: "PAYMENT_PENDING" };
   682→  }
   683→
   684→  // verification.status === "succeeded" — update status, then credit the balance
   685→  await db.tenantDepositPayment.update({
   686→    where: { id: deposit.id },
   687→    data: { status: "PAYMENT_SUCCEEDED" },
   688→  }).catch(() => {});
   689→
   690→  return creditDepositBalance({ depositPaymentId: deposit.id, tenantId: input.tenantId, userId: input.userId });
   691→}
   692→
   693→/**
   694→ * Credit the reseller balance after a successful deposit payment.
   695→ * Posts the ledger entry: Dr Cash, Cr Reseller Funds Liability.
   696→ *
   697→ * Atomic: balance update + TenantTransaction in one transaction.
   698→ * Idempotent: status-guarded (only BALANCE_POSTED or PAYMENT_SUCCEEDED → COMPLETED).
   699→ */
   700→export async function creditDepositBalance(input: {
   701→  depositPaymentId: string;
   702→  tenantId: string;
   703→  userId: string;
   704→}): Promise<{ status: string; balanceMinor: number }> {
   705→  await ensureChartOfAccounts();
   706→
   707→  const deposit = await db.tenantDepositPayment.findUnique({
   708→    where: { id: input.depositPaymentId },
   709→  });
   710→  if (!deposit) {
   711→    throw new AppError("not_found", "Deposit payment not found", 404, "Deposit payment not found.");
   712→  }
   713→
   714→  // Status-guarded: only credit if payment succeeded and not already completed
   715→  if (deposit.status === "COMPLETED") {
   716→    const bal = await getTenantBalanceMinor(input.tenantId);
   717→    return { status: "COMPLETED", balanceMinor: bal };
   718→  }
   719→
   720→  if (deposit.status !== "PAYMENT_SUCCEEDED" && deposit.status !== "BALANCE_POSTED") {
   721→    throw new AppError("conflict", `Cannot credit deposit in status ${deposit.status}`, 409, "The deposit payment has not been verified as succeeded.");
   722→  }
   723→
   724→  // Atomic balance credit + transaction record
   725→  const result = await db.$transaction(async (tx) => {
   726→    // Status-guarded transition: PAYMENT_SUCCEEDED → BALANCE_POSTED
   727→    const updated = await tx.tenantDepositPayment.updateMany({
   728→      where: { id: deposit.id, status: { in: ["PAYMENT_SUCCEEDED", "BALANCE_POSTED"] } },
   729→      data: { status: "BALANCE_POSTED" },
   730→    });
   731→    if (updated.count === 0) {
   732→      // Another concurrent call completed it
   733→      const existing = await tx.tenantDepositPayment.findUnique({ where: { id: deposit.id } });
   734→      const bal = await tx.tenantBalance.findUnique({ where: { tenantId: input.tenantId } });
   735→      return { balanceMinor: bal?.balanceMinor ?? 0, status: existing?.status ?? "COMPLETED", alreadyDone: true };
   736→    }
   737→
   738→    // Phase 2B.2.7: Lock the balance row for concurrency-safe sequence allocation.
   739→    await lockTenantBalance(tx, input.tenantId);
   740→
   741→    // Credit the balance
   742→    const balance = await tx.tenantBalance.upsert({
   743→      where: { tenantId: input.tenantId },
   744→      create: { tenantId: input.tenantId, balanceMinor: deposit.amountMinor, totalDepositedMinor: deposit.amountMinor },
   745→      update: {
   746→        balanceMinor: { increment: deposit.amountMinor },
   747→        totalDepositedMinor: { increment: deposit.amountMinor },
   748→      },
   749→    });
   750→
   751→    // Create transaction record
   752→    const depositSeq = await getNextSequenceNumber(tx, input.tenantId);
   753→    const txn = await tx.tenantTransaction.create({
   754→      data: {
   755→        tenantId: input.tenantId,
   756→        type: "deposit",
   757→        amountMinor: deposit.amountMinor,
   758→        balanceAfter: balance.balanceMinor,
   759→        description: `Deposit via ${deposit.paymentProvider}`,
   760→        idempotencyKey: `deposit_${deposit.idempotencyKey}`,
   761→        sequenceNumber: depositSeq,
   762→      },
   763→    });
   764→
   765→    return { balanceMinor: balance.balanceMinor, status: "BALANCE_POSTED" as const, transactionId: txn.id, alreadyDone: false };
   766→  }, { timeout: 30000, maxWait: 15000 });
   767→
   768→  if (result.alreadyDone) {
   769→    return { status: result.status, balanceMinor: result.balanceMinor };
   770→  }
   771→
   772→  // Post ledger entry (outside the balance transaction — idempotent)
   773→  let ledgerTxnId: string | null = null;
   774→  let ledgerFailed = false;
   775→  try {
   776→    ledgerTxnId = await ledgerResellerDeposit({
   777→      tenantId: input.tenantId,
   778→      userId: input.userId,
   779→      amountMinor: deposit.amountMinor,
   780→      idempotencyKey: `${deposit.idempotencyKey}:ledger`,
   781→    });
   782→  } catch (err) {
   783→    // Phase 2B.2 §3: Do NOT silently swallow. Mark as reconciliation_required.
   784→    ledgerFailed = true;
   785→    const errorMsg = err instanceof Error ? err.message : String(err);
   786→    logger.error("reseller.deposit_ledger_failed", { depositId: deposit.id, error: errorMsg });
   787→  }
   788→
   789→  // Link the transaction + ledger, and transition to COMPLETED (or RECONCILIATION_REQUIRED)
   790→  const finalStatus = ledgerFailed ? "RECONCILIATION_REQUIRED" : "COMPLETED";
   791→  await db.tenantDepositPayment.update({
   792→    where: { id: deposit.id },
   793→    data: { status: finalStatus, ledgerTransactionId: ledgerTxnId },
   794→  }).catch((err) => {
   795→    // Phase 2B.2 §3: If this update fails, log at CRITICAL — the balance was
   796→    // credited but the deposit record is stuck in BALANCE_POSTED. The
   797→    // reconciliation worker will find it via the stale-status scan.
   798→    const updateMsg = err instanceof Error ? err.message : String(err);
   799→    logger.error("reseller.deposit_status_update_failed", {
   800→      depositId: deposit.id,
   801→      ledgerFailed,
   802→      updateError: updateMsg,
   803→      message: "CRITICAL: Deposit balance was credited but status update failed. The reconciliation worker will recover it.",
   804→    });
   805→  });
   806→
   807→  // Link the ledger txn to the TenantTransaction (if we have both)
   808→  if (ledgerTxnId && "transactionId" in result && result.transactionId) {
   809→    await db.tenantTransaction.update({
   810→      where: { id: result.transactionId },
   811→      data: { ledgerTransactionId: ledgerTxnId },
   812→    }).catch((err) => {
   813→      // Phase 2B.2 §3: Log at CRITICAL — the balance + ledger are correct but
   814→      // the link between them is missing. The reconciliation worker can repair it.
   815→      const linkMsg = err instanceof Error ? err.message : String(err);
   816→      logger.error("reseller.deposit_link_failed", {
   817→        transactionId: result.transactionId,
   818→        ledgerTxnId,
   819→        linkError: linkMsg,
   820→        message: "CRITICAL: Deposit balance + ledger posted but transaction link failed. The reconciliation worker will repair it.",
   821→      });
   822→    });
   823→  }
   824→
   825→  await audit({
   826→    tenantId: input.tenantId,
   827→    userId: input.userId,
   828→    action: "reseller.balance_deposited",
   829→    entity: "tenant_balance",
   830→    entityId: input.tenantId,
   831→    detail: { amount: deposit.amountMinor, balanceAfter: result.balanceMinor, depositId: deposit.id, ledgerTxnId },
   832→  });
   833→  logger.info("reseller.balance_deposited", { tenantId: input.tenantId, amount: deposit.amountMinor, balance: result.balanceMinor, depositId: deposit.id });
   834→
   835→  return { status: finalStatus, balanceMinor: result.balanceMinor };
   836→}

   838→/**
   839→ * Process a deposit payment webhook (from the payment provider).
   840→ * Idempotent: if the deposit is already COMPLETED, returns without re-crediting.
   841→ */
   842→export async function handleDepositWebhook(input: {
   843→  providerReference: string;
   844→  status: "succeeded" | "failed" | "pending";
   845→}): Promise<{ handled: boolean }> {
   846→  const deposit = await db.tenantDepositPayment.findFirst({
   847→    where: { providerReference: input.providerReference },
   848→  });
   849→  if (!deposit) {
   850→    logger.warn("reseller.deposit_webhook_no_match", { providerReference: input.providerReference });
   851→    return { handled: false };
   852→  }
   853→
   854→  // Idempotent: if already completed, skip
   855→  if (deposit.status === "COMPLETED" || deposit.status === "BALANCE_POSTED") {
   856→    logger.info("reseller.deposit_webhook_idempotent", { depositId: deposit.id, status: deposit.status });
   857→    return { handled: true };
   858→  }
   859→
   860→  if (input.status === "succeeded") {
   861→    // Mark as PAYMENT_SUCCEEDED, then credit the balance
   862→    // Phase 2B.2.1 §7: Do NOT silently swallow the status update.
   863→    try {
   864→      await db.tenantDepositPayment.update({
   865→        where: { id: deposit.id, status: { notIn: ["COMPLETED", "BALANCE_POSTED", "PAYMENT_SUCCEEDED"] } },
   866→        data: { status: "PAYMENT_SUCCEEDED" },
   867→      });
   868→    } catch (err) {
   869→      const errMsg = err instanceof Error ? err.message : String(err);
   870→      logger.error("reseller.deposit_webhook_status_update_failed", {
   871→        depositId: deposit.id,
   872→        error: errMsg,
   873→        message: "CRITICAL: Webhook reported payment succeeded but status update failed. The reconciliation worker will recover it.",
   874→      });
   875→      return { handled: false };
   876→    }
   877→    // Credit the balance — if this fails, the deposit stays in PAYMENT_SUCCEEDED
   878→    // and the reconciliation worker will retry via processDueDepositReconciliation
   879→    // (which processes BALANCE_POSTED + RECONCILIATION_REQUIRED, and the
   880→    // processDueResellerReservationReconciliation stale scan catches
   881→    // PAYMENT_SUCCEEDED deposits that never got their balance credited).
   882→    try {
   883→      await creditDepositBalance({
   884→        depositPaymentId: deposit.id,
   885→        tenantId: deposit.tenantId,
   886→        userId: deposit.userId,
   887→      });
   888→    } catch (err) {
   889→      logger.error("reseller.deposit_webhook_credit_failed", {
   890→        depositId: deposit.id,
   891→        error: err instanceof Error ? err.message : String(err),
   892→        message: "CRITICAL: Webhook credited status but balance credit failed. The deposit is in PAYMENT_SUCCEEDED — the reconciliation worker will retry.",
   893→      });
   894→      // Mark as RECONCILIATION_REQUIRED so the worker picks it up
   895→      // Phase 2B.2.1 §7: Do NOT silently swallow this either — if the
   896→      // status update itself fails, log CRITICAL so an operator can intervene.
   897→      await db.tenantDepositPayment.updateMany({
   898→        where: { id: deposit.id, status: "PAYMENT_SUCCEEDED" },
   899→        data: { status: "RECONCILIATION_REQUIRED", failureReason: `Balance credit failed: ${err instanceof Error ? err.message : String(err)}` },
   900→      }).catch((updateErr) => {
   901→        const updateMsg = updateErr instanceof Error ? updateErr.message : String(updateErr);
   902→        logger.error("reseller.deposit_webhook_reconciliation_status_failed", {
   903→          depositId: deposit.id,
   904→          creditError: err instanceof Error ? err.message : String(err),
   905→          updateError: updateMsg,
   906→          message: "CRITICAL: Balance credit failed AND reconciliation status update failed. The deposit is stuck in PAYMENT_SUCCEEDED. The stale-status scan in processDueDepositReconciliation will recover it.",
   907→        });
   908→      });
   909→    }
   910→    return { handled: true };
   911→  }
   912→
   913→  if (input.status === "failed") {
   914→    // Phase 2B.2.1 §7: Do NOT silently swallow the status update.
   915→    try {
   916→      await db.tenantDepositPayment.update({
   917→        where: { id: deposit.id, status: { notIn: ["COMPLETED", "PAYMENT_FAILED"] } },
   918→        data: { status: "PAYMENT_FAILED", failureReason: "Webhook reported payment failed" },
   919→      });
   920→    } catch (err) {
   921→      const errMsg = err instanceof Error ? err.message : String(err);
   922→      logger.error("reseller.deposit_webhook_failed_status_update_error", {
   923→        depositId: deposit.id,
   924→        error: errMsg,
   925→        message: "CRITICAL: Webhook reported payment failed but status update failed. Manual review required.",
   926→      });
   927→      return { handled: false };
   928→    }
   929→    return { handled: true };
   930→  }
   931→
   932→  return { handled: false };
   933→}

   935→// ---------------------------------------------------------------------------
   936→// Reconciliation worker for stuck deposits (Phase 2B.2 §3)
   937→// ---------------------------------------------------------------------------
   938→
   939→/**
   940→ * Process deposits stuck in RECONCILIATION_REQUIRED or BALANCE_POSTED.
   941→ * Retries the ledger posting + status transition.
   942→ */
   943→export async function processDueDepositReconciliation(): Promise<{
   944→  retried: number;
   945→  repaired: number;
   946→  stillFailing: number;
   947→}> {
   948→  const result = { retried: 0, repaired: 0, stillFailing: 0 };
   949→
   950→  const due = await db.tenantDepositPayment.findMany({
   951→    where: { status: { in: ["RECONCILIATION_REQUIRED", "BALANCE_POSTED"] } },
   952→  });
   953→
   954→  for (const deposit of due) {
   955→    result.retried++;
   956→    try {
   957→      const ledgerTxnId = await ledgerResellerDeposit({
   958→        tenantId: deposit.tenantId,
   959→        userId: deposit.userId,
   960→        amountMinor: deposit.amountMinor,
   961→        idempotencyKey: `${deposit.idempotencyKey}:ledger`,
   962→      });
   963→
   964→      const updated = await db.tenantDepositPayment.updateMany({
   965→        where: { id: deposit.id, status: { in: ["RECONCILIATION_REQUIRED", "BALANCE_POSTED"] } },
   966→        data: { status: "COMPLETED", ledgerTransactionId: ledgerTxnId },
   967→      });
   968→
   969→      if (updated.count > 0) {
   970→        result.repaired++;
   971→        logger.info("reseller.deposit_reconciled", { depositId: deposit.id, ledgerTxnId });
   972→      }
   973→    } catch (err) {
   974→      result.stillFailing++;
   975→      logger.error("reseller.deposit_reconciliation_still_failing", { depositId: deposit.id, error: err instanceof Error ? err.message : String(err) });
   976→    }
   977→  }
   978→
   979→  return result;
   980→}

   982→// ---------------------------------------------------------------------------
   983→// Phase 2B.2.1 — Reservation reconciliation worker
   984→// Phase 2B.2.2: Safe reconciliation — NEVER settle based on age alone.
   985→// ---------------------------------------------------------------------------
   986→
   987→/**
   988→ * Process reservations that need reconciliation.
   989→ *
   990→ * Phase 2B.2.2: The worker NEVER infers fulfillment success from reservation
   991→ * age. It always consults the Order's authoritative fulfillment state before
   992→ * taking any financial action.
   993→ *
   994→ * Classification:
   995→ *   - RECONCILIATION_REQUIRED reservations:
   996→ *     → The order route already attempted settlement and the ledger failed.
   997→ *       The order's fulfillmentStatus is "success" (verified by the order route
   998→ *       before calling settleResellerReservation). So these are SETTLEMENT_ELIGIBLE.
   999→ *
  1000→ *   - Stale RESERVED reservations (older than threshold):
  1001→ *     → The order route may have crashed between reserve and settle.
  1002→ *       We must inspect the Order's fulfillmentStatus:
  1003→ *         "success"           → SETTLEMENT_ELIGIBLE (settle)
  1004→ *         "failed"            → RELEASE_ELIGIBLE (release)
  1005→ *         "pending"/"provisioning" → FULFILLMENT_PENDING (do nothing, retry later)
  1006→ *         "unknown"/"reconciliation_required" → FULFILLMENT_UNKNOWN (mark RECONCILIATION_REQUIRED, do not settle)
  1007→ *
  1008→ * Idempotency:
  1009→ *   - ledgerResellerPurchase replays if the idempotencyKey already exists
  1010→ *   - Status-guarded updateMany prevents duplicate transitions
  1011→ *   - TenantTransaction creation uses a unique idempotencyKey
  1012→ *   - Running this worker twice is safe — SETTLED/RELEASED records are skipped
  1013→ */
  1014→const STALE_RESERVED_THRESHOLD_MS = 5 * 60 * 1000; // 5 minutes
  1015→
  1016→type ReservationClassification =
  1017→  | "SETTLEMENT_ELIGIBLE"
  1018→  | "RELEASE_ELIGIBLE"
  1019→  | "FULFILLMENT_PENDING"
  1020→  | "FULFILLMENT_UNKNOWN";
  1021→
  1022→export async function processDueResellerReservationReconciliation(): Promise<{
  1023→  retried: number;
  1024→  repaired: number;
  1025→  released: number;
  1026→  pending: number;
  1027→  unknown: number;
  1028→  projectionRepaired: number;
  1029→  stillFailing: number;
  1030→}> {
  1031→  const result = { retried: 0, repaired: 0, released: 0, pending: 0, unknown: 0, projectionRepaired: 0, stillFailing: 0 };
  1032→
  1033→  // 1. Reservations explicitly in RECONCILIATION_REQUIRED (settlement was
  1034→  //    attempted but the ledger failed — the order route already verified
  1035→  //    fulfillment success before calling settleResellerReservation).
  1036→  // 2. Stale RESERVED reservations (older than threshold) — the order route
  1037→  //    may have crashed. We MUST inspect the Order's fulfillment state.
  1038→  const staleCutoff = new Date(Date.now() - STALE_RESERVED_THRESHOLD_MS);
  1039→  const due = await db.tenantBalanceReservation.findMany({
  1040→    where: {
  1041→      OR: [
  1042→        { state: "RECONCILIATION_REQUIRED" },
  1043→        { state: "RESERVED", updatedAt: { lt: staleCutoff } },
  1044→      ],
  1045→    },
  1046→    select: {
  1047→      id: true, tenantId: true, orderId: true, amountMinor: true,
  1048→      platformFeeMinor: true, idempotencyKey: true, state: true,
  1049→      ledgerTransactionId: true,
  1050→    },
  1051→  });
  1052→
  1053→  for (const reservation of due) {
  1054→    result.retried++;
  1055→
  1056→    // Phase 2B.2.3: ALWAYS inspect the Order's authoritative fulfillment state,
  1057→    // regardless of whether the reservation is RECONCILIATION_REQUIRED or stale RESERVED.
  1058→    // The reservation state alone NEVER proves fulfillment success — only the
  1059→    // Order's fulfillmentStatus does. This prevents the bug where a reservation
  1060→    // moved to RECONCILIATION_REQUIRED due to FULFILLMENT_UNKNOWN would be
  1061→    // incorrectly settled on the next worker run.
  1062→    let classification: ReservationClassification;
  1063→
  1064→    const order = await db.order.findUnique({
  1065→      where: { id: reservation.orderId },
  1066→      select: { fulfillmentStatus: true, status: true },
  1067→    });
  1068→
  1069→    if (!order) {
  1070→      // Order not found — can't determine fulfillment state. Fail closed.
  1071→      classification = "FULFILLMENT_UNKNOWN";
  1072→    } else {
  1073→      const fulfillmentStatus = order.fulfillmentStatus;
  1074→      if (fulfillmentStatus === "success") {
  1075→        classification = "SETTLEMENT_ELIGIBLE";
  1076→      } else if (fulfillmentStatus === "failed") {
  1077→        classification = "RELEASE_ELIGIBLE";
  1078→      } else if (fulfillmentStatus === "pending" || fulfillmentStatus === "provisioning") {
  1079→        classification = "FULFILLMENT_PENDING";
  1080→      } else {
  1081→        // "unknown" or "reconciliation_required" — can't determine. Fail closed.
  1082→        classification = "FULFILLMENT_UNKNOWN";
  1083→      }
  1084→    }
  1085→
  1086→    // Act on the classification
  1087→    if (classification === "FULFILLMENT_PENDING") {
  1088→      result.pending++;
  1089→      logger.info("reseller.reservation_fulfillment_pending", {
  1090→        reservationId: reservation.id,
  1091→        orderId: reservation.orderId,
  1092→      });
  1093→      continue; // do NOT settle, do NOT release
  1094→    }
  1095→
  1096→    if (classification === "FULFILLMENT_UNKNOWN") {
  1097→      result.unknown++;
  1098→      // Phase 2B.2.3: Transition to RECONCILIATION_REQUIRED with reason FULFILLMENT_UNKNOWN.
  1099→      // Do NOT settle, do NOT release. The reservation stays here until the Order's
  1100→      // fulfillmentStatus becomes "success" or "failed".
  1101→      // On the next worker run, the classification will re-check the Order's
  1102→      // fulfillmentStatus — if it's still unknown, it stays RECONCILIATION_REQUIRED.
  1103→      // If it becomes "success", it becomes SETTLEMENT_ELIGIBLE. If "failed", RELEASE_ELIGIBLE.
  1104→      await db.tenantBalanceReservation.updateMany({
  1105→        where: { id: reservation.id, state: { in: ["RESERVED", "RECONCILIATION_REQUIRED"] } },
  1106→        data: {
  1107→          state: "RECONCILIATION_REQUIRED",
  1108→          reconciliationReason: "FULFILLMENT_UNKNOWN",
  1109→          failureReason: "Fulfillment state unknown — manual review required",
  1110→        },
  1111→      }).catch((err) => {
  1112→        logger.error("reseller.reservation_unknown_classification_failed", {
  1113→          reservationId: reservation.id,
  1114→          error: err instanceof Error ? err.message : String(err),
  1115→        });
  1116→      });
  1117→      logger.warn("reseller.reservation_fulfillment_unknown", {
  1118→        reservationId: reservation.id,
  1119→        orderId: reservation.orderId,
  1120→        message: "Cannot determine fulfillment state — moved to RECONCILIATION_REQUIRED (FULFILLMENT_UNKNOWN). Will NOT settle until fulfillment is confirmed.",
  1121→      });
  1122→      continue;
  1123→    }
  1124→
  1125→    if (classification === "RELEASE_ELIGIBLE") {
  1126→      // Fulfillment failed — release the reservation (return funds).
  1127→      // We inline the release logic here (rather than calling releaseResellerReservation)
  1128→      // because the worker doesn't have a real userId for the audit log.
  1129→      try {
  1130→        const releaseResult = await db.$transaction(async (tx) => {
  1131→          const updated = await tx.tenantBalanceReservation.updateMany({
  1132→            where: { id: reservation.id, state: "RESERVED" },
  1133→            data: { state: "RELEASED", releasedAt: new Date(), failureReason: "Fulfillment failed (detected by reconciliation worker)" },
  1134→          });
  1135→          if (updated.count === 0) {
  1136→            return { alreadyDone: true as const };
  1137→          }
  1138→          // Phase 2B.2.7: Lock the balance row for concurrency-safe sequence allocation.
  1139→          await lockTenantBalance(tx, reservation.tenantId);
  1140→          const balance = await tx.tenantBalance.upsert({
  1141→            where: { tenantId: reservation.tenantId },
  1142→            create: { tenantId: reservation.tenantId, balanceMinor: reservation.amountMinor },
  1143→            update: { balanceMinor: { increment: reservation.amountMinor } },
  1144→          });
  1145→          const workerReleaseSeq = await getNextSequenceNumber(tx, reservation.tenantId);
  1146→          await tx.tenantTransaction.create({
  1147→            data: {
  1148→              tenantId: reservation.tenantId,
  1149→              type: "release",
  1150→              amountMinor: reservation.amountMinor,
  1151→              balanceAfter: balance.balanceMinor,
  1152→              orderId: reservation.orderId,
  1153→              description: "Reservation released by reconciliation worker (fulfillment failed)",
  1154→              idempotencyKey: `release_${reservation.idempotencyKey}`,
  1155→              sequenceNumber: workerReleaseSeq,
  1156→            },
  1157→          });
  1158→          return { alreadyDone: false as const, balanceMinor: balance.balanceMinor };
  1159→        }, { timeout: 30000, maxWait: 15000 });
  1160→
  1161→        if (!releaseResult.alreadyDone) {
  1162→          result.released++;
  1163→          logger.info("reseller.reservation_released_by_reconciliation", {
  1164→            reservationId: reservation.id,
  1165→            orderId: reservation.orderId,
  1166→            balanceMinor: releaseResult.balanceMinor,
  1167→          });
  1168→        }
  1169→      } catch (err) {
  1170→        result.stillFailing++;
  1171→        logger.error("reseller.reservation_release_failed", {
  1172→          reservationId: reservation.id,
  1173→          error: err instanceof Error ? err.message : String(err),
  1174→        });
  1175→      }
  1176→      continue;
  1177→    }
  1178→
  1179→    // SETTLEMENT_ELIGIBLE — retry the ledger posting (idempotent)
  1180→    try {
  1181→      const ledgerTxnId = await ledgerResellerPurchase({
  1182→        tenantId: reservation.tenantId,
  1183→        orderId: reservation.orderId,
  1184→        retailPriceMinor: reservation.amountMinor,
  1185→        platformFeeMinor: reservation.platformFeeMinor,
  1186→        idempotencyKey: `${reservation.idempotencyKey}:ledger`,
  1187→      });
  1188→
  1189→      // Transition to SETTLED (status-guarded)
  1190→      // Phase 2B.2.4: Clear BOTH failureReason AND reconciliationReason on successful settlement.
  1191→      const updated = await db.tenantBalanceReservation.updateMany({
  1192→        where: { id: reservation.id, state: { in: ["RESERVED", "RECONCILIATION_REQUIRED"] } },
  1193→        data: { state: "SETTLED", settledAt: new Date(), ledgerTransactionId: ledgerTxnId, failureReason: null, reconciliationReason: null },
  1194→      });
  1195→
  1196→      if (updated.count > 0) {
  1197→        result.repaired++;
  1198→        logger.info("reseller.reservation_reconciled", {
  1199→          reservationId: reservation.id,
  1200→          orderId: reservation.orderId,
  1201→          ledgerTxnId,
  1202→          wasStaleReserved: reservation.state === "RESERVED",
  1203→        });
  1204→
  1205→        // Create the TenantTransaction if it doesn't exist (idempotent via unique key)
  1206→        // Phase 2B.2.7: Use a transaction with FOR UPDATE lock for concurrency-safe sequence allocation.
  1207→        try {
  1208→          await db.$transaction(async (tx) => {
  1209→            await lockTenantBalance(tx, reservation.tenantId);
  1210→            const workerSettleSeq = await getNextSequenceNumber(tx, reservation.tenantId);
  1211→            const currentBalance = await getTenantBalanceMinorTx(tx, reservation.tenantId);
  1212→            await tx.tenantTransaction.create({
  1213→              data: {
  1214→                tenantId: reservation.tenantId,
  1215→                type: "purchase",
  1216→                amountMinor: -reservation.amountMinor,
  1217→                balanceAfter: currentBalance,
  1218→                orderId: reservation.orderId,
  1219→                description: "Connectivity purchase settled (reconciliation)",
  1220→                idempotencyKey: `settle_${reservation.idempotencyKey}`,
  1221→                ledgerTransactionId: ledgerTxnId,
  1222→                sequenceNumber: workerSettleSeq,
  1223→              },
  1224→            });
  1225→          }, { timeout: 30000, maxWait: 15000 });
  1226→
  1227→          // Phase 2B.2.5: Mark the projection as reconciled
  1228→          await db.tenantBalanceReservation.update({
  1229→            where: { id: reservation.id },
  1230→            data: { projectionReconciled: true },
  1231→          }).catch((err) => logProjectionUpdateFailure(reservation.id, "worker.settle.normal", err));
  1232→        } catch (txnErr) {
  1233→          const txnMsg = txnErr instanceof Error ? txnErr.message : String(txnErr);
  1234→          if (txnMsg.includes("Unique constraint") || txnMsg.includes("P2002")) {
  1235→            // Already exists — fine, mark as reconciled
  1236→            await db.tenantBalanceReservation.update({
  1237→              where: { id: reservation.id },
  1238→              data: { projectionReconciled: true },
  1239→            }).catch((err) => logProjectionUpdateFailure(reservation.id, "worker.settle.idempotent_replay", err));
  1240→          } else {
  1241→            logger.error("reseller.reservation_reconcile_txn_failed", {
  1242→              reservationId: reservation.id,
  1243→              orderId: reservation.orderId,
  1244→              ledgerTxnId,
  1245→              error: txnMsg,
  1246→              message: "CRITICAL: Reservation reconciled to SETTLED but TenantTransaction creation failed. The operational projection will be repaired on the next reconciliation run.",
  1247→            });
  1248→          }
  1249→        }
  1250→      } else {
  1251→        // Another concurrent call settled it — no-op
  1252→        logger.info("reseller.reservation_already_settled_during_reconciliation", { reservationId: reservation.id });
  1253→      }
  1254→    } catch (err) {
  1255→      result.stillFailing++;
  1256→      const errorMsg = err instanceof Error ? err.message : String(err);
  1257→      logger.error("reseller.reservation_reconciliation_still_failing", {
  1258→        reservationId: reservation.id,
  1259→        orderId: reservation.orderId,
  1260→        error: errorMsg,
  1261→      });
  1262→      // Leave as RECONCILIATION_REQUIRED for the next worker run
  1263→    }
  1264→  }
  1265→
  1266→  // Phase 2B.2.4: Repair SETTLED reservations that are missing their TenantTransaction.
  1267→  // This can happen if the TenantTransaction.create() call failed during settlement
  1268→  // (in either the order route or the reconciliation worker). The ledger is already
  1269→  // correct (ledgerTransactionId is set), so we only need to repair the operational
  1270→  // projection — we do NOT repost the ledger or change the balance.
  1271→  //
  1272→  // Phase 2B.2.5: Use the projectionReconciled flag to avoid scanning every historical
  1273→  // SETTLED reservation on every cron run. Only reservations with
  1274→  // projectionReconciled=false are checked. Once the TenantTransaction is confirmed
  1275→  // to exist (either it was already there or we just repaired it), the flag is set
  1276→  // to true and the worker skips it on future runs.
  1277→  //
  1278→  // Phase 2B.2.5: The repaired balanceAfter is the HISTORICAL balance immediately
  1279→  // after this transaction, NOT the current balance. We reconstruct it from the
  1280→  // ordered TenantTransaction history: find the transaction immediately before this
  1281→  // one (by createdAt), take its balanceAfter, add this transaction's amount.
  1282→  const settledReservations = await db.tenantBalanceReservation.findMany({
  1283→    where: {
  1284→      state: "SETTLED",
  1285→      ledgerTransactionId: { not: null },
  1286→      projectionReconciled: false,
  1287→    },
  1288→    select: {
  1289→      id: true, tenantId: true, orderId: true, amountMinor: true,
  1290→      idempotencyKey: true, ledgerTransactionId: true, createdAt: true,
  1291→    },
  1292→  });
  1293→
  1294→  for (const reservation of settledReservations) {
  1295→    // Check if the TenantTransaction already exists
  1296→    const expectedTxnKey = `settle_${reservation.idempotencyKey}`;
  1297→    const existingTxn = await db.tenantTransaction.findUnique({
  1298→      where: { idempotencyKey: expectedTxnKey },
  1299→    });
  1300→
  1301→    if (existingTxn) {
  1302→      // The TenantTransaction already exists — mark the projection as reconciled
  1303→      // so we don't scan this reservation again.
  1304→      await db.tenantBalanceReservation.update({
  1305→        where: { id: reservation.id },
  1306→        data: { projectionReconciled: true },
  1307→      }).catch((err) => logProjectionUpdateFailure(reservation.id, "worker.existing_txn_check", err));
  1308→      continue;
  1309→    }
  1310→
  1311→    // Phase 2B.2.6: Reconstruct the HISTORICAL balanceAfter using deterministic
  1312→    // sequenceNumber ordering (NOT createdAt, which can collide under concurrent
  1313→    // transactions). Find the TenantTransaction with the highest sequenceNumber
  1314→    // that is lower than the sequence this transaction would have had.
  1315→    //
  1316→    // Since the missing transaction was never created, we don't know its exact
  1317→    // sequenceNumber. But we can reconstruct the correct balanceAfter by finding
  1318→    // the transaction that was immediately before it in the historical sequence.
  1319→    // We use the reservation's createdAt as a rough filter to find the right
  1320→    // epoch, then use sequenceNumber for deterministic ordering within that epoch.
  1321→    //
  1322→    // The prior transaction's balanceAfter + this transaction's amount = the
  1323→    // correct historical balanceAfter for the repaired transaction.
  1324→    const priorTxn = await db.tenantTransaction.findFirst({
  1325→      where: {
  1326→        tenantId: reservation.tenantId,
  1327→        createdAt: { lt: reservation.createdAt },
  1328→      },
  1329→      orderBy: [{ sequenceNumber: "desc" }],
  1330→      select: { balanceAfter: true, sequenceNumber: true },
  1331→    });
  1332→
  1333→    const historicalBalanceBefore = priorTxn?.balanceAfter ?? 0;
  1334→    const transactionAmount = -reservation.amountMinor; // purchase is negative
  1335→    const historicalBalanceAfter = historicalBalanceBefore + transactionAmount;
  1336→
  1337→    // Phase 2B.2.7: Use a transaction with FOR UPDATE lock for concurrency-safe
  1338→    // sequence allocation. The repaired transaction gets the next available
  1339→    // sequenceNumber (at the end of the current history). This is correct
  1340→    // because the repaired transaction is being created NOW by the reconciliation
  1341→    // worker — its position in the operational history is the repair time, not
  1342→    // the original settlement time. The historicalBalanceAfter field preserves
  1343→    // the correct historical balance.
  1344→    try {
  1345→      await db.$transaction(async (tx) => {
  1346→        await lockTenantBalance(tx, reservation.tenantId);
  1347→        const repairSeq = await getNextSequenceNumber(tx, reservation.tenantId);
  1348→        await tx.tenantTransaction.create({
  1349→          data: {
  1350→            tenantId: reservation.tenantId,
  1351→            type: "purchase",
  1352→            amountMinor: transactionAmount,
  1353→            balanceAfter: historicalBalanceAfter,
  1354→            orderId: reservation.orderId,
  1355→            description: "Connectivity purchase settled (projection repair)",
  1356→            idempotencyKey: expectedTxnKey,
  1357→            ledgerTransactionId: reservation.ledgerTransactionId,
  1358→            sequenceNumber: repairSeq,
  1359→          },
  1360→        });
  1361→      }, { timeout: 30000, maxWait: 15000 });
  1362→
  1363→      // Mark the projection as reconciled so we don't scan this reservation again
  1364→      await db.tenantBalanceReservation.update({
  1365→        where: { id: reservation.id },
  1366→        data: { projectionReconciled: true },
  1367→      }).catch((err) => logProjectionUpdateFailure(reservation.id, "worker.repair.create", err));
  1368→
  1369→      result.projectionRepaired++;
  1370→      logger.info("reseller.projection_repaired", {
  1371→        reservationId: reservation.id,
  1372→        orderId: reservation.orderId,
  1373→        ledgerTxnId: reservation.ledgerTransactionId,
  1374→        historicalBalanceAfter,
  1375→      });
  1376→    } catch (txnErr) {
  1377→      const txnMsg = txnErr instanceof Error ? txnErr.message : String(txnErr);
  1378→      if (txnMsg.includes("Unique constraint") || txnMsg.includes("P2002")) {
  1379→        // Another concurrent call created it — mark as reconciled
  1380→        await db.tenantBalanceReservation.update({
  1381→          where: { id: reservation.id },
  1382→          data: { projectionReconciled: true },
  1383→        }).catch((err) => logProjectionUpdateFailure(reservation.id, "worker.repair.idempotent_replay", err));
  1384→        logger.info("reseller.projection_repair_idempotent_replay", { reservationId: reservation.id });
  1385→      } else {
  1386→        result.stillFailing++;
  1387→        logger.error("reseller.projection_repair_failed", {
  1388→          reservationId: reservation.id,
  1389→          orderId: reservation.orderId,
  1390→          ledgerTxnId: reservation.ledgerTransactionId,
  1391→          error: txnMsg,
  1392→          message: "CRITICAL: SETTLED reservation has a ledger transaction but TenantTransaction creation failed. The ledger is correct; the operational projection will be retried on the next reconciliation run.",
  1393→        });
  1394→      }
  1395→    }
  1396→  }
  1397→
  1398→  return result;
  1399→}

  1450→/**
  1451→ * @deprecated Phase 2B.2: Use reserveResellerBalance + settleResellerReservation
  1452→ * instead. This function is kept for backward compatibility with existing
  1453→ * tests but now uses the reservation lifecycle internally (reserve + settle).
  1454→ */
  1455→export async function debitResellerBalance(input: {
  1456→  tenantId: string;
  1457→  userId: string;
  1458→  orderId: string;
  1459→  amountMinor: number;
  1460→  platformFeeMinor: number;
  1461→  idempotencyKey: string;
  1462→  description?: string;
  1463→}): Promise<{ balanceMinor: number; transactionId: string; ledgerTransactionId: string }> {
  1464→  // Reserve the funds
  1465→  const reserveResult = await reserveResellerBalance({
  1466→    tenantId: input.tenantId,
  1467→    userId: input.userId,
  1468→    orderId: input.orderId,
  1469→    amountMinor: input.amountMinor,
  1470→    platformFeeMinor: input.platformFeeMinor,
